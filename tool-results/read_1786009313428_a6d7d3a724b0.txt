    48→    48→def get_next_token():
    49→    49→    """Round-robin token selection across all accounts.
    50→    50→    Auto-refreshes expired tokens. Skips known-exhausted accounts (from cache only)."""
    51→    51→    global _token_idx
    52→    52→    data = load_tokens()
    53→    53→    if not data["accounts"]:
    54→    54→        return None, None
    55→    55→
    56→    56→    n = len(data["accounts"])
    57→    57→    with _token_lock:
    58→    58→        idx = _token_idx % n
    59→    59→        _token_idx += 1
    60→    60→
    61→    61→    # Try each account starting from idx
    62→    62→    for i in range(n):
    63→    63→        acc = data["accounts"][(idx + i) % n]
    64→    64→        # Check token validity
    65→    65→        from config import ACCESS_TOKEN_TTL, REFRESH_MARGIN
    66→    66→        age = time.time() - acc.get("last_refreshed", 0)
    67→    67→        if age < ACCESS_TOKEN_TTL - REFRESH_MARGIN:
    68→    68→            # Token still valid — check exhausted cache (non-blocking, cached only)
    69→    69→            if not _is_cached_exhausted(acc):
    70→    70→                return acc["access_token"], acc
    71→    71→            else:
    72→    72→                continue
    73→    73→        # Try refresh
    74→    74→        from auth import refresh_token
    75→    75→        new_token = refresh_token(acc)
    76→    76→        if new_token:
    77→    77→            if not _is_cached_exhausted(acc):
    78→    78→                return new_token, acc
    79→    79→            else:
    80→    80→                continue
    81→    81→
    82→    82→    # Fallback: return first valid token even if exhausted
    83→    83→    for acc in data["accounts"]:
    84→    84→        return acc["access_token"], acc
    85→    85→    return None, None

    94→    94→def _is_cached_exhausted(acc):
    95→    95→    """Check if account is known to be exhausted (cache only, no API call)."""
    96→    96→    email = acc.get("email", "")
    97→    97→    now = time.time()
    98→    98→    cached = _exhausted_cache.get(email)
    99→    99→    if cached and (now - cached["checked"]) < _EXHAUSTED_CACHE_TTL:
   100→   100→        return cached["balance"] <= 0
   101→   101→    # No cache or expired — assume NOT exhausted (don't block chat)
   102→   102→    return False

   105→   105→def _refresh_exhausted_cache():
   106→   106→    """Background: check all accounts' wallet balances and update cache.
   107→   107→    Called periodically, not during chat requests."""
   108→   108→    data = load_tokens()
   109→   109→    for acc in data["accounts"]:
   110→   110→        email = acc.get("email", "")
   111→   111→        try:
   112→   112→            result = check_wallet(acc["access_token"])
   113→   113→            if result.get("code") == 0:
   114→   114→                balance = result["data"]["total_balance"]
   115→   115→                _exhausted_cache[email] = {"balance": balance, "checked": time.time()}
   116→   116→                if balance <= 0:
   117→   117→                    print(f"[proxy] Exhausted: {email} (0 pts)")
   118→   118→        except:
   119→   119→            pass

   149→   149→@app.route("/v1/chat/completions", methods=["POST"])
   150→   150→def chat_completions():
   151→   151→    """OpenAI-compatible chat completions proxy."""
   152→   152→    body = request.get_json(force=True)
   153→   153→    if not body:
   154→   154→        return jsonify({"error": {"message": "Invalid JSON body"}}), 400
   155→   155→
   156→   156→    # Model mapping
   157→   157→    client_model = body.get("model", DEFAULT_MODEL)
   158→   158→    upstream_model = MODEL_MAP.get(client_model, DEFAULT_MODEL)
   159→   159→
   160→   160→    # Force stream=True for upstream (DeepSeek models 500 on non-stream)
   161→   161→    upstream_body = dict(body)
   162→   162→    upstream_body["stream"] = True
   163→   163→    upstream_body["model"] = "x"  # ignored by upstream, but fill it
   164→   164→
   165→   165→    # Get token (round-robin)
   166→   166→    access_token, acc = get_next_token()
   167→   167→    if not access_token:
   168→   168→        return jsonify({
   169→   169→            "error": {
   170→   170→                "message": "No valid tokens. Add accounts first via /login or manually edit tokens.json",
   171→   171→                "type": "auth_error",
   172→   172→            }
   173→   173→        }), 401
   174→   174→
   175→   175→    # Increment request counter for this account
   176→   176→    used_email = acc.get("email", "unknown")
   177→   177→    _request_counts[used_email] = _request_counts.get(used_email, 0) + 1
   178→   178→
   179→   179→    headers = _sign_headers()
   180→   180→    # Token already has "Bearer " prefix — don't double it
   181→   181→    raw_token = access_token.replace("Bearer ", "")
   182→   182→    headers["X-Authorization"] = f"Bearer {raw_token}"
   183→   183→    headers["X-Request-Id"] = str(uuid.uuid4())
   184→   184→    headers["X-Request-Model"] = upstream_model
   185→   185→
   186→   186→    client_wants_stream = body.get("stream", False)
   187→   187→
   188→   188→    try:
   189→   189→        # Always request stream from upstream
   190→   190→        upstream_resp = req_lib.post(
   191→   191→            CHAT_COMPLETIONS,
   192→   192→            json=upstream_body,
   193→   193→            headers=headers,
   194→   194→            stream=True,
   195→   195→            timeout=600,
   196→   196→            verify=False,
   197→   197→        )
   198→   198→
   199→   199→        if upstream_resp.status_code != 200:
   200→   200→            error_text = upstream_resp.text[:1000]
   201→   201→            return jsonify({
   202→   202→                "error": {
   203→   203→                    "message": f"Upstream error {upstream_resp.status_code}: {error_text}",
   204→   204→                    "type": "upstream_error",
   205→   205→                    "code": upstream_resp.status_code,
   206→   206→                }
   207→   207→            }), upstream_resp.status_code
   208→   208→
   209→   209→        if client_wants_stream:
   210→   210→            # Pass through SSE stream — filter reasoning, only forward content chunks
   211→   211→            def generate():
   212→   212→                for line in upstream_resp.iter_lines():
   213→   213→                    if line:
   214→   214→                        if line.startswith(b"data:"):
   215→   215→                            raw = line[5:].strip()
   216→   216→                            if raw == b"[DONE]":
   217→   217→                                yield line + b"\n\n"
   218→   218→                                continue
   219→   219→                            try:
   220→   220→                                chunk = json.loads(raw)
   221→   221→                                choices = chunk.get("choices", [])
   222→   222→                                if choices:
   223→   223→                                    delta = choices[0].get("delta", {})
   224→   224→                                    # Strip reasoning fields — Cline/OpenAI clients don't understand them
   225→   225→                                    if "reasoning" in delta:
   226→   226→                                        del delta["reasoning"]
   227→   227→                                    if "reasoning_details" in delta:
   228→   228→                                        del delta["reasoning_details"]
   229→   229→                                    if "reasoning_content" in delta:
   230→   230→                                        del delta["reasoning_content"]
   231→   231→                                    # Skip chunks with empty content and no tool_calls (pure thinking)
   232→   232→                                    has_content = bool(delta.get("content"))
   233→   233→                                    has_tool_calls = bool(delta.get("tool_calls"))
   234→   234→                                    has_finish = bool(choices[0].get("finish_reason"))
   235→   235→                                    # Only forward if there's actual content/tool_calls/finish
   236→   236→                                    if not has_content and not has_tool_calls and not has_finish:
   237→   237→                                        continue
   238→   238→                                    # Clean native_finish_reason
   239→   239→                                    if "native_finish_reason" in choices[0]:
   240→   240→                                        del choices[0]["native_finish_reason"]
   241→   241→                                yield b"data: " + json.dumps(chunk).encode() + b"\n\n"
   242→   242→                            except (json.JSONDecodeError, KeyError):
   243→   243→                                yield line + b"\n\n"
   244→   244→                yield b"data: [DONE]\n\n"
   245→   245→            return Response(
   246→   246→                generate(),
   247→   247→                mimetype="text/event-stream",
   248→   248→                headers={
   249→   249→                    "Cache-Control": "no-cache",
   250→   250→                    "X-Accel-Buffering": "no",
   251→   251→                },
   252→   252→            )
   253→   253→        else:
   254→   254→            # Aggregate stream → single JSON response (OpenAI non-stream format)
   255→   255→            full_content = ""
   256→   256→            full_tool_calls = []
   257→   257→            finish_reason = None
   258→   258→            model_name = upstream_model
   259→   259→
   260→   260→            for line in upstream_resp.iter_lines():
   261→   261→                if not line:
   262→   262→                    continue
   263→   263→                line_str = line.decode("utf-8", errors="replace")
   264→   264→                if line_str.startswith("data: "):
   265→   265→                    data_str = line_str[6:]
   266→   266→                    if data_str.strip() == "[DONE]":
   267→   267→                        break
   268→   268→                    try:
   269→   269→                        chunk = json.loads(data_str)
   270→   270→                        choices = chunk.get("choices", [])
   271→   271→                        if choices:
   272→   272→                            delta = choices[0].get("delta", {})
   273→   273→                            if "content" in delta and delta["content"]:
   274→   274→                                full_content += delta["content"]
   275→   275→                            if "tool_calls" in delta and delta["tool_calls"]:
   276→   276→                                for tc in delta["tool_calls"]:
   277→   277→                                    idx = tc.get("index", 0)
   278→   278→                                    while len(full_tool_calls) <= idx:
   279→   279→                                        full_tool_calls.append({"id": "", "type": "function", "function": {"name": "", "arguments": ""}})
   280→   280→                                    if "id" in tc and tc["id"]:
   281→   281→                                        full_tool_calls[idx]["id"] = tc["id"]
   282→   282→                                    if "type" in tc and tc["type"]:
   283→   283→                                        full_tool_calls[idx]["type"] = tc["type"]
   284→   284→                                    if "function" in tc:
   285→   285→                                        fn = tc["function"]
   286→   286→                                        if "name" in fn and fn["name"]:
   287→   287→                                            full_tool_calls[idx]["function"]["name"] += fn["name"]
   288→   288→                                        if "arguments" in fn and fn["arguments"]:
   289→   289→                                            full_tool_calls[idx]["function"]["arguments"] += fn["arguments"]
   290→   290→                            if choices[0].get("finish_reason"):
   291→   291→                                finish_reason = choices[0]["finish_reason"]
   292→   292→                        if chunk.get("model"):
   293→   293→                            model_name = chunk["model"]
   294→   294→                        if chunk.get("usage"):
   295→   295→                            usage = chunk["usage"]
   296→   296→                    except json.JSONDecodeError:
   297→   297→                        pass
   298→   298→
   299→   299→            message = {"role": "assistant", "content": full_content if full_content else None}
   300→   300→            if full_tool_calls:
   301→   301→                message["tool_calls"] = full_tool_calls
   302→   302→                if not finish_reason:
   303→   303→                    finish_reason = "tool_calls"
   304→   304→
   305→   305→            response = {
   306→   306→                "id": f"chatcmpl-{uuid.uuid4().hex[:12]}",
   307→   307→                "object": "chat.completion",
   308→   308→                "created": int(time.time()),
   309→   309→                "model": client_model,
   310→   310→                "choices": [{
   311→   311→                    "index": 0,
   312→   312→                    "message": message,
   313→   313→                    "finish_reason": finish_reason or "stop",
   314→   314→                }],
   315→   315→                "usage": usage if 'usage' in dir() else {"prompt_tokens": 0, "completion_tokens": 0, "total_tokens": 0},
   316→   316→            }
   317→   317→            return jsonify(response)
   318→   318→
   319→   319→    except req_lib.exceptions.Timeout:
   320→   320→        return jsonify({"error": {"message": "Upstream timeout"}}), 504
   321→   321→    except Exception as e:
   322→   322→        return jsonify({"error": {"message": f"Proxy error: {str(e)}"}}), 500

   368→   368→def accounts_detail():
   369→   369→    """List all accounts with wallet balance + token expiry (bulk)."""
   370→   370→    import base64 as _b64
   371→   371→    data = load_tokens()
   372→   372→    result = []
   373→   373→    for acc in data["accounts"]:
   374→   374→        # Decode JWT for expiry
   375→   375→        exp = 0
   376→   376→        iat = 0
   377→   377→        try:
   378→   378→            token = acc["access_token"].replace("Bearer ", "")
   379→   379→            parts = token.split(".")
   380→   380→            payload = parts[1] + "=" * (4 - len(parts[1]) % 4)
   381→   381→            decoded = json.loads(_b64.urlsafe_b64decode(payload))
   382→   382→            exp = decoded.get("exp", 0)
   383→   383→            iat = decoded.get("iat", 0)
   384→   384→        except:
   385→   385→            pass
   386→   386→
   387→   387→        now = time.time()
   388→   388→        remaining_h = max(0, (exp - now) / 3600) if exp else 0
   389→   389→        expired = remaining_h <= 0
   390→   390→
   391→   391→        result.append({
   392→   392→            "email": acc["email"],
   393→   393→            "user_id": acc.get("user_id", ""),
   394→   394→            "device_id": acc.get("device_id", ""),
   395→   395→            "added_at": acc.get("added_at", 0),
   396→   396→            "last_refreshed": acc.get("last_refreshed", 0),
   397→   397→            "expires_at": exp,
   398→   398→            "remaining_hours": round(remaining_h, 1),
   399→   399→            "expired": expired,
   400→   400→            "has_refresh_token": bool(acc.get("refresh_token")),
   401→   401→            "request_count": _request_counts.get(acc.get("email", ""), 0),
   402→   402→        })
   403→   403→    return jsonify({"accounts": result, "total": len(result)})

   469→   469→_refresh_progress = {"total": 0, "done": 0, "success": 0, "fail": 0, "running": False}
   470→   470→
   471→   471→
   472→   472→@app.route("/refresh-all", methods=["POST"])
   473→   473→def do_refresh_all():
   474→   474→    # If already running, return current progress
   475→   475→    if _refresh_progress["running"]:
   476→   476→        return jsonify({"status": "running", **_refresh_progress})
   477→   477→
   478→   478→    # Backup tokens.json before refresh (preventive)
   479→   479→    import shutil
   480→   480→    try:
   481→   481→        shutil.copy2(TOKENS_FILE_FULL, TOKENS_FILE_FULL + ".bak")
   482→   482→    except Exception as e:
   483→   483→        print(f"[proxy] Backup before refresh-all failed: {e}")
   484→   484→
   485→   485→    # Start refresh in background thread
   486→   486→    data = load_tokens()
   487→   487→    _refresh_progress["total"] = len(data["accounts"])
   488→   488→    _refresh_progress["done"] = 0
   489→   489→    _refresh_progress["success"] = 0
   490→   490→    _refresh_progress["fail"] = 0
   491→   491→    _refresh_progress["running"] = True
   492→   492→
   493→   493→    def _bg_refresh():
   494→   494→        import auth as _auth
   495→   495→        from config import REFRESH_URL
   496→   496→        import requests as _req
   497→   497→        d = _auth.load_tokens()
   498→   498→        now = int(time.time())
   499→   499→        for acc in d["accounts"]:
   500→   500→            try:
   501→   501→                headers = _auth._sign_headers()
   502→   502→                body = {
   503→   503→                    "source_id": acc.get("source_id", "autoclaw"),
   504→   504→                    "device_id": acc["device_id"],
   505→   505→                    "refresh_token": acc["refresh_token"],
   506→   506→                }
   507→   507→                resp = _req.post(REFRESH_URL, json=body, headers=headers, timeout=15, verify=False)
   508→   508→                resp_data = resp.json()
   509→   509→                if resp_data.get("code") == 0 and "data" in resp_data:
   510→   510→                    new_access = resp_data["data"].get("access_token")
   511→   511→                    new_refresh = resp_data["data"].get("refresh_token", acc["refresh_token"])
   512→   512→                    if new_access:
   513→   513→                        acc["access_token"] = new_access
   514→   514→                        if new_refresh:
   515→   515→                            acc["refresh_token"] = new_refresh
   516→   516→                        acc["last_refreshed"] = now
   517→   517→                        _refresh_progress["success"] += 1
   518→   518→                    else:
   519→   519→                        _refresh_progress["fail"] += 1
   520→   520→                else:
   521→   521→                    _refresh_progress["fail"] += 1
   522→   522→            except Exception:
   523→   523→                _refresh_progress["fail"] += 1
   524→   524→            _refresh_progress["done"] += 1
   525→   525→        # Save once at end
   526→   526→        _auth.save_tokens(d)
   527→   527→        _refresh_progress["running"] = False
   528→   528→        print(f"[proxy] Refresh all done: {_refresh_progress['success']} ok, {_refresh_progress['fail']} fail")
   529→   529→
   530→   530→    t = threading.Thread(target=_bg_refresh, daemon=True)
   531→   531→    t.start()
   532→   532→
   533→   533→    return jsonify({"status": "started", **_refresh_progress})

   601→   601→@app.route("/auth/callback-google")
   602→   602→def auth_callback_google():
   603→   603→    """Google OAuth redirect target — auto-captures code+state, exchanges tokens."""
   604→   604→    from auth import google_oauth_login, add_token
   605→   605→
   606→   606→    code = request.args.get("code")
   607→   607→    state = request.args.get("state")
   608→   608→    error = request.args.get("error")
   609→   609→
   610→   610→    # Find pending login by state — MUST match, no fallback in concurrent mode
   611→   611→    pending = _pending_logins.get(state)
   612→   612→    if not pending:
   613→   613→        # No state match — do NOT fallback to "first pending" (causes cross-account
   614→   614→        # contamination in concurrent mode). Just fail this callback.
   615→   615→        print(f"[callback] Unknown state={state[:20]}... (not in {len(_pending_logins)} pending)")
   616→   616→        return "<html><body><h1>Login Failed</h1><p>Unknown state</p></body></html>", 400
   617→   617→
   618→   618→    if error:
   619→   619→        pending["error"] = error
   620→   620→        return f"<html><body><h1>Login Failed</h1><p>{error}</p></body></html>", 400
   621→   621→
   622→   622→    if not code:
   623→   623→        pending["error"] = "No code in callback"
   624→   624→        return "<html><body><h1>Login Failed</h1><p>No code</p></body></html>", 400
   625→   625→
   626→   626→    device_id = pending.get("device_id")
   627→   627→    proxy_used = pending.get("proxy")  # Reuse same proxy from URL generation
   628→   628→
   629→   629→    # Serialize token exchanges — AutoClaw API rate-limits per IP (630014).
   630→   630→    # Without this, 3 concurrent callbacks all hit the API simultaneously → all fail.
   631→   631→    # Lock ensures 1 exchange at a time, with small gap between.
   632→   632→    with _token_exchange_lock:
   633→   633→        print(f"[callback] Exchanging token for state={state[:12]}... device_id={device_id[:8]}... proxy={'yes' if proxy_used else 'none'}")
   634→   634→        result = google_oauth_login(code, state, device_id, proxy=proxy_used)
   635→   635→        time.sleep(2)  # Gap so next concurrent callback doesn't hit 630014
   636→   636→    if not result:
   637→   637→        print(f"[callback] Token exchange FAILED for state={state[:12]}... (google_oauth_login returned None)")
   638→   638→        pending["error"] = "Token exchange failed"
   639→   639→        return "<html><body><h1>Login Failed</h1><p>Token exchange failed</p></body></html>", 500
   640→   640→    print(f"[callback] Token exchange OK for state={state[:12]}... user_id={result.get('user_id')}")
   641→   641→
   642→   642→    # Extract email from JWT jti field
   643→   643→    import base64 as _b64
   644→   644→    email = result.get("user_name") or f"user_{result['user_id']}"
   645→   645→    try:
   646→   646→        token_part = result["access_token"].replace("Bearer ", "").split(".")[1]
   647→   647→        token_part += "=" * (4 - len(token_part) % 4)
   648→   648→        payload = json.loads(_b64.urlsafe_b64decode(token_part))
   649→   649→        if payload.get("jti"):
   650→   650→            email = payload["jti"]
   651→   651→    except:
   652→   652→        pass
   653→   653→
   654→   654→    add_token(
   655→   655→        email=email,
   656→   656→        access_token=result["access_token"],
   657→   657→        refresh_token=result["refresh_token"],
   658→   658→        user_id=result["user_id"],
   659→   659→        device_id=device_id,
   660→   660→    )
   661→   661→    pending["result"] = {
   662→   662→        "email": email,
   663→   663→        "user_id": result.get("user_id"),
   664→   664→        "first_login": result.get("first_login"),
   665→   665→    }
   666→   666→    return f"""<html><body>
   667→   667→<h1>Login Success!</h1>
   668→   668→<p>Email: {email}</p>
   669→   669→<p>User ID: {result.get('user_id')}</p>
   670→   670→<p>You can close this tab.</p>
   671→   671→</body></html>"""
