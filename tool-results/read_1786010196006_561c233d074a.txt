   269→// ── Load accounts + wallet ──
   270→async function loadAccounts() {
   271→  try {
   272→    // 1. Fetch accounts-detail first (fast — no API calls)
   273→    const detailResp = await fetch(API + '/api/accounts-detail').then(r => r.json());
   274→    const wasFirstLoad = accountsData.length === 0;
   275→    accountsData = detailResp.accounts || [];
   276→
   277→    const grid = document.getElementById('cards-grid');
   278→    document.getElementById('account-count').textContent = accountsData.length;
   279→
   280→    if (accountsData.length === 0) {
   281→      grid.innerHTML = '<div class="empty-state">No accounts. Use autoclaw_autologin.py to add accounts.</div>';
   282→      document.getElementById('credits-bar').style.display = 'none';
   283→      document.getElementById('pagination').style.display = 'none';
   284→      return;
   285→    }
   286→
   287→    // 2. Show credits bar immediately (starts at 0 if first load)
   288→    if (wasFirstLoad) {
   289→      document.getElementById('credits-bar').style.display = 'inline-flex';
   290→      document.getElementById('total-points').textContent = '0';
   291→      document.getElementById('total-max').textContent = '0';
   292→      // Clear wallet cache on first load
   293→      walletCache = {};
   294→    }
   295→
   296→    // 3. Render current page
   297→    renderPage();
   298→
   299→    // 4. Update totals from existing cache (instant)
   300→    updateTotals();
   301→
   302→    // 5. Fetch ALL wallets in background (limited concurrency) — credits bar counts up live
   303→    fetchAllWalletsBackground();
   304→
   305→  } catch (e) {
   306→    document.getElementById('cards-grid').innerHTML =
   307→      `<div class="empty-state" style="color:var(--red);">Failed to load: ${e.message}</div>`;
   308→  }
   309→}

   311→// ── Fetch all wallets in background with limited concurrency ──
   312→let _walletFetchInFlight = false;
   313→let _walletFetchAbort = false;
   314→
   315→async function fetchAllWalletsBackground() {
   316→  // Prevent overlapping fetches
   317→  if (_walletFetchInFlight) {
   318→    _walletFetchAbort = true;
   319→    // Wait a bit for previous to notice abort
   320→    await new Promise(r => setTimeout(r, 200));
   321→  }
   322→  _walletFetchInFlight = true;
   323→  _walletFetchAbort = false;
   324→
   325→  const toFetch = accountsData.filter(acc => {
   326→    const w = walletCache[acc.email];
   327→    return !w || w.balance == null;
   328→  });
   329→
   330→  if (toFetch.length === 0) {
   331→    _walletFetchInFlight = false;
   332→    return;
   333→  }
   334→
   335→  // Process in small batches to avoid overwhelming the server
   336→  const BATCH = 5;
   337→  for (let i = 0; i < toFetch.length; i += BATCH) {
   338→    if (_walletFetchAbort) break;
   339→
   340→    const batch = toFetch.slice(i, i + BATCH);
   341→    const promises = batch.map(async (acc) => {
   342→      if (_walletFetchAbort) return;
   343→      try {
   344→        const resp = await fetch(API + '/api/wallet/' + encodeURIComponent(acc.email));
   345→        const data = await resp.json();
   346→        if (data.code === 0 && data.data) {
   347→          walletCache[acc.email] = { email: acc.email, balance: data.data.total_balance, status: data.data.total_balance > 0 ? 'active' : 'exhausted' };
   348→        } else {
   349→          walletCache[acc.email] = { email: acc.email, balance: 0, status: 'error' };
   350→        }
   351→      } catch (e) {
   352→        walletCache[acc.email] = { email: acc.email, balance: 0, status: 'error' };
   353→      }
   354→      // Update totals after each wallet completes — credits bar counts up live
   355→      updateTotals();
   356→    });
   357→
   358→    await Promise.all(promises);
   359→  }
   360→
   361→  _walletFetchInFlight = false;
   362→}

   364→// ── Render single page of cards ──
   365→function renderPage() {
   366→  const grid = document.getElementById('cards-grid');
   367→  const total = accountsData.length;
   368→  const totalPages = Math.ceil(total / perPage);
   369→  if (currentPage > totalPages) currentPage = totalPages;
   370→  if (currentPage < 1) currentPage = 1;
   371→
   372→  const start = (currentPage - 1) * perPage;
   373→  const end = Math.min(start + perPage, total);
   374→  const pageAccounts = accountsData.slice(start, end);
   375→
   376→  // Render cards for this page
   377→  grid.innerHTML = '';
   378→  for (let i = 0; i < pageAccounts.length; i++) {
   379→    const acc = pageAccounts[i];
   380→    const cardId = 'card-' + acc.email.replace(/[^a-zA-Z0-9]/g, '');
   381→    const emailShort = acc.email.length > 32 ? acc.email.substring(0, 29) + '...' : acc.email;
   382→
   383→    // Token status
   384→    let tokenBadge, tokenDot, tokenBadgeClass;
   385→    if (acc.expired) {
   386→      tokenBadge = 'disabled'; tokenDot = 'dot-red'; tokenBadgeClass = 'badge-disabled';
   387→    } else if (acc.remaining_hours < 2) {
   388→      tokenBadge = 'expiring'; tokenDot = 'dot-yellow'; tokenBadgeClass = 'badge-expiring';
   389→    } else {
   390→      tokenBadge = 'active'; tokenDot = 'dot-green'; tokenBadgeClass = 'badge-active';
   391→    }
   392→
   393→    const reqCount = (acc.request_count || 0) + ' req';
   394→
   395→    // Check if wallet already cached
   396→    const w = walletCache[acc.email];
   397→    const balance = w ? w.balance : null;
   398→    const isExhausted = balance !== null && balance <= 0;
   399→    const maxPoints = 2300;
   400→    const pct = balance != null ? Math.min(100, (balance / maxPoints) * 100) : 0;
   401→
   402→    let walletBadgeClass = isExhausted ? 'badge-disabled' : 'badge-active';
   403→    let fillClass = 'fill-blue';
   404→    if (balance != null) {
   405→      if (balance <= 0) fillClass = 'fill-red';
   406→      else if (balance < 500) fillClass = 'fill-yellow';
   407→      else fillClass = 'fill-blue';
   408→    }
   409→
   410→    const walletText = balance != null ? `${balance.toLocaleString()} / ${maxPoints.toLocaleString()}` : 'Loading...';
   411→    const cardOpacity = isExhausted ? '0.5' : '1';
   412→
   413→    // Card
   414→    const card = document.createElement('div');
   415→    card.className = 'account-card';
   416→    card.id = cardId;
   417→    card.style.opacity = '0';
   418→    card.style.transform = 'translateY(8px)';
   419→    card.style.transition = 'opacity 0.3s, transform 0.3s';
   420→    card.innerHTML = `
   421→      <div class="card-row1">
   422→        <div class="card-name">
   423→          <span class="status-dot ${tokenDot}"></span>
   424→          <span>${emailShort}</span>
   425→        </div>
   426→        <div class="card-actions">
   427→          <button class="btn-icon" onclick="refreshToken('${acc.email}', this)" title="Refresh token">${ICONS.refresh}</button>
   428→          <button class="btn-icon" onclick="checkWalletSingle('${acc.email}', this)" title="Check wallet">${ICONS.wallet}</button>
   429→          <button class="btn-icon danger" onclick="deleteAccount('${acc.email}', this)" title="Delete">${ICONS.trash}</button>
   430→        </div>
   431→      </div>
   432→      <div class="card-row2">
   433→        <span class="badge ${tokenBadgeClass}">${tokenBadge}</span>
   434→        <span class="req-count">${reqCount}</span>
   435→        <span class="pill">Reward Points</span>
   436→      </div>
   437→      <div class="card-row3">
   438→        <span class="label">Points</span>
   439→        <span class="value"><span class="badge ${walletBadgeClass}" id="wallet-${cardId}">${walletText}</span></span>
   440→      </div>
   441→      <div class="quota-bar">
   442→        <div class="quota-fill ${fillClass}" id="bar-${cardId}" style="width: ${pct}%"></div>
   443→      </div>
   444→    `;
   445→    grid.appendChild(card);
   446→
   447→    // Animate in
   448→    requestAnimationFrame(() => {
   449→      card.style.opacity = cardOpacity;
   450→      card.style.transform = 'translateY(0)';
   451→    });
   452→
   453→    // Fetch wallet if not cached
   454→    if (balance === null) {
   455→      fetchWalletForCard(acc.email, cardId);
   456→    }
   457→  }
   458→
   459→  // Update pagination controls
   460→  const pagDiv = document.getElementById('pagination');
   461→  if (total > perPage) {
   462→    pagDiv.style.display = 'flex';
   463→    document.getElementById('page-info').textContent = `Page ${currentPage} / ${totalPages} (${total} accounts)`;
   464→    document.getElementById('prev-btn').disabled = currentPage <= 1;
   465→    document.getElementById('next-btn').disabled = currentPage >= totalPages;
   466→  } else {
   467→    pagDiv.style.display = 'none';
   468→  }
   469→}

   484→// ── Fetch wallet for single card (called from renderPage if cache miss) ──
   485→// Note: background fetchAllWalletsBackground handles the actual API calls.
   486→// This function just polls cache and updates the card when data arrives.
   487→async function fetchWalletForCard(email, cardId) {
   488→  const walletEl = document.getElementById('wallet-' + cardId);
   489→  const barEl = document.getElementById('bar-' + cardId);
   490→  if (!walletEl) return;
   491→
   492→  // Wait for background fetch to populate cache (poll up to 30s)
   493→  for (let i = 0; i < 60; i++) {
   494→    const w = walletCache[email];
   495→    if (w && w.balance != null) {
   496→      const balance = w.balance;
   497→      const maxPoints = 2300;
   498→      const pct = Math.min(100, (balance / maxPoints) * 100);
   499→      const isExhausted = balance <= 0;
   500→
   501→      let walletBadgeClass = isExhausted ? 'badge-disabled' : 'badge-active';
   502→      let fillClass = 'fill-blue';
   503→      if (isExhausted) fillClass = 'fill-red';
   504→      else if (balance < 500) fillClass = 'fill-yellow';
   505→
   506→      walletEl.className = 'badge ' + walletBadgeClass;
   507→      walletEl.textContent = `${balance.toLocaleString()} / ${maxPoints.toLocaleString()}`;
   508→
   509→      if (barEl) {
   510→        barEl.className = 'quota-fill ' + fillClass;
   511→        barEl.style.width = pct + '%';
   512→      }
   513→
   514→      if (isExhausted) {
   515→        const card = document.getElementById(cardId);
   516→        if (card) card.style.opacity = '0.5';
   517→      }
   518→      return;
   519→    }
   520→    await new Promise(r => setTimeout(r, 500));
   521→  }
   522→
   523→  // Timeout — show dash
   524→  walletEl.textContent = '—';
   525→  walletEl.className = 'badge badge-disabled';
   526→}

   528→// ── Update totals bar (incremental, called per-wallet) ──
   529→function updateTotals() {
   530→  let totalPoints = 0, totalMax = 0, loaded = 0;
   531→  const target = accountsData.length;
   532→  for (const acc of accountsData) {
   533→    const w = walletCache[acc.email];
   534→    if (w && w.balance != null) {
   535→      totalPoints += w.balance;
   536→      totalMax += 2300;
   537→      loaded++;
   538→    }
   539→  }
   540→  const bar = document.getElementById('credits-bar');
   541→  if (totalMax > 0 || loaded > 0 || target > 0) {
   542→    bar.style.display = 'inline-flex';
   543→    // Animate the number — smooth count up
   544→    const pointsEl = document.getElementById('total-points');
   545→    const maxEl = document.getElementById('total-max');
   546→    const progEl = document.getElementById('credits-progress');
   547→    const current = parseInt(pointsEl.textContent.replace(/[^0-9]/g, '')) || 0;
   548→    const target_val = totalPoints;
   549→
   550→    // Show loading progress
   551→    if (loaded < target) {
   552→      progEl.textContent = `(${loaded}/${target})`;
   553→    } else {
   554→      progEl.textContent = '';
   555→    }
   556→
   557→    // Smooth count animation — only start if not already animating
   558→    if (current !== target_val && pointsEl.dataset.animating !== '1') {
   559→      pointsEl.dataset.animating = '1';
   560→      requestAnimationFrame(function tick() {
   561→        const cur = parseInt(pointsEl.textContent.replace(/[^0-9]/g, '')) || 0;
   562→        const d = target_val - cur;
   563→        if (d === 0) {
   564→          pointsEl.textContent = target_val.toLocaleString();
   565→          pointsEl.dataset.animating = '';
   566→          return;
   567→        }
   568→        // Adaptive step: bigger jumps for large diffs, small for precision
   569→        const s = Math.max(1, Math.ceil(Math.abs(d) / 15));
   570→        if (Math.abs(d) <= s) {
   571→          pointsEl.textContent = target_val.toLocaleString();
   572→          pointsEl.dataset.animating = '';
   573→        } else {
   574→          pointsEl.textContent = (cur + Math.sign(d) * s).toLocaleString();
   575→          requestAnimationFrame(tick);
   576→        }
   577→      });
   578→    }
   579→    maxEl.textContent = totalMax.toLocaleString();
   580→    // Update dot color based on progress
   581→    const dot = bar.querySelector('.credits-dot');
   582→    if (dot) {
   583→      const pct = totalMax > 0 ? (totalPoints / totalMax) * 100 : 0;
   584→      dot.style.background = pct > 50 ? 'var(--green)' : pct > 20 ? 'var(--yellow)' : 'var(--red)';
   585→    }
   586→  } else {
   587→    bar.style.display = 'none';
   588→  }
   589→}

   609→// ── Refresh all tokens (with live counter) ──
   610→let _refreshAllPolling = null;
   611→
   612→async function refreshAllTokens() {
   613→  // Disable button, show spinner
   614→  const btn = event.target.closest('button');
   615→  if (btn) { btn.classList.add('spinning'); btn.disabled = true; }
   616→
   617→  try {
   618→    const resp = await fetch(API + '/refresh-all', { method: 'POST' });
   619→    const data = await resp.json();
   620→
   621→    if (data.status === 'running') {
   622→      // Already running — start polling
   623→      showToast('Refresh already in progress...', 'success');
   624→    } else if (data.status === 'started') {
   625→      showToast(`Refreshing ${data.total} tokens...`, 'success');
   626→    }
   627→
   628→    // Start polling progress
   629→    if (_refreshAllPolling) clearInterval(_refreshAllPolling);
   630→    _refreshAllPolling = setInterval(async () => {
   631→      try {
   632→        const pr = await fetch(API + '/api/refresh-progress').then(r => r.json());
   633→        const pct = pr.total > 0 ? Math.round((pr.done / pr.total) * 100) : 0;
   634→
   635→        // Update button text to show counter
   636→        if (btn) {
   637→          btn.innerHTML = btn.querySelector('svg').outerHTML + ` ${pr.done}/${pr.total}`;
   638→        }
   639→
   640→        if (!pr.running) {
   641→          clearInterval(_refreshAllPolling);
   642→          _refreshAllPolling = null;
   643→          const msg = `Refreshed: ${pr.success} ok, ${pr.fail} failed`;
   644→          showToast(msg, pr.fail > 0 ? 'error' : 'success');
   645→          // Restore button
   646→          if (btn) {
   647→            btn.classList.remove('spinning');
   648→            btn.disabled = false;
   649→            btn.innerHTML = '<svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"><path d="M3 12a9 9 0 0 1 9-9 9.75 9.75 0 0 1 6.74 2.74L21 8"/><path d="M21 3v5h-5"/><path d="M21 12a9 9 0 0 1-9 9 9.75 9.75 0 0 1-6.74-2.74L3 16"/><path d="M3 21v-5h5"/></svg> Refresh All';
   650→          }
   651→          loadAccounts();
   652→        }
   653→      } catch (e) {
   654→        // Polling error — keep going
   655→      }
   656→    }, 500);
   657→
   658→  } catch (e) {
   659→    showToast('Error: ' + e.message, 'error');
   660→    if (btn) {
   661→      btn.classList.remove('spinning');
   662→      btn.disabled = false;
   663→    }
   664→  }
   665→}
