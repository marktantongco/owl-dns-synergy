    53→    53→def get_next_token():
    54→    54→    """Round-robin token selection across all accounts.
    55→    55→    Auto-refreshes expired tokens. Skips known-exhausted accounts (from cache only)."""
    56→    56→    global _token_idx
    57→    57→    data = load_tokens()
    58→    58→    if not data["accounts"]:
    59→    59→        return None, None
    60→    60→
    61→    61→    n = len(data["accounts"])
    62→    62→    with _token_lock:
    63→    63→        idx = _token_idx % n
    64→    64→        _token_idx += 1
    65→    65→
    66→    66→    # Try each account starting from idx
    67→    67→    for i in range(n):
    68→    68→        acc = data["accounts"][(idx + i) % n]
    69→    69→        # Check token validity
    70→    70→        from config import ACCESS_TOKEN_TTL, REFRESH_MARGIN
    71→    71→        age = time.time() - acc.get("last_refreshed", 0)
    72→    72→        if age < ACCESS_TOKEN_TTL - REFRESH_MARGIN:
    73→    73→            # Token still valid — check exhausted cache (non-blocking, cached only)
    74→    74→            if not _is_cached_exhausted(acc):
    75→    75→                return acc["access_token"], acc
    76→    76→            else:
    77→    77→                continue
    78→    78→        # Try refresh
    79→    79→        from auth import refresh_token
    80→    80→        new_token = refresh_token(acc)
    81→    81→        if new_token:
    82→    82→            if not _is_cached_exhausted(acc):
    83→    83→                return new_token, acc
    84→    84→            else:
    85→    85→                continue
    86→    86→
    87→    87→    # Fallback: return first valid token even if exhausted
    88→    88→    for acc in data["accounts"]:
    89→    89→        return acc["access_token"], acc
    90→    90→    return None, None

    99→    99→def _is_cached_exhausted(acc):

   105→   105→        return cached["balance"] <= 0

   107→   107→    return False

   265→   265→                generate(),
   266→   266→                mimetype="text/event-stream",

   299→   299→                                    if "id" in tc and tc["id"]:
   300→   300→                                        full_tool_calls[idx]["id"] = tc["id"]
   301→   301→                                    if "type" in tc and tc["type"]:
   302→   302→                                        full_tool_calls[idx]["type"] = tc["type"]
   303→   303→                                    if "function" in tc:

   314→   314→                            usage = chunk["usage"]

   327→   327→                "created": int(time.time()),

   491→   491→@app.route("/refresh-all", methods=["POST"])
   492→   492→def do_refresh_all():
   493→   493→    # If already running, return current progress
   494→   494→    if _refresh_progress["running"]:
   495→   495→        return jsonify({"status": "running", **_refresh_progress})
   496→   496→
   497→   497→    # Backup tokens.json before refresh (preventive)
   498→   498→    import shutil
   499→   499→    try:
   500→   500→        shutil.copy2(TOKENS_FILE_FULL, TOKENS_FILE_FULL + ".bak")
   501→   501→    except Exception as e:
   502→   502→        logger.warning(f"Backup before refresh-all failed: {e}")
   503→   503→
   504→   504→    # Start refresh in background thread
   505→   505→    data = load_tokens()
   506→   506→    _refresh_progress["total"] = len(data["accounts"])
   507→   507→    _refresh_progress["done"] = 0
   508→   508→    _refresh_progress["success"] = 0
   509→   509→    _refresh_progress["fail"] = 0
   510→   510→    _refresh_progress["running"] = True
   511→   511→
   512→   512→    def _bg_refresh():
   513→   513→        import auth as _auth
   514→   514→        from config import REFRESH_URL
   515→   515→        import requests as _req
   516→   516→        d = _auth.load_tokens()
   517→   517→        now = int(time.time())
   518→   518→        for acc in d["accounts"]:
   519→   519→            try:
   520→   520→                headers = _auth._sign_headers()
   521→   521→                body = {
   522→   522→                    "source_id": acc.get("source_id", "autoclaw"),
   523→   523→                    "device_id": acc["device_id"],
   524→   524→                    "refresh_token": acc["refresh_token"],
   525→   525→                }
   526→   526→                resp = _req.post(REFRESH_URL, json=body, headers=headers, timeout=15, verify=False)
   527→   527→                resp_data = resp.json()
   528→   528→                if resp_data.get("code") == 0 and "data" in resp_data:
   529→   529→                    new_access = resp_data["data"].get("access_token")
   530→   530→                    new_refresh = resp_data["data"].get("refresh_token", acc["refresh_token"])
   531→   531→                    if new_access:
   532→   532→                        acc["access_token"] = new_access
   533→   533→                        if new_refresh:
   534→   534→                            acc["refresh_token"] = new_refresh
   535→   535→                        acc["last_refreshed"] = now
   536→   536→                        _refresh_progress["success"] += 1
   537→   537→                    else:
   538→   538→                        _refresh_progress["fail"] += 1
   539→   539→                else:
   540→   540→                    _refresh_progress["fail"] += 1
   541→   541→            except Exception:
   542→   542→                _refresh_progress["fail"] += 1
   543→   543→            _refresh_progress["done"] += 1
   544→   544→        # Save once at end
   545→   545→        _auth.save_tokens(d)
   546→   546→        _refresh_progress["running"] = False
   547→   547→        logger.info(f"Refresh all done: {_refresh_progress['success']} ok, {_refresh_progress['fail']} fail")
   548→   548→
   549→   549→    t = threading.Thread(target=_bg_refresh, daemon=True)
   550→   550→    t.start()
   551→   551→
   552→   552→    return jsonify({"status": "started", **_refresh_progress})

   620→   620→@app.route("/auth/callback-google")
   621→   621→def auth_callback_google():
   622→   622→    """Google OAuth redirect target — auto-captures code+state, exchanges tokens."""
   623→   623→    from auth import google_oauth_login, add_token
   624→   624→
   625→   625→    code = request.args.get("code")
   626→   626→    state = request.args.get("state")
   627→   627→    error = request.args.get("error")
   628→   628→
   629→   629→    # Find pending login by state — MUST match, no fallback in concurrent mode
   630→   630→    pending = _pending_logins.get(state)
   631→   631→    if not pending:
   632→   632→        # No state match — do NOT fallback to "first pending" (causes cross-account
   633→   633→        # contamination in concurrent mode). Just fail this callback.
   634→   634→        logger.warning(f"Unknown state={state[:20]}... (not in {len(_pending_logins)} pending)")
   635→   635→        return "<html><body><h1>Login Failed</h1><p>Unknown state</p></body></html>", 400
   636→   636→
   637→   637→    if error:
   638→   638→        pending["error"] = error
   639→   639→        return f"<html><body><h1>Login Failed</h1><p>{error}</p></body></html>", 400
   640→   640→
   641→   641→    if not code:
   642→   642→        pending["error"] = "No code in callback"
   643→   643→        return "<html><body><h1>Login Failed</h1><p>No code</p></body></html>", 400
   644→   644→
   645→   645→    device_id = pending.get("device_id")
   646→   646→    proxy_used = pending.get("proxy")  # Reuse same proxy from URL generation
   647→   647→
   648→   648→    # Serialize token exchanges — AutoClaw API rate-limits per IP (630014).
   649→   649→    # Without this, 3 concurrent callbacks all hit the API simultaneously → all fail.
   650→   650→    # Lock ensures 1 exchange at a time, with small gap between.
   651→   651→    with _token_exchange_lock:
   652→   652→        logger.info(f"Exchanging token for state={state[:12]}... device_id={device_id[:8]}... proxy={'yes' if proxy_used else 'none'}")
   653→   653→        result = google_oauth_login(code, state, device_id, proxy=proxy_used)
   654→   654→        time.sleep(2)  # Gap so next concurrent callback doesn't hit 630014
   655→   655→    if not result:
   656→   656→        logger.error(f"Token exchange FAILED for state={state[:12]}... (google_oauth_login returned None)")
   657→   657→        pending["error"] = "Token exchange failed"
   658→   658→        return "<html><body><h1>Login Failed</h1><p>Token exchange failed</p></body></html>", 500
   659→   659→    logger.info(f"Token exchange OK for state={state[:12]}... user_id={result.get('user_id')}")
   660→   660→
   661→   661→    # Extract email from JWT jti field
   662→   662→    import base64 as _b64
   663→   663→    email = result.get("user_name") or f"user_{result['user_id']}"
   664→   664→    try:
   665→   665→        token_part = result["access_token"].replace("Bearer ", "").split(".")[1]
   666→   666→        token_part += "=" * (4 - len(token_part) % 4)
   667→   667→        payload = json.loads(_b64.urlsafe_b64decode(token_part))
   668→   668→        if payload.get("jti"):
   669→   669→            email = payload["jti"]
   670→   670→    except:
   671→   671→        pass
   672→   672→
   673→   673→    add_token(
   674→   674→        email=email,
   675→   675→        access_token=result["access_token"],
   676→   676→        refresh_token=result["refresh_token"],
   677→   677→        user_id=result["user_id"],
   678→   678→        device_id=device_id,
   679→   679→    )
   680→   680→    pending["result"] = {
   681→   681→        "email": email,
   682→   682→        "user_id": result.get("user_id"),
   683→   683→        "first_login": result.get("first_login"),
   684→   684→    }
   685→   685→    return f"""<html><body>
   686→   686→<h1>Login Success!</h1>
   687→   687→<p>Email: {email}</p>
   688→   688→<p>User ID: {result.get('user_id')}</p>
   689→   689→<p>You can close this tab.</p>
   690→   690→</body></html>"""
