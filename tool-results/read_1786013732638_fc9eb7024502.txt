   129→class CircuitState(Enum):
   130→    """3-state circuit breaker: CLOSED → OPEN → HALF_OPEN → CLOSED"""
   131→    CLOSED = 0      # Normal operation — requests flow through
   132→    OPEN = 1        # Circuit tripped — all requests rejected
   133→    HALF_OPEN = 2  # Probe state — one request allowed to test recovery
   134→
   135→
   136→class ChannelCircuitBreaker:
   137→    """Per-channel 3-state circuit breaker with Prometheus tracking.
   138→
   139→    States:
   140→      CLOSED: Normal. All requests pass. Failure count tracked.
   141→      OPEN: Circuit tripped. All requests fail-fast. Timer starts.
   142→      HALF_OPEN: One probe request allowed. If success → CLOSED, if fail → OPEN.
   143→
   144→    Args:
   145→        channel_name: Name for logging/metrics
   146→        failure_threshold: Consecutive failures before opening (default 5)
   147→        recovery_timeout: Seconds before half-open probe (default 30)
   148→        success_threshold: Consecutive successes in HALF_OPEN to close (default 2)
   149→    """
   150→    def __init__(self, channel_name: str, failure_threshold: int = 5,
   151→                 recovery_timeout: float = 30.0, success_threshold: int = 2):
   152→        self.channel_name = channel_name
   153→        self.failure_threshold = failure_threshold
   154→        self.recovery_timeout = recovery_timeout
   155→        self.success_threshold = success_threshold
   156→        self._state = CircuitState.CLOSED
   157→        self._failure_count = 0
   158→        self._success_count = 0
   159→        self._opened_at: float = 0.0
   160→        self._lock = asyncio.Lock()
   161→
   162→    @property
   163→    def state(self) -> CircuitState:
   164→        """Current state, with automatic HALF_OPEN transition on timeout."""
   165→        if self._state == CircuitState.OPEN:
   166→            if time.time() - self._opened_at >= self.recovery_timeout:
   167→                self._state = CircuitState.HALF_OPEN
   168→        return self._state
   169→    @property
   170→    def is_open(self) -> bool:
   171→        """True if circuit is OPEN (requests should be rejected)."""
   172→        return self.state == CircuitState.OPEN
   173→    async def allow_request(self) -> bool:
   174→        """Check if a request should be allowed through."""
   175→        async with self._lock:
   176→            state = self.state
   177→            if state == CircuitState.CLOSED:
   178→                return True
   179→            elif state == CircuitState.HALF_OPEN:
   180→                # Allow ONE probe request through
   181→                return True
   182→            else:  # OPEN
   183→                return False
   184→    async def record_success(self):
   185→        """Record a successful request."""
   186→        async with self._lock:
   187→            if self._state == CircuitState.HALF_OPEN:
   188→                self._success_count += 1
   189→                if self._success_count >= self.success_threshold:
   190→                    self._state = CircuitState.CLOSED
   191→                    self._failure_count = 0
   192→                    self._success_count = 0
   193→                    logger.info(f"Circuit CLOSED for {self.channel_name} — recovered")
   194→            elif self._state == CircuitState.CLOSED:
   195→                self._failure_count = 0  # Reset on success
   196→    async def record_failure(self):
   197→        """Record a failed request."""
   198→        async with self._lock:
   199→            if self._state == CircuitState.HALF_OPEN:
   200→                # Probe failed → back to OPEN
   201→                self._state = CircuitState.OPEN
   202→                self._opened_at = time.time()
   203→                self._success_count = 0
   204→                logger.warning(f"Circuit back to OPEN for {self.channel_name} — probe failed")
   205→            elif self._state == CircuitState.CLOSED:
   206→                self._failure_count += 1
   207→                if self._failure_count >= self.failure_threshold:
   208→                    self._state = CircuitState.OPEN
   209→                    self._opened_at = time.time()
   210→                    logger.warning(
   211→                        f"Circuit OPEN for {self.channel_name} — "
   212→                        f"{self._failure_count} failures >= threshold {self.failure_threshold}"
   213→                    )

   228→@dataclass
   229→class DomainPreference:
   230→    """Per-domain channel preference with EMA-based learning.
   231→
   232→    Tracks success/failure counts per channel and uses exponential moving average
   233→    to compute a quality score. The channel with the highest EMA score becomes
   234→    the preferred_channel for subsequent requests to this domain.
   235→    """
   236→    domain: str
   237→    successes: Dict[str, int] = field(default_factory=lambda: {})
   238→    failures: Dict[str, int] = field(default_factory=lambda: {})
   239→    preferred_channel: str = "http_proxy"
   240→    last_updated: float = field(default_factory=time.time)
   241→    # EMA scores per channel (0.0-1.0)
   242→    _ema_scores: Dict[str, float] = field(default_factory=lambda: {})
   243→    _ema_alpha: float = 0.3  # EMA smoothing factor (higher = more responsive)
   244→
   245→    def record_channel_result(self, channel: str, success: bool):
   246→        """Record a channel result and update EMA score."""
   247→        if success:
   248→            self.successes[channel] = self.successes.get(channel, 0) + 1
   249→        else:
   250→            self.failures[channel] = self.failures.get(channel, 0) + 1
   251→        # EMA update: new_score = alpha * sample + (1 - alpha) * old_score
   252→        old_score = self._ema_scores.get(channel, 0.5)
   253→        sample = 1.0 if success else 0.0
   254→        self._ema_scores[channel] = self._ema_alpha * sample + (1 - self._ema_alpha) * old_score
   255→        self.last_updated = time.time()
   256→        # Re-compute preferred channel from EMA scores
   257→        if self._ema_scores:
   258→            best_channel = max(self._ema_scores, key=self._ema_scores.get)
   259→            if self._ema_scores[best_channel] > 0.3:  # Minimum quality threshold
   260→                self.preferred_channel = best_channel
   261→
   262→    def get_channel_score(self, channel: str) -> float:
   263→        """Get EMA quality score for a channel (0.0-1.0)."""
   264→        return self._ema_scores.get(channel, 0.5)

   271→class OpenRouterKeyRotator:
   272→    """Round-robin API key rotation with failover on 429/401/403."""
   273→
   274→    def __init__(self, keys: List[str] = None, base_url: str = "https://openrouter.ai/api/v1"):
   275→        self._keys = keys or []
   276→        self._current_index = 0
   277→        self._base_url = base_url
   278→        self._key_errors: Dict[int, int] = {}
   279→        self._key_cooldown: Dict[int, float] = {}
   280→        self._lock = threading.Lock()
   281→
   282→    @classmethod
   283→    def from_env(cls) -> "OpenRouterKeyRotator":
   284→        keys = []
   285→        primary = os.getenv("OPENAI_API_KEY")
   286→        if primary:
   287→            keys.append(primary)
   288→        for i in range(1, 10):
   289→            key = os.getenv(f"OPENROUTER_KEY_{i}")
   290→            if key:
   291→                keys.append(key)
   292→        base_url = os.getenv("OPENAI_BASE_URL", "https://openrouter.ai/api/v1")
   293→        return cls(keys=keys, base_url=base_url)
   294→
   295→    @property
   296→    def base_url(self) -> str:
   297→        return self._base_url
   298→
   299→    def get_active_key(self) -> Optional[str]:
   300→        with self._lock:
   301→            if not self._keys:
   302→                return None
   303→            now = time.time()
   304→            if now >= self._key_cooldown.get(self._current_index, 0):
   305→                return self._keys[self._current_index]
   306→            for offset in range(len(self._keys)):
   307→                idx = (self._current_index + offset) % len(self._keys)
   308→                if now >= self._key_cooldown.get(idx, 0):
   309→                    self._current_index = idx
   310→                    KEY_ROTATION_COUNT.inc()
   311→                    return self._keys[idx]
   312→            return None
   313→
   314→    def report_error(self, status_code: int, error_type: str = "unknown"):
   315→        with self._lock:
   316→            idx = self._current_index
   317→            self._key_errors[idx] = self._key_errors.get(idx, 0) + 1
   318→            if status_code == 429:
   319→                self._key_cooldown[idx] = time.time() + 60
   320→            elif status_code in (401, 403):
   321→                self._key_cooldown[idx] = time.time() + 300
   322→            else:
   323→                self._key_cooldown[idx] = time.time() + 10
   324→            self._rotate_to_next()
   325→
   326→    def _rotate_to_next(self):
   327→        if len(self._keys) <= 1:
   328→            return
   329→        now = time.time()
   330→        for offset in range(1, len(self._keys)):
   331→            idx = (self._current_index + offset) % len(self._keys)
   332→            if now >= self._key_cooldown.get(idx, 0):
   333→                self._current_index = idx
   334→                KEY_ROTATION_COUNT.inc()
   335→                return
   336→        self._current_index = (self._current_index + 1) % len(self._keys)
   337→        KEY_ROTATION_COUNT.inc()
   338→
   339→    @property
   340→    def total_keys(self) -> int:
   341→        return len(self._keys)
   342→
   343→    @property
   344→    def available_keys(self) -> int:
   345→        now = time.time()
   346→        return sum(1 for i in range(len(self._keys))
   347→                   if now >= self._key_cooldown.get(i, 0))
   348→
   349→    def get_status(self) -> Dict[str, Any]:
   350→        now = time.time()
   351→        return {
   352→            "total_keys": self.total_keys,
   353→            "available_keys": self.available_keys,
   354→            "current_index": self._current_index,
   355→            "key_errors": dict(self._key_errors),
   356→            "cooldowns": {
   357→                i: max(0, self._key_cooldown.get(i, 0) - now)
   358→                for i in range(len(self._keys))
   359→            }
   360→        }

   364→# 2. DNS Flood Protector (from v2)
   365→# ═══════════════════════════════════════════════════════════════
   366→
   367→class DNSFloodProtector:
   368→    """Token-bucket + per-client rate limiting for DNS queries."""
   369→
   370→    def __init__(self, max_qps: int = 50, burst: int = 100):
   371→        self.max_qps = max_qps
   372→        self.burst = burst
   373→        self._tokens: float = float(burst)
   374→        self._last_refill: float = time.time()
   375→        self._lock = asyncio.Lock()
   376→        self._blocked_count = 0
   377→        self._client_queries: Dict[str, deque] = {}
   378→
   379→    async def allow(self, client_ip: str = "default") -> bool:
   380→        async with self._lock:
   381→            now = time.time()
   382→            # Per-client rate check FIRST (before consuming global token)
   383→            if client_ip not in self._client_queries:
   384→                self._client_queries[client_ip] = deque(maxlen=100)
   385→            self._client_queries[client_ip].append(now)
   386→            recent = sum(1 for t in self._client_queries[client_ip] if now - t < 1.0)
   387→            if recent > 10:
   388→                self._blocked_count += 1
   389→                DNS_FLOOD_BLOCKED.inc()
   390→                return False
   391→            # Global token bucket check (after per-client passes)
   392→            elapsed = now - self._last_refill
   393→            self._tokens = min(self.burst, self._tokens + elapsed * self.max_qps)
   394→            self._last_refill = now
   395→            if self._tokens >= 1.0:
   396→                self._tokens -= 1.0
   397→                return True
   398→            self._blocked_count += 1
   399→            DNS_FLOOD_BLOCKED.inc()
   400→            return False

   405→# ═══════════════════════════════════════════════════════════════

   407→@dataclass
   408→class ProxyEntry:
   409→    """Represents a single proxy endpoint (prox5-compatible)."""
   410→    endpoint: str           # host:port
   411→    protocol: str = "socks5"  # socks5, socks4, http
   412→    username: str = ""
   413→    password: str = ""
   414→    proxied_ip: str = ""
   415→    last_validated: float = 0.0
   416→    successes: int = 0
   417→    failures: int = 0
   418→    score: float = 1.0      # EMA quality score
   419→    region: str = ""
   420→
   421→    @property
   422→    def url(self) -> str:
   423→        if self.username and self.password:
   424→            return f"{self.protocol}://{self.username}:{self.password}@{self.endpoint}"
   425→        return f"{self.protocol}://{self.endpoint}"
   426→
   427→    @property
   428→    def is_stale(self) -> bool:
   429→        return time.time() - self.last_validated > 1800  # 30 min
   430→
   431→
   432→class ProxyPoolAdapter:
   433→    """
   434→    Python adapter for prox5-style proxy pool management.
   435→    Loads proxies from file (prox5/proxies.txt format), validates,
   436→    and rotates using Mystery Dialer pattern (retry on failure).
   437→    """
   438→
   439→    def __init__(self, proxy_file: str = None, max_workers: int = 10):
   440→        self._pool: List[ProxyEntry] = []
   441→        self._current_index = 0
   442→        self._lock = threading.Lock()
   443→        self._max_workers = max_workers
   444→        self._proxy_file = proxy_file
   445→        if proxy_file and os.path.exists(proxy_file):
   446→            self.load_from_file(proxy_file)
   447→
   448→    def load_from_file(self, filepath: str):
   449→        """
   450→        Load proxies from file. Supports formats:
   451→          - host:port:user:pass  (autoclaw/prox5 format)
   452→          - protocol://host:port  (URL format)
   453→          - host:port  (simple format)
   454→        """
   455→        count = 0
   456→        with open(filepath) as f:
   457→            for line in f:
   458→                line = line.strip()
   459→                if not line or line.startswith('#'):
   460→                    continue
   461→                entry = self._parse_proxy_line(line)
   462→                if entry:
   463→                    self._pool.append(entry)
   464→                    count += 1
   465→        logger.info(f"Loaded {count} proxies from {filepath}")
   466→        PROXY_POOL_SIZE.labels(protocol="all").set(len(self._pool))
   467→
   468→    def _parse_proxy_line(self, line: str) -> Optional[ProxyEntry]:
   469→        """Parse a proxy line in any supported format."""
   470→        if '://' in line:
   471→            # URL format: socks5://user:pass@host:port
   472→            parsed = urlparse(line)
   473→            protocol = parsed.scheme or "socks5"
   474→            host = parsed.hostname or ""
   475→            port = parsed.port or 1080
   476→            return ProxyEntry(
   477→                endpoint=f"{host}:{port}",
   478→                protocol=protocol,
   479→                username=parsed.username or "",
   480→                password=parsed.password or "",
   481→            )
   482→        parts = line.split(':')
   483→        if len(parts) == 4:
   484→            # host:port:user:pass format
   485→            return ProxyEntry(
   486→                endpoint=f"{parts[0]}:{parts[1]}",
   487→                username=parts[2],
   488→                password=parts[3],
   489→            )
   490→        if len(parts) == 2:
   491→            try:
   492→                int(parts[1])
   493→                return ProxyEntry(endpoint=line)
   494→            except ValueError:
   495→                pass
   496→        return None
   497→
   498→    def add_proxy(self, endpoint: str, protocol: str = "socks5",
   499→                  username: str = "", password: str = ""):
   500→        with self._lock:
   501→            self._pool.append(ProxyEntry(
   502→                endpoint=endpoint, protocol=protocol,
   503→                username=username, password=password
   504→            ))
   505→
   506→    def get_next(self) -> Optional[ProxyEntry]:
   507→        """Get next proxy using round-robin with Mystery Dialer retry."""
   508→        with self._lock:
   509→            if not self._pool:
   510→                return None
   511→            # Try current proxy first
   512→            for offset in range(len(self._pool)):
   513→                idx = (self._current_index + offset) % len(self._pool)
   514→                proxy = self._pool[idx]
   515→                if proxy.score > 0.2 and not proxy.is_stale:
   516→                    self._current_index = idx
   517→                    return proxy
   518→            # All stale/low-score, return best available
   519→            best = max(self._pool, key=lambda p: p.score)
   520→            return best
   521→
   522→    def report_success(self, endpoint: str, latency_ms: float = 0):
   523→        with self._lock:
   524→            for p in self._pool:
   525→                if p.endpoint == endpoint:
   526→                    p.successes += 1
   527→                    p.last_validated = time.time()
   528→                    # EMA update: alpha=0.3
   529→                    p.score = 0.3 * 1.0 + 0.7 * p.score
   530→                    break
   531→
   532→    def report_failure(self, endpoint: str):
   533→        with self._lock:
   534→            for p in self._pool:
   535→                if p.endpoint == endpoint:
   536→                    p.failures += 1
   537→                    p.score = 0.3 * 0.0 + 0.7 * p.score
   538→                    break
   539→
   540→    @property
   541→    def size(self) -> int:
   542→        return len(self._pool)
   543→
   544→    @property
   545→    def valid_count(self) -> int:
   546→        return sum(1 for p in self._pool if p.score > 0.2 and not p.is_stale)
   547→
   548→    def get_status(self) -> Dict[str, Any]:
   549→        return {
   550→            "total": self.size,
   551→            "valid": self.valid_count,
   552→            "current_index": self._current_index,
   553→            "protocols": {
   554→                proto: sum(1 for p in self._pool if p.protocol == proto)
   555→                for proto in set(p.protocol for p in self._pool)
   556→            },
   557→            "top_5": [
   558→                {"endpoint": p.endpoint, "protocol": p.protocol,
   559→                 "score": round(p.score, 3), "successes": p.successes,

  1106→        """Select the optimal channel for a domain."""
  1107→        if force_channel:
  1108→            try:
  1109→                return Channel(force_channel)
  1110→            except ValueError:
  1111→                pass
  1112→
  1113→        pref = self._prefs.get(domain)
  1114→        if pref and pref.preferred_channel:
  1115→            try:
  1116→                return Channel(pref.preferred_channel)
  1117→            except ValueError:
  1118→                pass
  1119→
  1120→        # Default priority: proxy pool → SOCKS → DNS → MITM → direct
  1121→        if self.proxy_pool.valid_count > 0:
  1122→            return Channel.HTTP_PROXY
  1123→        return Channel.DNS_TUNNEL

  1127→    async def _try_http_proxy(self, url: str, domain: str, **kwargs) -> ChannelResult:
  1128→        """Route through proxy pool (prox5/https_proxy)."""
  1129→        import httpx
  1130→        start = time.time()
  1131→        proxy = self.proxy_pool.get_next()
  1132→
  1133→        if not proxy:
  1134→            return ChannelResult("http_proxy", False, error="No proxies available")
  1135→
  1136→        try:
  1137→            api_key = self.key_rotator.get_active_key()
  1138→            headers = {}
  1139→            if api_key:
  1140→                headers["Authorization"] = f"Bearer {api_key}"
  1141→
  1142→            async with httpx.AsyncClient(
  1143→                proxy=proxy.url, timeout=30.0
  1144→            ) as client:
  1145→                resp = await client.get(url, headers=headers)
  1146→                latency = (time.time() - start) * 1000
  1147→
  1148→                if resp.status_code in (401, 403, 429):
  1149→                    self.key_rotator.report_error(
  1150→                        resp.status_code,
  1151→                        "rate_limit" if resp.status_code == 429 else "auth_error"
  1152→                    )
  1153→
  1154→                self.proxy_pool.report_success(proxy.endpoint, latency)
  1155→
  1156→                # Only treat 2xx as success — non-2xx should trigger failover
  1157→                is_success = 200 <= resp.status_code < 300
  1158→                if is_success:
  1159→                    self._cache_set(url, resp.content)
  1160→
  1161→                return ChannelResult(
  1162→                    "http_proxy", is_success,
  1163→                    data=resp.content, latency_ms=latency,
  1164→                    status_code=resp.status_code, proxy_used=proxy.endpoint
  1165→                )
  1166→        except Exception as e:
  1167→            latency = (time.time() - start) * 1000
  1168→            self.proxy_pool.report_failure(proxy.endpoint)
  1169→            return ChannelResult("http_proxy", False, error=str(e), latency_ms=latency)

  1171→    async def _try_socks_pool(self, url: str, domain: str, **kwargs) -> ChannelResult:
  1172→        """Route through SOCKS5 pool directly (prox5 Mystery Dialer pattern)."""

  1195→    async def _try_dns_tunnel(self, url: str, domain: str, **kwargs) -> ChannelResult:
  1196→        """Fall back to DNS tunneling (llm-dns-proxy).
  1197→        Verifies DNS server is actually reachable by sending a health-check query."""
  1198→        start = time.time()
  1199→        try:
  1200→            import socket
  1201→            dns_host = os.getenv("DNS_SERVER_HOST", "127.0.0.1")
  1202→            dns_port = int(os.getenv("DNS_SERVER_PORT", "5353"))
  1203→            sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
  1204→            sock.settimeout(2)
  1205→            # Send a minimal DNS query to verify the server is actually running
  1206→            # DNS header: ID=0x1234, RD=1, QDCOUNT=1, QNAME=., QTYPE=TXT, QCLASS=IN
  1207→            query = b'\x12\x34\x01\x00\x00\x01\x00\x00\x00\x00\x00\x00\x00\x00\x10\x00\x01'
  1208→            sock.sendto(query, (dns_host, dns_port))
  1209→            # Wait for any response (even SERVFAIL proves the server is alive)
  1210→            data, _ = sock.recvfrom(512)
  1211→            sock.close()
  1212→            if not data:
  1213→                return ChannelResult("dns_tunnel", False, error="DNS server returned empty response")
  1214→            return ChannelResult(
  1215→                "dns_tunnel", True,
  1216→                data=f"[DNS tunnel verified for {domain}]",
  1217→                latency_ms=(time.time() - start) * 1000
  1218→            )
  1219→        except socket.timeout:
  1220→            return ChannelResult("dns_tunnel", False, error="DNS server not reachable (timeout)")
  1221→        except ConnectionRefusedError:
  1222→            return ChannelResult("dns_tunnel", False, error="DNS server not running (connection refused)")
  1223→        except Exception as e:
  1224→            return ChannelResult("dns_tunnel", False, error=str(e))

  1258→    async def fetch(self, url: str, force_channel: str = None, **kwargs) -> ChannelResult:
  1259→        """
  1260→        Fetch a URL through the optimal channel with automatic failover.
  1261→        Now includes per-channel circuit breaker and EMA domain preference learning.
  1262→
  1263→        Channel selection priority:
  1264→          cached → http_proxy → socks_pool → dns_tunnel → mitm_stealth → connect_chain → http_direct
  1265→        """
  1266→        domain = self._extract_domain(url)
  1267→
  1268→        # Check cache
  1269→        cached = self._cache_get(url)
  1270→        if cached and not force_channel:
  1271→            return ChannelResult("cached", True, data=cached)
  1272→
  1273→        # Select channel (uses DomainPreference EMA if available)
  1274→        channel = self._select_channel(domain, force_channel)
  1275→
  1276→        # Channel implementations
  1277→        channel_map = {
  1278→            Channel.HTTP_PROXY: self._try_http_proxy,
  1279→            Channel.SOCKS_POOL: self._try_socks_pool,
  1280→            Channel.DNS_TUNNEL: self._try_dns_tunnel,
  1281→            Channel.MITM_STEALTH: self._try_mitm_stealth,
  1282→            Channel.CONNECT_CHAIN: self._try_connect_chain,
  1283→            Channel.HTTP_DIRECT: self._try_http_direct,
  1284→        }
  1285→
  1286→        # Try preferred channel (with circuit breaker check)
  1287→        if channel in channel_map:
  1288→            cb = self._circuit_breakers.get(channel.value)
  1289→            if cb and not await cb.allow_request():
  1290→                logger.debug(f"Circuit OPEN for {channel.value} — skipping")
  1291→            else:
  1292→                result = await channel_map[channel](url, domain, **kwargs)
  1293→                # Record result in circuit breaker and EMA
  1294→                if cb:
  1295→                    if result.success:
  1296→                        await cb.record_success()
  1297→                    else:
  1298→                        await cb.record_failure()
  1299→                # Update domain preference EMA
  1300→                pref = self._prefs.get(domain)
  1301→                if not pref:
  1302→                    pref = DomainPreference(domain=domain)
  1303→                    self._prefs[domain] = pref
  1304→                pref.record_channel_result(result.channel, result.success)
  1305→
  1306→                if result.success:
  1307→                    REQUESTS_TOTAL.labels(channel=result.channel, domain=domain, status="success").inc()
  1308→                    REQUESTS_DURATION.labels(channel=result.channel, domain=domain).observe(result.latency_ms / 1000)
  1309→                    return result
  1310→                # Record failure in Prometheus
  1311→                REQUESTS_TOTAL.labels(channel=result.channel, domain=domain, status="error").inc()
  1312→
  1313→        # Fallback: try all channels in priority order
  1314→        # NOTE: CONNECT_CHAIN was missing — added between MITM and HTTP_DIRECT
  1315→        fallback_order = [
  1316→            Channel.HTTP_PROXY, Channel.SOCKS_POOL, Channel.DNS_TUNNEL,
  1317→            Channel.MITM_STEALTH, Channel.CONNECT_CHAIN, Channel.HTTP_DIRECT,
  1318→        ]
  1319→        for ch in fallback_order:
  1320→            if ch == channel:
  1321→                continue  # Already tried
  1322→            if ch not in channel_map:
  1323→                continue
  1324→
  1325→            # Circuit breaker check
  1326→            cb = self._circuit_breakers.get(ch.value)
  1327→            if cb and not await cb.allow_request():
  1328→                logger.debug(f"Circuit OPEN for {ch.value} — skipping fallback")
  1329→                continue
  1330→
  1331→            result = await channel_map[ch](url, domain, **kwargs)
  1332→
  1333→            # Record in circuit breaker and EMA
  1334→            if cb:
  1335→                if result.success:
  1336→                    await cb.record_success()
  1337→                else:
  1338→                    await cb.record_failure()
  1339→            pref = self._prefs.get(domain)
  1340→            if pref:
  1341→                pref.record_channel_result(result.channel, result.success)
  1342→
  1343→            if result.success:
  1344→                CHANNEL_SWITCHES.labels(
  1345→                    from_channel=channel.value, to_channel=ch.value
  1346→                ).inc()
  1347→                REQUESTS_TOTAL.labels(channel=result.channel, domain=domain, status="success").inc()
  1348→                REQUESTS_DURATION.labels(channel=result.channel, domain=domain).observe(result.latency_ms / 1000)
  1349→                return result
  1350→            # Record failure
  1351→            REQUESTS_TOTAL.labels(channel=result.channel, domain=domain, status="error").inc()
  1352→
  1353→        REQUESTS_TOTAL.labels(channel="none", domain=domain, status="all_failed").inc()
  1354→        return ChannelResult("none", False, error=f"All channels exhausted for {domain}")
