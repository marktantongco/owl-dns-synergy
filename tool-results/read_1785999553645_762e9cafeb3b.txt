    22→    22→class SimpleSpinner:
    23→    23→    def __init__(self, message="Loading"):
    24→    24→        self.message = message
    25→    25→        self.running = False
    26→    26→        self.thread = None
    27→    27→        self.spinner_chars = "|/-\\"
    28→    28→        self.current = 0
    29→    29→
    30→    30→    def _spin(self):
    31→    31→        while self.running:
    32→    32→            sys.stdout.write(f"\r{self.message} {self.spinner_chars[self.current % len(self.spinner_chars)]}")
    33→    33→            sys.stdout.flush()
    34→    34→            self.current += 1
    35→    35→            time.sleep(0.1)
    36→    36→        sys.stdout.write("\r" + " " * (len(self.message) + 2) + "\r")
    37→    37→        sys.stdout.flush()
    38→    38→
    39→    39→    def start(self):
    40→    40→        self.running = True
    41→    41→        self.thread = threading.Thread(target=self._spin)
    42→    42→        self.thread.start()
    43→    43→
    44→    44→    def stop(self):
    45→    45→        self.running = False
    46→    46→        if self.thread:
    47→    47→            self.thread.join()

    63→    63→        # Use 3-digit session ID (000-999) for much better uniqueness
    64→    64→        # Mix timestamp and random for collision avoidance
    65→    65→        time_component = int(time.time()) % 1000
    66→    66→        random_component = uuid.uuid4().int % 1000
    67→    67→        session_num = (time_component + random_component) % 1000

    70→    70→    def _send_dns_query(self, query_name: str) -> Optional[str]:
    71→    71→        """Send a DNS query and return the TXT record response."""
    72→    72→        try:
    73→    73→            # Manual DNS record construction to avoid dnslib issues
    74→    74→            question = DNSQuestion(query_name, QTYPE.TXT)
    75→    75→            header = DNSHeader(id=random.randint(1, 65535))
    76→    76→            q = DNSRecord(header, q=question)
    77→    77→            query_data = q.pack()
    78→    78→
    79→    79→            self.sock.sendto(query_data, (self.server_host, self.server_port))
    80→    80→
    81→    81→            response_data, _ = self.sock.recvfrom(4096)
    82→    82→            response = DNSRecord.parse(response_data)
    83→    83→
    84→    84→            for rr in response.rr:
    85→    85→                if rr.rtype == QTYPE.TXT:
    86→    86→                    # Handle TXT RDATA which can be a list of strings (each ≤255 bytes)
    87→    87→                    rdata = rr.rdata
    88→    88→
    89→    89→                    # dnslib TXT RDATA has a 'data' attribute that's a list of byte strings
    90→    90→                    if hasattr(rdata, 'data') and isinstance(rdata.data, list):
    91→    91→                        # Join all byte strings and decode
    92→    92→                        txt_bytes = b''.join(rdata.data)
    93→    93→                        txt_data = txt_bytes.decode('utf-8', errors='ignore')
    94→    94→                    else:
    95→    95→                        # Fallback to string conversion
    96→    96→                        txt_data = str(rdata)
    97→    97→                        # Handle quoted TXT records
    98→    98→                        if txt_data.startswith('"') and txt_data.endswith('"'):
    99→    99→                            txt_data = txt_data[1:-1]
   100→   100→
   101→   101→                    return txt_data
   102→   102→
   103→   103→            return None
   104→   104→
   105→   105→        except Exception as e:
   106→   106→            if self.verbose:
   107→   107→                click.echo(f"DNS query error: {e}")
   108→   108→            return None

   110→   110→    def send_message(self, message: str, show_spinner: bool = True, streaming: bool = True) -> Optional[str]:
   111→   111→        """Send a message to the LLM through DNS and get the response.
   112→   112→
   113→   113→        Args:
   114→   114→            message: The message to send
   115→   115→            show_spinner: Whether to show spinner during processing
   116→   116→            streaming: Whether to display streaming response (default True)
   117→   117→        """
   118→   118→        # Use the persistent session ID for this client instance
   119→   119→        session_id = self.session_id
   120→   120→        if self.verbose:
   121→   121→            click.echo(f"Using session ID: {session_id}")
   122→   122→        spinner = None
   123→   123→
   124→   124→        try:
   125→   125→            if self.verbose:
   126→   126→                click.echo(f"Encrypting message...")
   127→   127→            encrypted_data = self.crypto.encrypt(message)
   128→   128→
   129→   129→            if self.verbose:
   130→   130→                click.echo(f"Creating message chunks...")
   131→   131→            chunks = self.chunker.create_chunks(encrypted_data, session_id)
   132→   132→
   133→   133→            if show_spinner and not self.verbose:
   134→   134→                spinner = SimpleSpinner("...")
   135→   135→                spinner.start()
   136→   136→
   137→   137→            if self.verbose:
   138→   138→                click.echo(f"Sending {len(chunks)} chunks...")
   139→   139→
   140→   140→            for i, chunk in enumerate(chunks):
   141→   141→                if self.verbose:
   142→   142→                    click.echo(f"Sending chunk {i+1}/{len(chunks)}")
   143→   143→                response = self._send_dns_query(chunk)
   144→   144→                if response != "OK" and self.verbose:
   145→   145→                    click.echo(f"Warning: Unexpected response for chunk {i+1}: {response}")
   146→   146→
   147→   147→            if spinner:
   148→   148→                spinner.stop()
   149→   149→
   150→   150→            if streaming and not self.verbose:
   151→   151→                return self._handle_streaming_response(session_id)
   152→   152→            else:
   153→   153→                return self._handle_traditional_response(session_id, show_spinner)
   154→   154→
   155→   155→        finally:
   156→   156→            if spinner:
   157→   157→                spinner.stop()

   159→   159→    def _handle_streaming_response(self, session_id: str) -> Optional[str]:
   160→   160→        """Handle streaming response by displaying individual chunks immediately."""
   161→   161→        import sys
   162→   162→
   163→   163→        displayed_chunks = set()  # Track which chunks we've already displayed
   164→   164→        final_response = ""
   165→   165→        start_time = time.time()
   166→   166→        max_wait_time = 60  # Maximum wait time in seconds
   167→   167→        last_chunk_time = time.time()
   168→   168→
   169→   169→        # Initial wait for processing to start
   170→   170→        time.sleep(self.poll_interval)
   171→   171→
   172→   172→        while True:
   173→   173→            # Check for timeout
   174→   174→            current_time = time.time()
   175→   175→            if current_time - start_time > max_wait_time:
   176→   176→                if self.verbose:
   177→   177→                    click.echo(f"\nTimeout waiting for response after {max_wait_time}s")
   178→   178→                break
   179→   179→
   180→   180→            # Check if no new chunks for too long (indicates completion)
   181→   181→            if current_time - last_chunk_time > 10:  # 10 seconds without new chunks
   182→   182→                if final_response and '[EOS]' in final_response:
   183→   183→                    break
   184→   184→                elif len(final_response) > 50:  # Only timeout if we have substantial content
   185→   185→                    if not final_response.endswith('\n'):
   186→   186→                        sys.stdout.write('\n')
   187→   187→                        sys.stdout.flush()
   188→   188→                    final_response += "[Response may be incomplete - connection timed out]"
   189→   189→                    break
   190→   190→
   191→   191→            response_chunks = self._get_current_response_chunks(session_id)
   192→   192→
   193→   193→            if not response_chunks:
   194→   194→                time.sleep(self.poll_interval)
   195→   195→                continue
   196→   196→
   197→   197→            # Process each new chunk individually and display immediately
   198→   198→            new_chunks_found = False
   199→   199→            for chunk_index, chunk_data in response_chunks.items():
   200→   200→                if chunk_index not in displayed_chunks:
   201→   201→                    displayed_chunks.add(chunk_index)
   202→   202→                    new_chunks_found = True
   203→   203→                    last_chunk_time = current_time
   204→   204→
   205→   205→                    try:
   206→   206→                        # Try to decrypt this individual chunk
   207→   207→                        parts = chunk_data.split(':', 2)
   208→   208→                        if len(parts) == 3:
   209→   209→                            encrypted_data = parts[2].encode('ascii')
   210→   210→                            try:
   211→   211→                                # Try streaming chunk decryption
   212→   212→                                chunk_text = self.crypto.decrypt_chunk(encrypted_data)
   213→   213→                            except Exception:
   214→   214→                                # Fall back to regular decryption if chunk method fails
   215→   215→                                chunk_text = self.crypto.decrypt(encrypted_data)
   216→   216→
   217→   217→                            # Display chunk content immediately (remove [EOS] marker)
   218→   218→                            display_text = chunk_text.replace('[EOS]', '')
   219→   219→                            if display_text:
   220→   220→                                sys.stdout.write(display_text)
   221→   221→                                sys.stdout.flush()
   222→   222→                                final_response += display_text
   223→   223→
   224→   224→                            # Check if this chunk contains [EOS] marker
   225→   225→                            if '[EOS]' in chunk_text:
   226→   226→                                if not final_response.endswith('\n'):
   227→   227→                                    sys.stdout.write('\n')
   228→   228→                                    sys.stdout.flush()
   229→   229→                                return final_response.rstrip()
   230→   230→
   231→   231→                    except Exception as e:
   232→   232→                        if self.verbose:
   233→   233→                            click.echo(f"\nError decrypting chunk {chunk_index}: {e}")
   234→   234→                        continue
   235→   235→
   236→   236→            if not new_chunks_found:
   237→   237→                time.sleep(self.poll_interval)
   238→   238→
   239→   239→        # Return final response (timeout case)
   240→   240→        return final_response.rstrip() if final_response else None

   242→   242→    def _get_current_response_chunks(self, session_id: str) -> dict:
   243→   243→        """Get current response chunks from server."""
   244→   244→        response_chunks = {}
   245→   245→        chunk_index = 0
   246→   246→        max_chunks = 100  # Increased limit for large responses
   247→   247→        total_chunks = None
   248→   248→        consecutive_not_found = 0
   249→   249→        max_consecutive_not_found = 50  # Allow more gaps - server may be generating chunks slowly
   250→   250→
   251→   251→        while chunk_index < max_chunks and consecutive_not_found < max_consecutive_not_found:
   252→   252→            query = format_dns_query("g", session_id, chunk_index)
   253→   253→            response = self._send_dns_query(query)
   254→   254→
   255→   255→            if response == "NOT_FOUND":
   256→   256→                consecutive_not_found += 1
   257→   257→                chunk_index += 1
   258→   258→                continue
   259→   259→            else:
   260→   260→                consecutive_not_found = 0  # Reset counter on successful retrieval
   261→   261→
   262→   262→            if response and ':' in response:
   263→   263→                parts = response.split(':', 2)
   264→   264→                if len(parts) == 3:
   265→   265→                    try:
   266→   266→                        current_index = int(parts[0])
   267→   267→                        chunk_total = int(parts[1])
   268→   268→                        response_chunks[current_index] = response
   269→   269→
   270→   270→                        # Update total_chunks if we haven't seen it yet or if it's larger
   271→   271→                        if total_chunks is None or chunk_total > total_chunks:
   272→   272→                            total_chunks = chunk_total
   273→   273→                            # Don't limit max_chunks yet - server might still be generating more
   274→   274→
   275→   275→                        # If we have all chunks we expect, we can break early
   276→   276→                        if total_chunks and len(response_chunks) >= total_chunks:
   277→   277→                            # Verify we have a contiguous sequence from 0 to total_chunks-1
   278→   278→                            if all(i in response_chunks for i in range(total_chunks)):
   279→   279→                                break
   280→   280→                    except (ValueError, IndexError):
   281→   281→                        if self.verbose:
   282→   282→                            click.echo(f"Error parsing chunk response: {response}")
   283→   283→
   284→   284→            chunk_index += 1
   285→   285→
   286→   286→        # If we know how many chunks we should have, try to wait for missing ones
   287→   287→        if total_chunks and len(response_chunks) < total_chunks:
   288→   288→            missing_chunks = [i for i in range(total_chunks) if i not in response_chunks]
   289→   289→            if self.verbose:
   290→   290→                click.echo(f"Waiting for {len(missing_chunks)} missing chunks: {missing_chunks}")
   291→   291→
   292→   292→            # Implement chunk-level retries with exponential backoff for flaky networks
   293→   293→            max_retries = 5
   294→   294→            for retry in range(max_retries):
   295→   295→                if not missing_chunks:
   296→   296→                    break
   297→   297→
   298→   298→                if self.verbose:
   299→   299→                    click.echo(f"Retry {retry + 1}/{max_retries}: Requesting {len(missing_chunks)} missing chunks...")
   300→   300→
   301→   301→                # Exponential backoff: 0.5, 1.0, 2.0, 4.0, 8.0 seconds
   302→   302→                import time
   303→   303→                wait_time = 0.5 * (2 ** retry)
   304→   304→                time.sleep(wait_time)
   305→   305→
   306→   306→                # Request specific missing chunks
   307→   307→                chunks_retrieved_this_round = 0
   308→   308→                for missing_idx in missing_chunks[:]:
   309→   309→                    query = format_dns_query("g", session_id, missing_idx)
   310→   310→                    response = self._send_dns_query(query)
   311→   311→
   312→   312→                    if self._validate_chunk_response(response, missing_idx):
   313→   313→                        response_chunks[missing_idx] = response
   314→   314→                        missing_chunks.remove(missing_idx)
   315→   315→                        chunks_retrieved_this_round += 1
   316→   316→                        if self.verbose:
   317→   317→                            click.echo(f"  ✓ Retrieved missing chunk {missing_idx}")
   318→   318→                    elif self.verbose and response and response != "NOT_FOUND":
   319→   319→                        click.echo(f"  ✗ Invalid chunk {missing_idx}: {response[:30]}...")
   320→   320→
   321→   321→                if self.verbose:
   322→   322→                    click.echo(f"  Retrieved {chunks_retrieved_this_round} chunks this round")
   323→   323→
   324→   324→                if not missing_chunks:  # All chunks found
   325→   325→                    break
   326→   326→
   327→   327→            if self.verbose and missing_chunks:
   328→   328→                click.echo(f"Still missing {len(missing_chunks)} chunks after all retries: {missing_chunks}")
   329→   329→
   330→   330→        return response_chunks

   332→   332→    def _validate_chunk_response(self, response: str, expected_index: int) -> bool:
   333→   333→        """Validate chunk response format and integrity.
   334→   334→
   335→   335→        Args:
   336→   336→            response: Raw TXT record response
   337→   337→            expected_index: Expected chunk index
   338→   338→
   339→   339→        Returns:
   340→   340→            True if valid, False otherwise
   341→   341→        """
   342→   342→        if not response or response == "NOT_FOUND":
   343→   343→            return False
   344→   344→
   345→   345→        parts = response.split(':', 2)
   346→   346→        if len(parts) != 3:
   347→   347→            return False
   348→   348→
   349→   349→        try:
   350→   350→            actual_index = int(parts[0])
   351→   351→            total_chunks = int(parts[1])
   352→   352→            data = parts[2]
   353→   353→
   354→   354→            # Validate index matches expectation
   355→   355→            if actual_index != expected_index:
   356→   356→                return False
   357→   357→
   358→   358→            # Validate data is not empty
   359→   359→            if not data:
   360→   360→                return False
   361→   361→
   362→   362→            # Basic format validation - should be base64-like Fernet token
   363→   363→            if len(data) < 20:  # Fernet tokens are at least ~44 chars
   364→   364→                return False
   365→   365→
   366→   366→            return True
   367→   367→
   368→   368→        except (ValueError, IndexError):
   369→   369→            return False

   371→   371→    def _handle_traditional_response(self, session_id: str, show_spinner: bool) -> Optional[str]:
   372→   372→        """Handle traditional non-streaming response."""
   373→   373→        spinner = None
   374→   374→
   375→   375→        try:
   376→   376→            if show_spinner and not self.verbose:
   377→   377→                spinner = SimpleSpinner("-->")
   378→   378→                spinner.start()
   379→   379→            elif self.verbose:
   380→   380→                click.echo("Message sent, waiting for processing...")
   381→   381→
   382→   382→            # Wait longer for server to generate response, especially for long responses
   383→   383→            time.sleep(1)  # Increased wait time for server generation
   384→   384→
   385→   385→            if spinner:
   386→   386→                spinner.stop()
   387→   387→                spinner = SimpleSpinner("<--")
   388→   388→                spinner.start()
   389→   389→            elif self.verbose:
   390→   390→                click.echo("Retrieving response chunks...")
   391→   391→
   392→   392→            # Use the improved chunk retrieval method
   393→   393→            # Try multiple times until we get a complete response (marked with [EOS])
   394→   394→            response_chunks = {}
   395→   395→            max_attempts = 5
   396→   396→
   397→   397→            for attempt in range(max_attempts):
   398→   398→                response_chunks = self._get_current_response_chunks(session_id)
   399→   399→
   400→   400→                if response_chunks:
   401→   401→                    # Try to reassemble and check if it's complete
   402→   402→                    try:
   403→   403→                        reassembled = self.chunker.reassemble_response(response_chunks)
   404→   404→                        if reassembled:
   405→   405→                            decrypted = self.crypto.decrypt(reassembled)
   406→   406→                            if '[EOS]' in decrypted:
   407→   407→                                # Response is complete
   408→   408→                                break
   409→   409→                    except Exception:
   410→   410→                        pass  # Decryption failed, keep trying
   411→   411→
   412→   412→                if attempt < max_attempts - 1:  # Don't sleep on last attempt
   413→   413→                    if self.verbose:
   414→   414→                        click.echo(f"Response incomplete, waiting... (attempt {attempt + 1}/{max_attempts})")
   415→   415→                    time.sleep(2)  # Wait before retrying
   416→   416→
   417→   417→            if spinner:
   418→   418→                spinner.stop()
   419→   419→
   420→   420→            if not response_chunks:
   421→   421→                if not self.verbose:
   422→   422→                    click.echo("No response received from server")
   423→   423→                return None
   424→   424→
   425→   425→            if self.verbose:
   426→   426→                click.echo(f"Received {len(response_chunks)} response chunks")
   427→   427→
   428→   428→            try:
   429→   429→                complete_encrypted_response = self.chunker.reassemble_response(response_chunks)
   430→   430→                decrypted_response = self.crypto.decrypt(complete_encrypted_response)
   431→   431→                return decrypted_response
   432→   432→            except Exception as e:
   433→   433→                if self.verbose:
   434→   434→                    click.echo(f"Error decrypting response: {e}")
   435→   435→                    click.echo(f"Encrypted data length: {len(complete_encrypted_response) if complete_encrypted_response else 0}")
   436→   436→                    if complete_encrypted_response:
   437→   437→                        click.echo(f"Encrypted data starts with: {complete_encrypted_response[:50]}")
   438→   438→                return None
   439→   439→
   440→   440→        finally:
   441→   441→            if spinner:
   442→   442→                spinner.stop()

   479→   479→            self.sock.sendto(query_data, (self.server_host, self.server_port))
   480→   480→            response_data, _ = self.sock.recvfrom(4096)
   481→   481→
   482→   482→            # Restore original timeout
   483→   483→            self.sock.settimeout(original_timeout)
   484→   484→
   485→   485→            # If we got any response, connection is working
   486→   486→            return True
   487→   487→        except Exception:
   488→   488→            return False

   490→   490→    def cleanup_session(self):
   491→   491→        """Send cleanup signal to server to clear session data."""
   492→   492→        try:
   493→   493→            # Send a cleanup query to help server clear session data
   494→   494→            cleanup_query = format_dns_query("c", self.session_id)  # 'c' for cleanup
   495→   495→            self._send_dns_query(cleanup_query)
   496→   496→        except Exception:
   497→   497→            pass  # Cleanup is best-effort
   498→   498→
   499→   499→    def close(self):
   500→   500→        """Close the client connection."""
   501→   501→        self.cleanup_session()
   502→   502→        self.sock.close()
