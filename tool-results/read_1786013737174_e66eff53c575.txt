   129→   129→class CircuitState(Enum):
   130→   130→    """3-state circuit breaker: CLOSED → OPEN → HALF_OPEN → CLOSED"""
   131→   131→    CLOSED = 0      # Normal operation — requests flow through
   132→   132→    OPEN = 1        # Circuit tripped — all requests rejected
   133→   133→    HALF_OPEN = 2  # Probe state — one request allowed to test recovery
   134→   134→
   135→   135→
   136→   136→class ChannelCircuitBreaker:
   137→   137→    """Per-channel 3-state circuit breaker with Prometheus tracking.
   138→   138→
   139→   139→    States:
   140→   140→      CLOSED: Normal. All requests pass. Failure count tracked.
   141→   141→      OPEN: Circuit tripped. All requests fail-fast. Timer starts.
   142→   142→      HALF_OPEN: One probe request allowed. If success → CLOSED, if fail → OPEN.
   143→   143→
   144→   144→    Args:
   145→   145→        channel_name: Name for logging/metrics
   146→   146→        failure_threshold: Consecutive failures before opening (default 5)
   147→   147→        recovery_timeout: Seconds before half-open probe (default 30)
   148→   148→        success_threshold: Consecutive successes in HALF_OPEN to close (default 2)
   149→   149→    """
   150→   150→    def __init__(self, channel_name: str, failure_threshold: int = 5,
   151→   151→                 recovery_timeout: float = 30.0, success_threshold: int = 2):
   152→   152→        self.channel_name = channel_name
   153→   153→        self.failure_threshold = failure_threshold
   154→   154→        self.recovery_timeout = recovery_timeout
   155→   155→        self.success_threshold = success_threshold
   156→   156→        self._state = CircuitState.CLOSED
   157→   157→        self._failure_count = 0
   158→   158→        self._success_count = 0
   159→   159→        self._opened_at: float = 0.0
   160→   160→        self._lock = asyncio.Lock()
   161→   161→
   162→   162→    @property
   163→   163→    def state(self) -> CircuitState:
   164→   164→        """Current state, with automatic HALF_OPEN transition on timeout."""
   165→   165→        if self._state == CircuitState.OPEN:
   166→   166→            if time.time() - self._opened_at >= self.recovery_timeout:
   167→   167→                self._state = CircuitState.HALF_OPEN
   168→   168→        return self._state
   169→   169→    @property
   170→   170→    def is_open(self) -> bool:
   171→   171→        """True if circuit is OPEN (requests should be rejected)."""
   172→   172→        return self.state == CircuitState.OPEN
   173→   173→    async def allow_request(self) -> bool:
   174→   174→        """Check if a request should be allowed through."""
   175→   175→        async with self._lock:
   176→   176→            state = self.state
   177→   177→            if state == CircuitState.CLOSED:
   178→   178→                return True
   179→   179→            elif state == CircuitState.HALF_OPEN:
   180→   180→                # Allow ONE probe request through
   181→   181→                return True
   182→   182→            else:  # OPEN
   183→   183→                return False
   184→   184→    async def record_success(self):
   185→   185→        """Record a successful request."""
   186→   186→        async with self._lock:
   187→   187→            if self._state == CircuitState.HALF_OPEN:
   188→   188→                self._success_count += 1
   189→   189→                if self._success_count >= self.success_threshold:
   190→   190→                    self._state = CircuitState.CLOSED
   191→   191→                    self._failure_count = 0
   192→   192→                    self._success_count = 0
   193→   193→                    logger.info(f"Circuit CLOSED for {self.channel_name} — recovered")
   194→   194→            elif self._state == CircuitState.CLOSED:
   195→   195→                self._failure_count = 0  # Reset on success
   196→   196→    async def record_failure(self):
   197→   197→        """Record a failed request."""
   198→   198→        async with self._lock:
   199→   199→            if self._state == CircuitState.HALF_OPEN:
   200→   200→                # Probe failed → back to OPEN
   201→   201→                self._state = CircuitState.OPEN
   202→   202→                self._opened_at = time.time()
   203→   203→                self._success_count = 0
   204→   204→                logger.warning(f"Circuit back to OPEN for {self.channel_name} — probe failed")
   205→   205→            elif self._state == CircuitState.CLOSED:
   206→   206→                self._failure_count += 1
   207→   207→                if self._failure_count >= self.failure_threshold:
   208→   208→                    self._state = CircuitState.OPEN
   209→   209→                    self._opened_at = time.time()
   210→   210→                    logger.warning(
   211→   211→                        f"Circuit OPEN for {self.channel_name} — "
   212→   212→                        f"{self._failure_count} failures >= threshold {self.failure_threshold}"
   213→   213→                    )

   228→   228→@dataclass
   229→   229→class DomainPreference:
   230→   230→    """Per-domain channel preference with EMA-based learning.
   231→   231→
   232→   232→    Tracks success/failure counts per channel and uses exponential moving average
   233→   233→    to compute a quality score. The channel with the highest EMA score becomes
   234→   234→    the preferred_channel for subsequent requests to this domain.
   235→   235→    """
   236→   236→    domain: str
   237→   237→    successes: Dict[str, int] = field(default_factory=lambda: {})
   238→   238→    failures: Dict[str, int] = field(default_factory=lambda: {})
   239→   239→    preferred_channel: str = "http_proxy"
   240→   240→    last_updated: float = field(default_factory=time.time)
   241→   241→    # EMA scores per channel (0.0-1.0)
   242→   242→    _ema_scores: Dict[str, float] = field(default_factory=lambda: {})
   243→   243→    _ema_alpha: float = 0.3  # EMA smoothing factor (higher = more responsive)
   244→   244→
   245→   245→    def record_channel_result(self, channel: str, success: bool):
   246→   246→        """Record a channel result and update EMA score."""
   247→   247→        if success:
   248→   248→            self.successes[channel] = self.successes.get(channel, 0) + 1
   249→   249→        else:
   250→   250→            self.failures[channel] = self.failures.get(channel, 0) + 1
   251→   251→        # EMA update: new_score = alpha * sample + (1 - alpha) * old_score
   252→   252→        old_score = self._ema_scores.get(channel, 0.5)
   253→   253→        sample = 1.0 if success else 0.0
   254→   254→        self._ema_scores[channel] = self._ema_alpha * sample + (1 - self._ema_alpha) * old_score
   255→   255→        self.last_updated = time.time()
   256→   256→        # Re-compute preferred channel from EMA scores
   257→   257→        if self._ema_scores:
   258→   258→            best_channel = max(self._ema_scores, key=self._ema_scores.get)
   259→   259→            if self._ema_scores[best_channel] > 0.3:  # Minimum quality threshold
   260→   260→                self.preferred_channel = best_channel
   261→   261→
   262→   262→    def get_channel_score(self, channel: str) -> float:
   263→   263→        """Get EMA quality score for a channel (0.0-1.0)."""
   264→   264→        return self._ema_scores.get(channel, 0.5)

   317→   317→            self._key_errors[idx] = self._key_errors.get(idx, 0) + 1
   318→   318→            if status_code == 429:
   319→   319→                self._key_cooldown[idx] = time.time() + 60
   320→   320→            elif status_code in (401, 403):

   331→   331→            idx = (self._current_index + offset) % len(self._keys)

   364→   364→# 2. DNS Flood Protector (from v2)
   365→   365→# ═══════════════════════════════════════════════════════════════
   366→   366→
   367→   367→class DNSFloodProtector:
   368→   368→    """Token-bucket + per-client rate limiting for DNS queries."""
   369→   369→
   370→   370→    def __init__(self, max_qps: int = 50, burst: int = 100):
   371→   371→        self.max_qps = max_qps
   372→   372→        self.burst = burst
   373→   373→        self._tokens: float = float(burst)
   374→   374→        self._last_refill: float = time.time()
   375→   375→        self._lock = asyncio.Lock()
   376→   376→        self._blocked_count = 0
   377→   377→        self._client_queries: Dict[str, deque] = {}
   378→   378→
   379→   379→    async def allow(self, client_ip: str = "default") -> bool:
   380→   380→        async with self._lock:
   381→   381→            now = time.time()
   382→   382→            # Per-client rate check FIRST (before consuming global token)
   383→   383→            if client_ip not in self._client_queries:
   384→   384→                self._client_queries[client_ip] = deque(maxlen=100)
   385→   385→            self._client_queries[client_ip].append(now)
   386→   386→            recent = sum(1 for t in self._client_queries[client_ip] if now - t < 1.0)
   387→   387→            if recent > 10:
   388→   388→                self._blocked_count += 1
   389→   389→                DNS_FLOOD_BLOCKED.inc()
   390→   390→                return False
   391→   391→            # Global token bucket check (after per-client passes)
   392→   392→            elapsed = now - self._last_refill
   393→   393→            self._tokens = min(self.burst, self._tokens + elapsed * self.max_qps)
   394→   394→            self._last_refill = now
   395→   395→            if self._tokens >= 1.0:
   396→   396→                self._tokens -= 1.0
   397→   397→                return True
   398→   398→            self._blocked_count += 1
   399→   399→            DNS_FLOOD_BLOCKED.inc()
   400→   400→            return False

   432→   432→class ProxyPoolAdapter:
   433→   433→    """
   434→   434→    Python adapter for prox5-style proxy pool management.
   435→   435→    Loads proxies from file (prox5/proxies.txt format), validates,
   436→   436→    and rotates using Mystery Dialer pattern (retry on failure).
   437→   437→    """
   438→   438→
   439→   439→    def __init__(self, proxy_file: str = None, max_workers: int = 10):
   440→   440→        self._pool: List[ProxyEntry] = []
   441→   441→        self._current_index = 0
   442→   442→        self._lock = threading.Lock()
   443→   443→        self._max_workers = max_workers
   444→   444→        self._proxy_file = proxy_file
   445→   445→        if proxy_file and os.path.exists(proxy_file):
   446→   446→            self.load_from_file(proxy_file)
   447→   447→
   448→   448→    def load_from_file(self, filepath: str):
   449→   449→        """
   450→   450→        Load proxies from file. Supports formats:
   451→   451→          - host:port:user:pass  (autoclaw/prox5 format)
   452→   452→          - protocol://host:port  (URL format)
   453→   453→          - host:port  (simple format)
   454→   454→        """
   455→   455→        count = 0
   456→   456→        with open(filepath) as f:
   457→   457→            for line in f:
   458→   458→                line = line.strip()
   459→   459→                if not line or line.startswith('#'):
   460→   460→                    continue

  1258→  1258→    async def fetch(self, url: str, force_channel: str = None, **kwargs) -> ChannelResult:
  1259→  1259→        """
  1260→  1260→        Fetch a URL through the optimal channel with automatic failover.
  1261→  1261→        Now includes per-channel circuit breaker and EMA domain preference learning.
  1262→  1262→
  1263→  1263→        Channel selection priority:
  1264→  1264→          cached → http_proxy → socks_pool → dns_tunnel → mitm_stealth → connect_chain → http_direct
  1265→  1265→        """
  1266→  1266→        domain = self._extract_domain(url)
  1267→  1267→
  1268→  1268→        # Check cache
  1269→  1269→        cached = self._cache_get(url)
  1270→  1270→        if cached and not force_channel:
  1271→  1271→            return ChannelResult("cached", True, data=cached)
  1272→  1272→
  1273→  1273→        # Select channel (uses DomainPreference EMA if available)
  1274→  1274→        channel = self._select_channel(domain, force_channel)
  1275→  1275→
  1276→  1276→        # Channel implementations
  1277→  1277→        channel_map = {
  1278→  1278→            Channel.HTTP_PROXY: self._try_http_proxy,
  1279→  1279→            Channel.SOCKS_POOL: self._try_socks_pool,
  1280→  1280→            Channel.DNS_TUNNEL: self._try_dns_tunnel,
  1281→  1281→            Channel.MITM_STEALTH: self._try_mitm_stealth,
  1282→  1282→            Channel.CONNECT_CHAIN: self._try_connect_chain,
  1283→  1283→            Channel.HTTP_DIRECT: self._try_http_direct,
  1284→  1284→        }
  1285→  1285→
  1286→  1286→        # Try preferred channel (with circuit breaker check)
  1287→  1287→        if channel in channel_map:
  1288→  1288→            cb = self._circuit_breakers.get(channel.value)
  1289→  1289→            if cb and not await cb.allow_request():
  1290→  1290→                logger.debug(f"Circuit OPEN for {channel.value} — skipping")
  1291→  1291→            else:
  1292→  1292→                result = await channel_map[channel](url, domain, **kwargs)
  1293→  1293→                # Record result in circuit breaker and EMA
  1294→  1294→                if cb:
  1295→  1295→                    if result.success:
  1296→  1296→                        await cb.record_success()
  1297→  1297→                    else:
  1298→  1298→                        await cb.record_failure()
  1299→  1299→                # Update domain preference EMA
  1300→  1300→                pref = self._prefs.get(domain)
  1301→  1301→                if not pref:
  1302→  1302→                    pref = DomainPreference(domain=domain)
  1303→  1303→                    self._prefs[domain] = pref
  1304→  1304→                pref.record_channel_result(result.channel, result.success)
  1305→  1305→
  1306→  1306→                if result.success:
  1307→  1307→                    REQUESTS_TOTAL.labels(channel=result.channel, domain=domain, status="success").inc()
  1308→  1308→                    REQUESTS_DURATION.labels(channel=result.channel, domain=domain).observe(result.latency_ms / 1000)
  1309→  1309→                    return result
  1310→  1310→                # Record failure in Prometheus
  1311→  1311→                REQUESTS_TOTAL.labels(channel=result.channel, domain=domain, status="error").inc()
  1312→  1312→
  1313→  1313→        # Fallback: try all channels in priority order
  1314→  1314→        # NOTE: CONNECT_CHAIN was missing — added between MITM and HTTP_DIRECT
  1315→  1315→        fallback_order = [
  1316→  1316→            Channel.HTTP_PROXY, Channel.SOCKS_POOL, Channel.DNS_TUNNEL,
  1317→  1317→            Channel.MITM_STEALTH, Channel.CONNECT_CHAIN, Channel.HTTP_DIRECT,
  1318→  1318→        ]
  1319→  1319→        for ch in fallback_order:
  1320→  1320→            if ch == channel:
  1321→  1321→                continue  # Already tried
  1322→  1322→            if ch not in channel_map:
  1323→  1323→                continue
