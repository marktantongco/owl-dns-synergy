   351→# ─── Data Classes (memory-optimised) ──────────────────────────
   352→@dataclass(slots=True)
   353→class CachedResponse:
   354→    status: int
   355→    content: bytes
   356→    headers: Dict[str, str]
   357→    timestamp: float
   358→    ttl: int
   359→    protocol: str = "http/1.1"
   360→    def is_fresh(self) -> bool:
   361→        return time.time() - self.timestamp < self.ttl
   362→
   363→@dataclass(slots=True)
   364→class TokenBucket:
   365→    rate: float
   366→    capacity: float
   367→    tokens: float = 0.0
   368→    last_update: float = field(default_factory=time.time)
   369→    lock: asyncio.Lock = field(default_factory=asyncio.Lock)
   370→    async def _replenish(self):
   371→        now = time.time()
   372→        elapsed = now - self.last_update
   373→        async with self.lock:
   374→            self.tokens = min(self.capacity, self.tokens + elapsed * self.rate)
   375→            self.last_update = now
   376→    async def acquire(self, tokens: float = 1.0) -> bool:
   377→        await self._replenish()
   378→        async with self.lock:
   379→            if self.tokens >= tokens:
   380→                self.tokens -= tokens
   381→                return True
   382→        wait_time = (tokens - self.tokens) / self.rate
   383→        await asyncio.sleep(wait_time)
   384→        return await self.acquire(tokens)
   385→
   386→@dataclass(slots=True)
   387→class ProxyEntry:
   388→    url: str
   389→    healthy: bool = True
   390→    last_check: float = 0.0
   391→    fail_count: int = 0
   392→    ban_until: float = 0.0
   393→    def is_banned(self) -> bool:
   394→        return time.time() < self.ban_until
   395→    def mark_failed(self):
   396→        self.fail_count += 1
   397→        self.ban_until = time.time() + 60
   398→        self.healthy = False
   399→        logger.warning(f"Proxy banned (60s): {self.url}")
   400→    def mark_success(self):
   401→        self.fail_count = 0
   402→        self.healthy = True
   403→        self.last_check = time.time()
   404→
   405→# ─── HTTPCache (LRU + periodic cleanup) ──────────────────────
   406→class HTTPCache:
   407→    def __init__(self, ttl: int = DEFAULT_TTL, max_size: int = MAX_CACHED_RESPONSES):
   408→        self.ttl = ttl
   409→        self._max_size = max_size
   410→        self._memory: Dict[str, CachedResponse] = {}
   411→        self._lock = asyncio.Lock()
   412→        self._cleanup_task: Optional[asyncio.Task] = None
   413→
   414→    async def start_cleaner(self):
   415→        if self._cleanup_task is None or self._cleanup_task.done():
   416→            self._cleanup_task = asyncio.create_task(self._cleanup_loop())
   417→
   418→    async def _cleanup_loop(self):
   419→        while True:
   420→            await asyncio.sleep(60)
   421→            now = time.time()
   422→            async with self._lock:
   423→                expired = [k for k, v in self._memory.items() if not v.is_fresh()]
   424→                for k in expired:
   425→                    del self._memory[k]
   426→                if len(self._memory) > self._max_size:
   427→                    sorted_items = sorted(self._memory.items(), key=lambda x: x[1].timestamp)
   428→                    for k, _ in sorted_items[:len(self._memory)-self._max_size]:
   429→                        del self._memory[k]
   430→
   431→    def _key(self, method: str, url: str, params: Optional[Dict] = None, protocol: str = "http/1.1") -> str:
   432→        return hashlib.sha256(f"{method}:{url}:{json.dumps(params or {}, sort_keys=True)}:{protocol}".encode()).hexdigest()
   433→
   434→    async def get(self, method: str, url: str, params: Optional[Dict] = None, protocol: str = "http/1.1") -> Optional[CachedResponse]:
   435→        key = self._key(method, url, params, protocol)
   436→        if key in self._memory and self._memory[key].is_fresh():
   437→            return self._memory[key]
   438→        path = CACHE_DIR / f"{key}.json"
   439→        if path.exists():
   440→            try:
   441→                async with aiofiles.open(path, 'r') as f:
   442→                    data = json.loads(await f.read())
   443→                cached = CachedResponse(
   444→                    status=data["status"], content=data["content"].encode('utf-8', errors='replace'),
   445→                    headers=data["headers"], timestamp=data["timestamp"], ttl=data["ttl"], protocol=data.get("protocol", "http/1.1")
   446→                )
   447→                if cached.is_fresh():
   448→                    async with self._lock:
   449→                        self._memory[key] = cached
   450→                    return cached
   451→                else:
   452→                    path.unlink()
   453→            except Exception:
   454→                pass
   455→        return None

   467→# ─── RequestDeduplicator ──────────────────────────────────────
   468→class RequestDeduplicator:
   469→    def __init__(self):
   470→        self._in_flight: Dict[str, asyncio.Future] = {}
   471→        self._lock = asyncio.Lock()
   472→    def _key(self, method: str, url: str, params: Optional[Dict] = None, protocol: str = "http/1.1") -> str:
   473→        return hashlib.sha256(f"{method}:{url}:{json.dumps(params or {}, sort_keys=True)}:{protocol}".encode()).hexdigest()
   474→    async def execute(self, method: str, url: str, params: Optional[Dict], protocol: str, factory: Callable[[], Awaitable[CachedResponse]]) -> CachedResponse:
   475→        key = self._key(method, url, params, protocol)
   476→        async with self._lock:
   477→            if key in self._in_flight:
   478→                return await self._in_flight[key]
   479→            future = asyncio.Future()
   480→            self._in_flight[key] = future
   481→        try:
   482→            result = await factory()
   483→            future.set_result(result)
   484→            return result
   485→        except Exception as e:
   486→            future.set_exception(e)
   487→            raise
   488→        finally:
   489→            async with self._lock:
   490→                self._in_flight.pop(key, None)

   492→# ─── QualityScorer ─────────────────────────────────────────────
   493→class QualityScorer:
   494→    """Weighted scoring for proxies based on success rate and latency."""
   495→    def __init__(self, decay_factor: float = QUALITY_DECAY):
   496→        self._scores: Dict[str, float] = {}
   497→        self._history: Dict[str, List[float]] = {}  # last 100 latencies
   498→        self._decay = decay_factor
   499→
   500→    def update(self, proxy_url: str, success: bool, latency_ms: float = 9999.0):
   501→        old_score = self._scores.get(proxy_url, 0.5)
   502→        # Exponential moving average: success contributes 0.1 weight
   503→        new_score = old_score * self._decay + (1.0 if success else 0.0) * (1 - self._decay)
   504→        self._scores[proxy_url] = new_score
   505→        if proxy_url not in self._history:
   506→            self._history[proxy_url] = []
   507→        self._history[proxy_url].append(latency_ms)
   508→        if len(self._history[proxy_url]) > 100:
   509→            self._history[proxy_url] = self._history[proxy_url][-100:]
   510→
   511→    def get_score(self, proxy_url: str) -> float:
   512→        return self._scores.get(proxy_url, 0.5)
   513→
   514→    def get_best_proxy(self, proxies: List[str]) -> Optional[str]:
   515→        if not proxies:
   516→            return None
   517→        return max(proxies, key=lambda p: self.get_score(p))
   518→
   519→    def get_all_scores(self) -> Dict[str, float]:
   520→        return self._scores.copy()

   522→# ─── AdaptiveRateLimiter ──────────────────────────────────────
   523→class AdaptiveRateLimiter:
   524→    """Dynamically adjusts per-domain request rate based on response codes."""
   525→    def __init__(self, base_rate: float = DEFAULT_RATE,
   526→                 min_rate: float = ADAPTIVE_MIN_RATE,
   527→                 max_rate: float = ADAPTIVE_MAX_RATE):
   528→        self.base_rate = base_rate
   529→        self.min_rate = min_rate
   530→        self.max_rate = max_rate
   531→        self._rates: Dict[str, float] = {}
   532→        self._lock = asyncio.Lock()
   533→
   534→    async def adjust(self, domain: str, status: int):
   535→        async with self._lock:
   536→            current = self._rates.get(domain, self.base_rate)
   537→            if status in (429, 503):
   538→                # Too many requests – slow down aggressively
   539→                new_rate = max(self.min_rate, current * 0.5)
   540→            elif 200 <= status < 300:
   541→                # Success – speed up gradually
   542→                new_rate = min(self.max_rate, current * 1.1)
   543→            else:
   544→                # Other errors – keep same
   545→                new_rate = current
   546→            self._rates[domain] = new_rate
   547→            logger.debug(f"Rate for {domain}: {new_rate:.2f} req/s")
   548→
   549→    async def get_rate(self, domain: str) -> float:
   550→        return self._rates.get(domain, self.base_rate)
   551→
   552→    async def get_all_rates(self) -> Dict[str, float]:
   553→        return self._rates.copy()

   599→class ProxyPoolManager:
   600→    def __init__(self, max_queue: int = 50, cache_file: Path = PROXY_CACHE_FILE,
   601→                 cache_max: int = MAX_PROXY_CACHE, countries: Optional[List[str]] = None):

   603→        self._lock = asyncio.Lock()

   605→        self._cache_file = cache_file
   606→        self._cache_max = cache_max
   607→        self._proxies: List[ProxyEntry] = []

   612→    async def start(self):
   613→        self._running = True
   614→        self._load_cache()
   615→        for p in self._proxies:
   616→            if p.healthy and not p.is_banned():
   617→                await self.queue.put(p.url)

   620→    def _load_cache(self):
   621→        if not self._cache_file.exists():
   622→            return
   623→        try:
   624→            with open(self._cache_file, 'r') as f:
   625→                data = json.load(f)
   626→            for item in data.get("proxies", []):
   627→                p = ProxyEntry(
   628→                    url=item["url"],
   629→                    healthy=item.get("healthy", True),
   630→                    last_check=item.get("last_check", 0.0),
   631→                    fail_count=item.get("fail_count", 0),
   632→                    ban_until=item.get("ban_until", 0.0)
   633→                )

   635→                self._url_set.add(p.url)
   636→            if len(self._proxies) > self._cache_max:
   637→                self._proxies = self._proxies[-self._cache_max:]
   638→                self._url_set = {p.url for p in self._proxies}
   639→            logger.info(f"Loaded {len(self._proxies)} proxies from cache")
   640→        except Exception as e:
   641→            logger.warning(f"Failed to load proxy cache: {e}")
   642→
   643→    def _save_cache(self):
   644→        data = {
   645→            "proxies": [
   646→                {
   647→                    "url": p.url,
   648→                    "healthy": p.healthy,
   649→                    "last_check": p.last_check,
   650→                    "fail_count": p.fail_count,
   651→                    "ban_until": p.ban_until
   652→                } for p in self._proxies
   653→            ]
   654→        }
   655→        try:
   656→            with open(self._cache_file, 'w') as f:

   661→    async def _discovery_loop(self):
   662→        while self._running:
   663→            try:
   664→                async for proxy in self._broker.find(
   665→                    types=['HTTP', 'HTTPS', 'SOCKS4', 'SOCKS5'],
   666→                    anonymity=['high', 'anonymous'],
   667→                    country=self.countries,
   668→                    limit=0
   669→                ):
   670→                    if not self._running:
   671→                        break
   672→                    url = f"{proxy.protocol}://{proxy.host}:{proxy.port}"
   673→                    if url not in self._url_set:
   674→                        p = ProxyEntry(url=url)
   675→                        self._proxies.append(p)
   676→                        self._url_set.add(url)
   677→                        await self.queue.put(url)
   678→                        logger.debug(f"Discovered proxy: {url} (country: {proxy.country})")
   679→                        if len(self._proxies) > self._cache_max:
   680→                            self._proxies = self._proxies[-self._cache_max:]
   681→                            self._url_set = {p.url for p in self._proxies}
   682→                        self._save_cache()
   683→            except Exception as e:
   684→                logger.error(f"ProxyBroker2 discovery error: {e}")
   685→                await asyncio.sleep(10)

   687→    async def get(self) -> Optional[str]:
   688→        try:
   689→            url = await asyncio.wait_for(self.queue.get(), timeout=5.0)
   690→            return url
   691→        except asyncio.TimeoutError:
   692→            return None

   695→        return [p.url for p in self._proxies if p.healthy and not p.is_banned()]

   697→    def stop(self):
   698→        self._running = False
   699→        self._save_cache()

   885→            )

   888→        if cached:
   889→            return cached
   890→
   891→        async def factory():

   895→    async def _execute_with_retry(self, method, url, params, headers, **kwargs):
   896→        domain = urlparse(url).netloc
   897→        breaker = self.circuit_breakers.get(domain)
   898→
   899→        if not breaker.can_execute():
   900→            raise RuntimeError(f"Circuit breaker open for {domain}")
   901→
   902→        # Get adaptive rate for this domain
   903→        rate = await self.limiter.get_rate(domain)
   904→        # Use a token bucket with the adaptive rate (simplified: just sleep)
   905→        # In production, use a proper token bucket with dynamic rate
   906→
   907→        for attempt in range(self.max_retries):
   908→            start = time.time()
   909→            try:
   910→                response = await self._execute_with_resilient_client(method, url, params, headers, **kwargs)
   911→
   912→                if self.use_curl_cffi:
   913→                    content = response.content
   914→                    status = response.status_code
   915→                    resp_headers = dict(response.headers)
   916→                else:
   917→                    content = await response.aread()
   918→                    status = response.status_code
   919→                    resp_headers = dict(response.headers)
   920→
   921→                latency = (time.time() - start) * 1000
   922→
   923→                cached_response = CachedResponse(
   924→                    status=status,
   925→                    content=content,
   926→                    headers=resp_headers,
   927→                    timestamp=time.time(),
   928→                    ttl=self.cache.ttl
   929→                )
   930→
   931→                # Update quality scorer with success
   932→                proxy_url = kwargs.get('proxy_url') or await self._get_best_proxy()
   933→                if proxy_url:
   934→                    self.scorer.update(proxy_url, success=True, latency_ms=latency)
   935→
   936→                # Update adaptive rate limiter
   937→                await self.limiter.adjust(domain, status)
   938→
   939→                breaker.success()
   940→
   941→                # --- Retry-After handling ---
   942→                if status in (429, 503) and 'retry-after' in resp_headers:
   943→                    retry_after = resp_headers['retry-after']
   944→                    try:
   945→                        seconds = int(retry_after)
   946→                    except ValueError:
   947→                        try:
   948→                            retry_date = email.utils.parsedate_to_datetime(retry_after)
   949→                            seconds = max(0, (retry_date - datetime.datetime.now(datetime.timezone.utc)).total_seconds())
   950→                        except Exception:
   951→                            seconds = 5
   952→                    logger.info(f"Retry-After: sleeping {seconds}s before retry")
   953→                    await asyncio.sleep(seconds)
   954→                    continue
   955→
   956→                await self.cache.set(method, url, cached_response, params)
   957→                await self._save_state()
   958→                return cached_response
   959→
   960→            except Exception as e:
   961→                breaker.failure()
   962→                proxy_url = kwargs.get('proxy_url') or await self._get_best_proxy()
   963→                if proxy_url:
   964→                    self.scorer.update(proxy_url, success=False)
   965→                await self.limiter.adjust(domain, 500)
   966→                logger.warning(f"Request failed (attempt {attempt+1}/{self.max_retries}): {e}")
   967→                continue
   968→
   969→        # Direct fallback
   970→        logger.info("All proxies exhausted, attempting direct connection...")
   971→        try:

   989→            await self.cache.set(method, url, cached_response, params)
   990→            return cached_response
   991→        except Exception as e:
