   149→# 1. OpenRouter Key Rotator (from v2)
   150→# ═══════════════════════════════════════════════════════════════
   151→
   152→class OpenRouterKeyRotator:
   153→    """Round-robin API key rotation with failover on 429/401/403."""
   154→
   155→    def __init__(self, keys: List[str] = None, base_url: str = "https://openrouter.ai/api/v1"):
   156→        self._keys = keys or []
   157→        self._current_index = 0
   158→        self._base_url = base_url
   159→        self._key_errors: Dict[int, int] = {}
   160→        self._key_cooldown: Dict[int, float] = {}
   161→        self._lock = threading.Lock()
   162→
   163→    @classmethod
   164→    def from_env(cls) -> "OpenRouterKeyRotator":
   165→        keys = []
   166→        primary = os.getenv("OPENAI_API_KEY")
   167→        if primary:
   168→            keys.append(primary)
   169→        for i in range(1, 10):
   170→            key = os.getenv(f"OPENROUTER_KEY_{i}")
   171→            if key:
   172→                keys.append(key)
   173→        base_url = os.getenv("OPENAI_BASE_URL", "https://openrouter.ai/api/v1")
   174→        return cls(keys=keys, base_url=base_url)
   175→
   176→    @property
   177→    def base_url(self) -> str:
   178→        return self._base_url
   179→
   180→    def get_active_key(self) -> Optional[str]:
   181→        with self._lock:
   182→            if not self._keys:
   183→                return None
   184→            now = time.time()
   185→            if now >= self._key_cooldown.get(self._current_index, 0):
   186→                return self._keys[self._current_index]
   187→            for offset in range(len(self._keys)):
   188→                idx = (self._current_index + offset) % len(self._keys)
   189→                if now >= self._key_cooldown.get(idx, 0):
   190→                    self._current_index = idx
   191→                    KEY_ROTATION_COUNT.inc()
   192→                    return self._keys[idx]
   193→            return None
   194→
   195→    def report_error(self, status_code: int, error_type: str = "unknown"):
   196→        with self._lock:
   197→            idx = self._current_index
   198→            self._key_errors[idx] = self._key_errors.get(idx, 0) + 1
   199→            if status_code == 429:
   200→                self._key_cooldown[idx] = time.time() + 60
   201→            elif status_code in (401, 403):
   202→                self._key_cooldown[idx] = time.time() + 300
   203→            else:
   204→                self._key_cooldown[idx] = time.time() + 10
   205→            self._rotate_to_next()
   206→
   207→    def _rotate_to_next(self):
   208→        if len(self._keys) <= 1:
   209→            return
   210→        now = time.time()
   211→        for offset in range(1, len(self._keys)):
   212→            idx = (self._current_index + offset) % len(self._keys)
   213→            if now >= self._key_cooldown.get(idx, 0):
   214→                self._current_index = idx
   215→                KEY_ROTATION_COUNT.inc()
   216→                return
   217→        self._current_index = (self._current_index + 1) % len(self._keys)
   218→        KEY_ROTATION_COUNT.inc()
   219→
   220→    @property
   221→    def total_keys(self) -> int:
   222→        return len(self._keys)
   223→
   224→    @property
   225→    def available_keys(self) -> int:
   226→        now = time.time()
   227→        return sum(1 for i in range(len(self._keys))
   228→                   if now >= self._key_cooldown.get(i, 0))
   229→
   230→    def get_status(self) -> Dict[str, Any]:
   231→        now = time.time()
   232→        return {
   233→            "total_keys": self.total_keys,
   234→            "available_keys": self.available_keys,
   235→            "current_index": self._current_index,
   236→            "key_errors": dict(self._key_errors),
   237→            "cooldowns": {
   238→                i: max(0, self._key_cooldown.get(i, 0) - now)
   239→                for i in range(len(self._keys))
   240→            }
   241→        }

   245→# 2. DNS Flood Protector (from v2)
   246→# ═══════════════════════════════════════════════════════════════
   247→
   248→class DNSFloodProtector:
   249→    """Token-bucket + per-client rate limiting for DNS queries."""
   250→
   251→    def __init__(self, max_qps: int = 50, burst: int = 100):
   252→        self.max_qps = max_qps
   253→        self.burst = burst
   254→        self._tokens: float = float(burst)
   255→        self._last_refill: float = time.time()
   256→        self._lock = asyncio.Lock()
   257→        self._blocked_count = 0
   258→        self._client_queries: Dict[str, deque] = {}
   259→
   260→    async def allow(self, client_ip: str = "default") -> bool:
   261→        async with self._lock:
   262→            now = time.time()
   263→            elapsed = now - self._last_refill
   264→            self._tokens = min(self.burst, self._tokens + elapsed * self.max_qps)
   265→            self._last_refill = now
   266→            if self._tokens >= 1.0:
   267→                self._tokens -= 1.0
   268→            else:
   269→                self._blocked_count += 1
   270→                DNS_FLOOD_BLOCKED.inc()
   271→                return False
   272→            if client_ip not in self._client_queries:
   273→                self._client_queries[client_ip] = deque(maxlen=100)
   274→            self._client_queries[client_ip].append(now)
   275→            recent = sum(1 for t in self._client_queries[client_ip] if now - t < 1.0)
   276→            if recent > 10:
   277→                self._blocked_count += 1
   278→                DNS_FLOOD_BLOCKED.inc()
   279→                return False
   280→            return True

   284→# 3. Proxy Pool Adapter — prox5-compatible SOCKS/HTTP pool
   285→# ═══════════════════════════════════════════════════════════════
   286→
   287→@dataclass
   288→class ProxyEntry:
   289→    """Represents a single proxy endpoint (prox5-compatible)."""
   290→    endpoint: str           # host:port
   291→    protocol: str = "socks5"  # socks5, socks4, http
   292→    username: str = ""
   293→    password: str = ""
   294→    proxied_ip: str = ""
   295→    last_validated: float = 0.0
   296→    successes: int = 0
   297→    failures: int = 0
   298→    score: float = 1.0      # EMA quality score
   299→    region: str = ""
   300→
   301→    @property
   302→    def url(self) -> str:
   303→        if self.username and self.password:
   304→            return f"{self.protocol}://{self.username}:{self.password}@{self.endpoint}"
   305→        return f"{self.protocol}://{self.endpoint}"
   306→
   307→    @property
   308→    def is_stale(self) -> bool:
   309→        return time.time() - self.last_validated > 1800  # 30 min
   310→
   311→
   312→class ProxyPoolAdapter:
   313→    """
   314→    Python adapter for prox5-style proxy pool management.
   315→    Loads proxies from file (prox5/proxies.txt format), validates,
   316→    and rotates using Mystery Dialer pattern (retry on failure).
   317→    """
   318→
   319→    def __init__(self, proxy_file: str = None, max_workers: int = 10):
   320→        self._pool: List[ProxyEntry] = []
   321→        self._current_index = 0
   322→        self._lock = threading.Lock()
   323→        self._max_workers = max_workers
   324→        self._proxy_file = proxy_file
   325→        if proxy_file and os.path.exists(proxy_file):
   326→            self.load_from_file(proxy_file)
   327→
   328→    def load_from_file(self, filepath: str):
   329→        """
   330→        Load proxies from file. Supports formats:
   331→          - host:port:user:pass  (autoclaw/prox5 format)
   332→          - protocol://host:port  (URL format)
   333→          - host:port  (simple format)
   334→        """
   335→        count = 0
   336→        with open(filepath) as f:
   337→            for line in f:
   338→                line = line.strip()
   339→                if not line or line.startswith('#'):
   340→                    continue
   341→                entry = self._parse_proxy_line(line)
   342→                if entry:
   343→                    self._pool.append(entry)
   344→                    count += 1
   345→        logger.info(f"Loaded {count} proxies from {filepath}")
   346→        PROXY_POOL_SIZE.labels(protocol="all").set(len(self._pool))
   347→
   348→    def _parse_proxy_line(self, line: str) -> Optional[ProxyEntry]:
   349→        """Parse a proxy line in any supported format."""
   350→        if '://' in line:
   351→            # URL format: socks5://user:pass@host:port
   352→            parsed = urlparse(line)
   353→            protocol = parsed.scheme or "socks5"
   354→            host = parsed.hostname or ""
   355→            port = parsed.port or 1080
   356→            return ProxyEntry(
   357→                endpoint=f"{host}:{port}",
   358→                protocol=protocol,
   359→                username=parsed.username or "",
   360→                password=parsed.password or "",
   361→            )
   362→        parts = line.split(':')
   363→        if len(parts) == 4:
   364→            # host:port:user:pass format
   365→            return ProxyEntry(
   366→                endpoint=f"{parts[0]}:{parts[1]}",
   367→                username=parts[2],
   368→                password=parts[3],
   369→            )
   370→        if len(parts) == 2:
   371→            try:
   372→                int(parts[1])
   373→                return ProxyEntry(endpoint=line)
   374→            except ValueError:
   375→                pass
   376→        return None
   377→
   378→    def add_proxy(self, endpoint: str, protocol: str = "socks5",
   379→                  username: str = "", password: str = ""):
   380→        with self._lock:
   381→            self._pool.append(ProxyEntry(
   382→                endpoint=endpoint, protocol=protocol,
   383→                username=username, password=password
   384→            ))
   385→
   386→    def get_next(self) -> Optional[ProxyEntry]:
   387→        """Get next proxy using round-robin with Mystery Dialer retry."""
   388→        with self._lock:
   389→            if not self._pool:
   390→                return None
   391→            # Try current proxy first
   392→            for offset in range(len(self._pool)):
   393→                idx = (self._current_index + offset) % len(self._pool)
   394→                proxy = self._pool[idx]
   395→                if proxy.score > 0.2 and not proxy.is_stale:
   396→                    self._current_index = idx
   397→                    return proxy
   398→            # All stale/low-score, return best available
   399→            best = max(self._pool, key=lambda p: p.score)
   400→            return best
   401→
   402→    def report_success(self, endpoint: str, latency_ms: float = 0):
   403→        with self._lock:
   404→            for p in self._pool:
   405→                if p.endpoint == endpoint:
   406→                    p.successes += 1
   407→                    p.last_validated = time.time()
   408→                    # EMA update: alpha=0.3
   409→                    p.score = 0.3 * 1.0 + 0.7 * p.score
   410→                    break
   411→
   412→    def report_failure(self, endpoint: str):
   413→        with self._lock:
   414→            for p in self._pool:
   415→                if p.endpoint == endpoint:
   416→                    p.failures += 1
   417→                    p.score = 0.3 * 0.0 + 0.7 * p.score
   418→                    break
   419→
   420→    @property
   421→    def size(self) -> int:
   422→        return len(self._pool)
   423→
   424→    @property
   425→    def valid_count(self) -> int:
   426→        return sum(1 for p in self._pool if p.score > 0.2 and not p.is_stale)
   427→
   428→    def get_status(self) -> Dict[str, Any]:
   429→        return {
   430→            "total": self.size,
   431→            "valid": self.valid_count,
   432→            "current_index": self._current_index,
   433→            "protocols": {
   434→                proto: sum(1 for p in self._pool if p.protocol == proto)
   435→                for proto in set(p.protocol for p in self._pool)
   436→            },
   437→            "top_5": [
   438→                {"endpoint": p.endpoint, "protocol": p.protocol,
   439→                 "score": round(p.score, 3), "successes": p.successes,
   440→                 "failures": p.failures}
   441→                for p in sorted(self._pool, key=lambda x: x.score, reverse=True)[:5]
   442→            ]
   443→        }

   486→    def __init__(self, binary_path: str = None, config_path: str = None):

   492→        self._config = StealthProxyConfig()

   494→    def _find_binary(self) -> Optional[str]:
   495→        """Find https_proxy binary in PATH or repos."""

   497→            os.path.expanduser("~/my-project/repos/https_proxy/target/release/https_proxy"),
   498→            os.path.expanduser("~/.owl-dns-synergy/bin/https_proxy"),
   499→            "/usr/local/bin/https_proxy",

   508→        yaml_content = config.to_yaml()
   509→        os.makedirs(os.path.dirname(self._config_path), exist_ok=True)
   510→        with open(self._config_path, 'w') as f:
   511→            f.write(yaml_content)
   512→        return yaml_content

   514→    def start(self) -> bool:
   515→        """Start the https_proxy process."""
   516→        if not self._binary:
   517→            logger.warning("https_proxy binary not found — stealth proxy disabled")
   518→            return False
   519→        try:
   520→            self._process = subprocess.Popen(
   521→                [self._binary, "run", "-c", self._config_path],
   522→                stdout=subprocess.PIPE,
   523→                stderr=subprocess.PIPE,

   525→            logger.info(f"https_proxy started (PID {self._process.pid})")
   526→            return True
   527→        except Exception as e:
   528→            logger.error(f"Failed to start https_proxy: {e}")
   529→            return False
   530→
   531→    def stop(self):
   532→        if self._process and self._process.poll() is None:
   533→            self._process.terminate()
   534→            self._process.wait(timeout=5)
   535→            logger.info("https_proxy stopped")
   536→
   537→    @property
   538→    def is_running(self) -> bool:
   539→        return self._process is not None and self._process.poll() is None

   547→        return f"http://{host}:{port}"

   570→            if os.path.exists(path):
   571→                return path
   572→        return None
   573→
   574→    def build_command(
   575→        self,
   576→        proxy: str,
   577→        destination: str,
   578→        remproxy: str = None,
   579→        proxyauth: str = None,
   580→        remproxyauth: str = None,
   581→        encrypt: bool = False,
   582→        encrypt_proxy: bool = False,
   583→        encrypt_remproxy: bool = False,
   584→        ntlm: bool = False,
   585→        custom_headers: List[str] = None,
   586→    ) -> List[str]:
   587→        """
   588→        Build proxytunnel command with full flag support.
   589→        Chain: Client → proxy (-p) → [remproxy (-r)] → destination (-d)
   590→        """
   591→        if not self._binary:
   592→            return []
   593→
   594→        cmd = [self._binary, "-p", proxy, "-d", destination]
   595→        if remproxy:
   596→            cmd.extend(["-r", remproxy])
   597→        if proxyauth:
   598→            cmd.extend(["-P", proxyauth])
   599→        if remproxyauth:
   600→            cmd.extend(["-R", remproxyauth])
   601→        if encrypt:
   602→            cmd.append("-e")
   603→        if encrypt_proxy:
   604→            cmd.append("-E")
   605→        if encrypt_remproxy:
   606→            cmd.append("-X")
   607→        if ntlm:
   608→            cmd.append("-N")
   609→        if custom_headers:
   610→            for h in custom_headers:
   611→                cmd.extend(["-H", h])
   612→        return cmd
   613→
   614→    async def tunnel(
   615→        self,
   616→        proxy: str,
   617→        destination: str,
   618→        **kwargs
   619→    ) -> ChannelResult:
   620→        """Execute proxytunnel to establish a CONNECT tunnel."""
   621→        start = time.time()
   622→        cmd = self.build_command(proxy, destination, **kwargs)
   623→        if not cmd:
   624→            return ChannelResult("connect_chain", False, error="proxytunnel binary not found")
   625→
   626→        try:
   627→            proc = await asyncio.create_subprocess_exec(
   628→                *cmd,
   629→                stdout=asyncio.subprocess.PIPE,
   630→                stderr=asyncio.subprocess.PIPE,
   631→            )
   632→            await asyncio.wait_for(proc.wait(), timeout=30)
   633→            latency = (time.time() - start) * 1000
   634→            chain_depth = 2 if kwargs.get('remproxy') else 1
   635→            CONNECT_TUNNELS.labels(chain_depth=str(chain_depth)).inc()
   636→
   637→            if proc.returncode == 0:
   638→                return ChannelResult(
   639→                    "connect_chain", True, latency_ms=latency, chain_depth=chain_depth
   640→                )
   641→            else:
   642→                stderr = await proc.stderr.read()
   643→                return ChannelResult(
   644→                    "connect_chain", False, error=stderr.decode()[:200], latency_ms=latency
   645→                )
   646→        except asyncio.TimeoutError:
   647→            return ChannelResult("connect_chain", False, error="proxytunnel timeout (30s)")
   648→        except Exception as e:
   649→            return ChannelResult("connect_chain", False, error=str(e))

  1045→    async def fetch(self, url: str, force_channel: str = None, **kwargs) -> ChannelResult:
  1046→        """
  1047→        Fetch a URL through the optimal channel with automatic failover.
  1048→
  1049→        Channel selection priority:
  1050→          cached → http_proxy → socks_pool → dns_tunnel → mitm_stealth → connect_chain → http_direct
  1051→        """
  1052→        domain = self._extract_domain(url)
  1053→
  1054→        # Check cache
  1055→        cached = self._cache_get(url)
  1056→        if cached and not force_channel:
  1057→            return ChannelResult("cached", True, data=cached)
  1058→
  1059→        # Select channel
  1060→        channel = self._select_channel(domain, force_channel)
  1061→
  1062→        # Channel implementations
  1063→        channel_map = {
  1064→            Channel.HTTP_PROXY: self._try_http_proxy,
  1065→            Channel.SOCKS_POOL: self._try_socks_pool,
  1066→            Channel.DNS_TUNNEL: self._try_dns_tunnel,
  1067→            Channel.MITM_STEALTH: self._try_mitm_stealth,
  1068→            Channel.CONNECT_CHAIN: self._try_connect_chain,
  1069→            Channel.HTTP_DIRECT: self._try_http_direct,
  1070→        }
  1071→
  1072→        # Try preferred channel
  1073→        if channel in channel_map:
  1074→            result = await channel_map[channel](url, domain, **kwargs)
  1075→            if result.success:
  1076→                REQUESTS_TOTAL.labels(channel=result.channel, domain=domain, status="success").inc()
  1077→                REQUESTS_DURATION.labels(channel=result.channel, domain=domain).observe(result.latency_ms / 1000)
  1078→                return result
  1079→
  1080→        # Fallback: try all channels in priority order
  1081→        fallback_order = [
  1082→            Channel.HTTP_PROXY, Channel.SOCKS_POOL, Channel.DNS_TUNNEL,
  1083→            Channel.MITM_STEALTH, Channel.HTTP_DIRECT,
  1084→        ]
  1085→        for ch in fallback_order:
  1086→            if ch == channel:
  1087→                continue  # Already tried
  1088→            if ch in channel_map:
  1089→                result = await channel_map[ch](url, domain, **kwargs)
  1090→                if result.success:
  1091→                    CHANNEL_SWITCHES.labels(
  1092→                        from_channel=channel.value, to_channel=ch.value
  1093→                    ).inc()
  1094→                    REQUESTS_TOTAL.labels(channel=result.channel, domain=domain, status="success").inc()
  1095→                    return result
  1096→
  1097→        REQUESTS_TOTAL.labels(channel="none", domain=domain, status="all_failed").inc()
  1098→        return ChannelResult("none", False, error=f"All channels exhausted for {domain}")
