    49→    49→def get_next_token():
    50→    50→    """Round-robin token selection across all accounts.
    51→    51→    Auto-refreshes expired tokens. Skips known-exhausted accounts (from cache only)."""
    52→    52→    global _token_idx
    53→    53→    data = load_tokens()
    54→    54→    if not data["accounts"]:
    55→    55→        return None, None
    56→    56→
    57→    57→    n = len(data["accounts"])
    58→    58→    with _token_lock:
    59→    59→        idx = _token_idx % n
    60→    60→        _token_idx += 1
    61→    61→
    62→    62→    # Try each account starting from idx
    63→    63→    for i in range(n):
    64→    64→        acc = data["accounts"][(idx + i) % n]
    65→    65→        # Check token validity
    66→    66→        from config import ACCESS_TOKEN_TTL, REFRESH_MARGIN
    67→    67→        age = time.time() - acc.get("last_refreshed", 0)
    68→    68→        if age < ACCESS_TOKEN_TTL - REFRESH_MARGIN:
    69→    69→            # Token still valid — check exhausted cache (non-blocking, cached only)
    70→    70→            if not _is_cached_exhausted(acc):
    71→    71→                return acc["access_token"], acc
    72→    72→            else:
    73→    73→                continue
    74→    74→        # Try refresh
    75→    75→        from auth import refresh_token
    76→    76→        new_token = refresh_token(acc)
    77→    77→        if new_token:
    78→    78→            if not _is_cached_exhausted(acc):
    79→    79→                return new_token, acc
    80→    80→            else:
    81→    81→                continue
    82→    82→
    83→    83→    # Fallback: return first valid token even if exhausted
    84→    84→    for acc in data["accounts"]:
    85→    85→        return acc["access_token"], acc
    86→    86→    return None, None

   487→   487→@app.route("/refresh-all", methods=["POST"])
   488→   488→def do_refresh_all():
   489→   489→    # If already running, return current progress
   490→   490→    if _refresh_progress["running"]:
   491→   491→        return jsonify({"status": "running", **_refresh_progress})
   492→   492→
   493→   493→    # Backup tokens.json before refresh (preventive)
   494→   494→    import shutil
   495→   495→    try:
   496→   496→        shutil.copy2(TOKENS_FILE_FULL, TOKENS_FILE_FULL + ".bak")
   497→   497→    except Exception as e:
   498→   498→        print(f"[proxy] Backup before refresh-all failed: {e}")
   499→   499→
   500→   500→    # Start refresh in background thread
   501→   501→    data = load_tokens()
   502→   502→    _refresh_progress["total"] = len(data["accounts"])
   503→   503→    _refresh_progress["done"] = 0
   504→   504→    _refresh_progress["success"] = 0
   505→   505→    _refresh_progress["fail"] = 0
   506→   506→    _refresh_progress["running"] = True
   507→   507→
   508→   508→    def _bg_refresh():
   509→   509→        import auth as _auth
   510→   510→        from config import REFRESH_URL
   511→   511→        import requests as _req
   512→   512→        d = _auth.load_tokens()
   513→   513→        now = int(time.time())
   514→   514→        for acc in d["accounts"]:
   515→   515→            try:
   516→   516→                headers = _auth._sign_headers()
   517→   517→                body = {
   518→   518→                    "source_id": acc.get("source_id", "autoclaw"),
   519→   519→                    "device_id": acc["device_id"],
   520→   520→                    "refresh_token": acc["refresh_token"],
   521→   521→                }
   522→   522→                resp = _req.post(REFRESH_URL, json=body, headers=headers, timeout=15, verify=False)
   523→   523→                resp_data = resp.json()
   524→   524→                if resp_data.get("code") == 0 and "data" in resp_data:
   525→   525→                    new_access = resp_data["data"].get("access_token")
   526→   526→                    new_refresh = resp_data["data"].get("refresh_token", acc["refresh_token"])
   527→   527→                    if new_access:
   528→   528→                        acc["access_token"] = new_access
   529→   529→                        if new_refresh:
   530→   530→                            acc["refresh_token"] = new_refresh
   531→   531→                        acc["last_refreshed"] = now
   532→   532→                        _refresh_progress["success"] += 1
   533→   533→                    else:
   534→   534→                        _refresh_progress["fail"] += 1
   535→   535→                else:
   536→   536→                    _refresh_progress["fail"] += 1
   537→   537→            except Exception:
   538→   538→                _refresh_progress["fail"] += 1
   539→   539→            _refresh_progress["done"] += 1
   540→   540→        # Save once at end
   541→   541→        _auth.save_tokens(d)
   542→   542→        _refresh_progress["running"] = False
   543→   543→        print(f"[proxy] Refresh all done: {_refresh_progress['success']} ok, {_refresh_progress['fail']} fail")
   544→   544→
   545→   545→    t = threading.Thread(target=_bg_refresh, daemon=True)
   546→   546→    t.start()
   547→   547→
   548→   548→    return jsonify({"status": "started", **_refresh_progress})

   616→   616→@app.route("/auth/callback-google")
   617→   617→def auth_callback_google():
   618→   618→    """Google OAuth redirect target — auto-captures code+state, exchanges tokens."""
   619→   619→    from auth import google_oauth_login, add_token
   620→   620→
   621→   621→    code = request.args.get("code")
   622→   622→    state = request.args.get("state")
   623→   623→    error = request.args.get("error")
   624→   624→
   625→   625→    # Find pending login by state — MUST match, no fallback in concurrent mode
   626→   626→    pending = _pending_logins.get(state)
   627→   627→    if not pending:
   628→   628→        # No state match — do NOT fallback to "first pending" (causes cross-account
   629→   629→        # contamination in concurrent mode). Just fail this callback.
   630→   630→        print(f"[callback] Unknown state={state[:20]}... (not in {len(_pending_logins)} pending)")
   631→   631→        return "<html><body><h1>Login Failed</h1><p>Unknown state</p></body></html>", 400
   632→   632→
   633→   633→    if error:
   634→   634→        pending["error"] = error
   635→   635→        return f"<html><body><h1>Login Failed</h1><p>{error}</p></body></html>", 400
   636→   636→
   637→   637→    if not code:
   638→   638→        pending["error"] = "No code in callback"
   639→   639→        return "<html><body><h1>Login Failed</h1><p>No code</p></body></html>", 400
   640→   640→
   641→   641→    device_id = pending.get("device_id")
   642→   642→    proxy_used = pending.get("proxy")  # Reuse same proxy from URL generation
   643→   643→
   644→   644→    # Serialize token exchanges — AutoClaw API rate-limits per IP (630014).
   645→   645→    # Without this, 3 concurrent callbacks all hit the API simultaneously → all fail.
   646→   646→    # Lock ensures 1 exchange at a time, with small gap between.
   647→   647→    with _token_exchange_lock:
   648→   648→        print(f"[callback] Exchanging token for state={state[:12]}... device_id={device_id[:8]}... proxy={'yes' if proxy_used else 'none'}")
   649→   649→        result = google_oauth_login(code, state, device_id, proxy=proxy_used)
   650→   650→        time.sleep(2)  # Gap so next concurrent callback doesn't hit 630014
   651→   651→    if not result:
   652→   652→        print(f"[callback] Token exchange FAILED for state={state[:12]}... (google_oauth_login returned None)")
   653→   653→        pending["error"] = "Token exchange failed"
   654→   654→        return "<html><body><h1>Login Failed</h1><p>Token exchange failed</p></body></html>", 500
   655→   655→    print(f"[callback] Token exchange OK for state={state[:12]}... user_id={result.get('user_id')}")
   656→   656→
   657→   657→    # Extract email from JWT jti field
   658→   658→    import base64 as _b64
   659→   659→    email = result.get("user_name") or f"user_{result['user_id']}"
   660→   660→    try:
   661→   661→        token_part = result["access_token"].replace("Bearer ", "").split(".")[1]
   662→   662→        token_part += "=" * (4 - len(token_part) % 4)
   663→   663→        payload = json.loads(_b64.urlsafe_b64decode(token_part))
   664→   664→        if payload.get("jti"):
   665→   665→            email = payload["jti"]
   666→   666→    except:
   667→   667→        pass
   668→   668→
   669→   669→    add_token(
   670→   670→        email=email,
   671→   671→        access_token=result["access_token"],
   672→   672→        refresh_token=result["refresh_token"],
   673→   673→        user_id=result["user_id"],
   674→   674→        device_id=device_id,
   675→   675→    )
   676→   676→    pending["result"] = {
   677→   677→        "email": email,
   678→   678→        "user_id": result.get("user_id"),
   679→   679→        "first_login": result.get("first_login"),
   680→   680→    }
   681→   681→    return f"""<html><body>
   682→   682→<h1>Login Success!</h1>
   683→   683→<p>Email: {email}</p>
