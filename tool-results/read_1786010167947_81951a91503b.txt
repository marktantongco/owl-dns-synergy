   245→   245→# 2. DNS Flood Protector (from v2)
   246→   246→# ═══════════════════════════════════════════════════════════════
   247→   247→
   248→   248→class DNSFloodProtector:
   249→   249→    """Token-bucket + per-client rate limiting for DNS queries."""
   250→   250→
   251→   251→    def __init__(self, max_qps: int = 50, burst: int = 100):
   252→   252→        self.max_qps = max_qps
   253→   253→        self.burst = burst
   254→   254→        self._tokens: float = float(burst)
   255→   255→        self._last_refill: float = time.time()
   256→   256→        self._lock = asyncio.Lock()
   257→   257→        self._blocked_count = 0
   258→   258→        self._client_queries: Dict[str, deque] = {}
   259→   259→
   260→   260→    async def allow(self, client_ip: str = "default") -> bool:
   261→   261→        async with self._lock:
   262→   262→            now = time.time()
   263→   263→            elapsed = now - self._last_refill
   264→   264→            self._tokens = min(self.burst, self._tokens + elapsed * self.max_qps)
   265→   265→            self._last_refill = now
   266→   266→            if self._tokens >= 1.0:
   267→   267→                self._tokens -= 1.0
   268→   268→            else:
   269→   269→                self._blocked_count += 1
   270→   270→                DNS_FLOOD_BLOCKED.inc()
   271→   271→                return False
   272→   272→            if client_ip not in self._client_queries:
   273→   273→                self._client_queries[client_ip] = deque(maxlen=100)
   274→   274→            self._client_queries[client_ip].append(now)
   275→   275→            recent = sum(1 for t in self._client_queries[client_ip] if now - t < 1.0)
   276→   276→            if recent > 10:
   277→   277→                self._blocked_count += 1
   278→   278→                DNS_FLOOD_BLOCKED.inc()
   279→   279→                return False
   280→   280→            return True

   284→   284→# 3. Proxy Pool Adapter — prox5-compatible SOCKS/HTTP pool
   285→   285→# ═══════════════════════════════════════════════════════════════
   286→   286→
   287→   287→@dataclass
   288→   288→class ProxyEntry:
   289→   289→    """Represents a single proxy endpoint (prox5-compatible)."""
   290→   290→    endpoint: str           # host:port
   291→   291→    protocol: str = "socks5"  # socks5, socks4, http
   292→   292→    username: str = ""
   293→   293→    password: str = ""
   294→   294→    proxied_ip: str = ""
   295→   295→    last_validated: float = 0.0
   296→   296→    successes: int = 0
   297→   297→    failures: int = 0
   298→   298→    score: float = 1.0      # EMA quality score
   299→   299→    region: str = ""
   300→   300→
   301→   301→    @property
   302→   302→    def url(self) -> str:
   303→   303→        if self.username and self.password:
   304→   304→            return f"{self.protocol}://{self.username}:{self.password}@{self.endpoint}"
   305→   305→        return f"{self.protocol}://{self.endpoint}"
   306→   306→
   307→   307→    @property
   308→   308→    def is_stale(self) -> bool:
   309→   309→        return time.time() - self.last_validated > 1800  # 30 min
   310→   310→
   311→   311→
   312→   312→class ProxyPoolAdapter:
   313→   313→    """
   314→   314→    Python adapter for prox5-style proxy pool management.
   315→   315→    Loads proxies from file (prox5/proxies.txt format), validates,
   316→   316→    and rotates using Mystery Dialer pattern (retry on failure).
   317→   317→    """
   318→   318→
   319→   319→    def __init__(self, proxy_file: str = None, max_workers: int = 10):
   320→   320→        self._pool: List[ProxyEntry] = []
   321→   321→        self._current_index = 0
   322→   322→        self._lock = threading.Lock()
   323→   323→        self._max_workers = max_workers
   324→   324→        self._proxy_file = proxy_file
   325→   325→        if proxy_file and os.path.exists(proxy_file):
   326→   326→            self.load_from_file(proxy_file)
   327→   327→
   328→   328→    def load_from_file(self, filepath: str):
   329→   329→        """
   330→   330→        Load proxies from file. Supports formats:
   331→   331→          - host:port:user:pass  (autoclaw/prox5 format)
   332→   332→          - protocol://host:port  (URL format)
   333→   333→          - host:port  (simple format)
   334→   334→        """
   335→   335→        count = 0
   336→   336→        with open(filepath) as f:
   337→   337→            for line in f:
   338→   338→                line = line.strip()
   339→   339→                if not line or line.startswith('#'):
   340→   340→                    continue
   341→   341→                entry = self._parse_proxy_line(line)
   342→   342→                if entry:
   343→   343→                    self._pool.append(entry)
   344→   344→                    count += 1
   345→   345→        logger.info(f"Loaded {count} proxies from {filepath}")
   346→   346→        PROXY_POOL_SIZE.labels(protocol="all").set(len(self._pool))
   347→   347→
   348→   348→    def _parse_proxy_line(self, line: str) -> Optional[ProxyEntry]:
   349→   349→        """Parse a proxy line in any supported format."""
   350→   350→        if '://' in line:
   351→   351→            # URL format: socks5://user:pass@host:port
   352→   352→            parsed = urlparse(line)
   353→   353→            protocol = parsed.scheme or "socks5"
   354→   354→            host = parsed.hostname or ""
   355→   355→            port = parsed.port or 1080
   356→   356→            return ProxyEntry(
   357→   357→                endpoint=f"{host}:{port}",
   358→   358→                protocol=protocol,
   359→   359→                username=parsed.username or "",
   360→   360→                password=parsed.password or "",
   361→   361→            )
   362→   362→        parts = line.split(':')
   363→   363→        if len(parts) == 4:
   364→   364→            # host:port:user:pass format
   365→   365→            return ProxyEntry(
   366→   366→                endpoint=f"{parts[0]}:{parts[1]}",
   367→   367→                username=parts[2],
   368→   368→                password=parts[3],
   369→   369→            )
   370→   370→        if len(parts) == 2:
   371→   371→            try:
   372→   372→                int(parts[1])
   373→   373→                return ProxyEntry(endpoint=line)
   374→   374→            except ValueError:
   375→   375→                pass
   376→   376→        return None
   377→   377→
   378→   378→    def add_proxy(self, endpoint: str, protocol: str = "socks5",
   379→   379→                  username: str = "", password: str = ""):
   380→   380→        with self._lock:
   381→   381→            self._pool.append(ProxyEntry(
   382→   382→                endpoint=endpoint, protocol=protocol,
   383→   383→                username=username, password=password
   384→   384→            ))
   385→   385→
   386→   386→    def get_next(self) -> Optional[ProxyEntry]:
   387→   387→        """Get next proxy using round-robin with Mystery Dialer retry."""
   388→   388→        with self._lock:
   389→   389→            if not self._pool:
   390→   390→                return None
   391→   391→            # Try current proxy first
   392→   392→            for offset in range(len(self._pool)):
   393→   393→                idx = (self._current_index + offset) % len(self._pool)
   394→   394→                proxy = self._pool[idx]
   395→   395→                if proxy.score > 0.2 and not proxy.is_stale:
   396→   396→                    self._current_index = idx
   397→   397→                    return proxy
   398→   398→            # All stale/low-score, return best available
   399→   399→            best = max(self._pool, key=lambda p: p.score)
   400→   400→            return best
   401→   401→
   402→   402→    def report_success(self, endpoint: str, latency_ms: float = 0):
   403→   403→        with self._lock:
   404→   404→            for p in self._pool:
   405→   405→                if p.endpoint == endpoint:
   406→   406→                    p.successes += 1
   407→   407→                    p.last_validated = time.time()
   408→   408→                    # EMA update: alpha=0.3
   409→   409→                    p.score = 0.3 * 1.0 + 0.7 * p.score
   410→   410→                    break
   411→   411→
   412→   412→    def report_failure(self, endpoint: str):
   413→   413→        with self._lock:
   414→   414→            for p in self._pool:
   415→   415→                if p.endpoint == endpoint:
   416→   416→                    p.failures += 1
   417→   417→                    p.score = 0.3 * 0.0 + 0.7 * p.score
   418→   418→                    break
   419→   419→
   420→   420→    @property
   421→   421→    def size(self) -> int:
   422→   422→        return len(self._pool)
   423→   423→
   424→   424→    @property
   425→   425→    def valid_count(self) -> int:
   426→   426→        return sum(1 for p in self._pool if p.score > 0.2 and not p.is_stale)
   427→   427→
   428→   428→    def get_status(self) -> Dict[str, Any]:
   429→   429→        return {
   430→   430→            "total": self.size,
   431→   431→            "valid": self.valid_count,
   432→   432→            "current_index": self._current_index,
   433→   433→            "protocols": {
   434→   434→                proto: sum(1 for p in self._pool if p.protocol == proto)
   435→   435→                for proto in set(p.protocol for p in self._pool)
   436→   436→            },
   437→   437→            "top_5": [
   438→   438→                {"endpoint": p.endpoint, "protocol": p.protocol,
   439→   439→                 "score": round(p.score, 3), "successes": p.successes,
   440→   440→                 "failures": p.failures}
   441→   441→                for p in sorted(self._pool, key=lambda x: x.score, reverse=True)[:5]
   442→   442→            ]
   443→   443→        }

  1045→  1045→    async def fetch(self, url: str, force_channel: str = None, **kwargs) -> ChannelResult:
  1046→  1046→        """
  1047→  1047→        Fetch a URL through the optimal channel with automatic failover.
  1048→  1048→
  1049→  1049→        Channel selection priority:
  1050→  1050→          cached → http_proxy → socks_pool → dns_tunnel → mitm_stealth → connect_chain → http_direct
  1051→  1051→        """
  1052→  1052→        domain = self._extract_domain(url)
  1053→  1053→
  1054→  1054→        # Check cache
  1055→  1055→        cached = self._cache_get(url)
  1056→  1056→        if cached and not force_channel:
  1057→  1057→            return ChannelResult("cached", True, data=cached)
  1058→  1058→
  1059→  1059→        # Select channel
  1060→  1060→        channel = self._select_channel(domain, force_channel)
  1061→  1061→
  1062→  1062→        # Channel implementations
  1063→  1063→        channel_map = {
  1064→  1064→            Channel.HTTP_PROXY: self._try_http_proxy,
  1065→  1065→            Channel.SOCKS_POOL: self._try_socks_pool,
  1066→  1066→            Channel.DNS_TUNNEL: self._try_dns_tunnel,
  1067→  1067→            Channel.MITM_STEALTH: self._try_mitm_stealth,
  1068→  1068→            Channel.CONNECT_CHAIN: self._try_connect_chain,
  1069→  1069→            Channel.HTTP_DIRECT: self._try_http_direct,
  1070→  1070→        }
  1071→  1071→
  1072→  1072→        # Try preferred channel
  1073→  1073→        if channel in channel_map:
  1074→  1074→            result = await channel_map[channel](url, domain, **kwargs)
  1075→  1075→            if result.success:
  1076→  1076→                REQUESTS_TOTAL.labels(channel=result.channel, domain=domain, status="success").inc()
  1077→  1077→                REQUESTS_DURATION.labels(channel=result.channel, domain=domain).observe(result.latency_ms / 1000)
  1078→  1078→                return result
  1079→  1079→
  1080→  1080→        # Fallback: try all channels in priority order
  1081→  1081→        fallback_order = [
  1082→  1082→            Channel.HTTP_PROXY, Channel.SOCKS_POOL, Channel.DNS_TUNNEL,
  1083→  1083→            Channel.MITM_STEALTH, Channel.HTTP_DIRECT,
  1084→  1084→        ]
  1085→  1085→        for ch in fallback_order:
  1086→  1086→            if ch == channel:
  1087→  1087→                continue  # Already tried
  1088→  1088→            if ch in channel_map:
  1089→  1089→                result = await channel_map[ch](url, domain, **kwargs)
  1090→  1090→                if result.success:
  1091→  1091→                    CHANNEL_SWITCHES.labels(
  1092→  1092→                        from_channel=channel.value, to_channel=ch.value
  1093→  1093→                    ).inc()
  1094→  1094→                    REQUESTS_TOTAL.labels(channel=result.channel, domain=domain, status="success").inc()
  1095→  1095→                    return result
  1096→  1096→
  1097→  1097→        REQUESTS_TOTAL.labels(channel="none", domain=domain, status="all_failed").inc()
  1098→  1098→        return ChannelResult("none", False, error=f"All channels exhausted for {domain}")
