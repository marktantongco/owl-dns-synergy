   129→class CircuitState(Enum):
   130→    """3-state circuit breaker: CLOSED → OPEN → HALF_OPEN → CLOSED"""
   131→    CLOSED = 0      # Normal operation — requests flow through
   132→    OPEN = 1        # Circuit tripped — all requests rejected
   133→    HALF_OPEN = 2  # Probe state — one request allowed to test recovery
   134→
   135→
   136→class ChannelCircuitBreaker:
   137→    """Per-channel 3-state circuit breaker with Prometheus tracking.
   138→
   139→    States:
   140→      CLOSED: Normal. All requests pass. Failure count tracked.
   141→      OPEN: Circuit tripped. All requests fail-fast. Timer starts.
   142→      HALF_OPEN: One probe request allowed. If success → CLOSED, if fail → OPEN.
   143→
   144→    Args:
   145→        channel_name: Name for logging/metrics
   146→        failure_threshold: Consecutive failures before opening (default 5)
   147→        recovery_timeout: Seconds before half-open probe (default 30)
   148→        success_threshold: Consecutive successes in HALF_OPEN to close (default 2)
   149→    """
   150→    def __init__(self, channel_name: str, failure_threshold: int = 5,
   151→                 recovery_timeout: float = 30.0, success_threshold: int = 2):
   152→        self.channel_name = channel_name
   153→        self.failure_threshold = failure_threshold
   154→        self.recovery_timeout = recovery_timeout
   155→        self.success_threshold = success_threshold
   156→        self._state = CircuitState.CLOSED
   157→        self._failure_count = 0
   158→        self._success_count = 0
   159→        self._opened_at: float = 0.0
   160→        self._lock = asyncio.Lock()
   161→
   162→    @property
   163→    def state(self) -> CircuitState:
   164→        """Current state, with automatic HALF_OPEN transition on timeout."""
   165→        if self._state == CircuitState.OPEN:
   166→            if time.time() - self._opened_at >= self.recovery_timeout:
   167→                self._state = CircuitState.HALF_OPEN
   168→        return self._state
   169→    @property
   170→    def is_open(self) -> bool:
   171→        """True if circuit is OPEN (requests should be rejected)."""
   172→        return self.state == CircuitState.OPEN
   173→    async def allow_request(self) -> bool:
   174→        """Check if a request should be allowed through."""
   175→        async with self._lock:
   176→            state = self.state
   177→            if state == CircuitState.CLOSED:
   178→                return True
   179→            elif state == CircuitState.HALF_OPEN:
   180→                # Allow ONE probe request through
   181→                return True
   182→            else:  # OPEN
   183→                return False
   184→    async def record_success(self):
   185→        """Record a successful request."""
   186→        async with self._lock:
   187→            if self._state == CircuitState.HALF_OPEN:
   188→                self._success_count += 1
   189→                if self._success_count >= self.success_threshold:
   190→                    self._state = CircuitState.CLOSED
   191→                    self._failure_count = 0
   192→                    self._success_count = 0
   193→                    logger.info(f"Circuit CLOSED for {self.channel_name} — recovered")
   194→            elif self._state == CircuitState.CLOSED:
   195→                self._failure_count = 0  # Reset on success
   196→    async def record_failure(self):
   197→        """Record a failed request."""
   198→        async with self._lock:
   199→            if self._state == CircuitState.HALF_OPEN:
   200→                # Probe failed → back to OPEN
   201→                self._state = CircuitState.OPEN
   202→                self._opened_at = time.time()
   203→                self._success_count = 0
   204→                logger.warning(f"Circuit back to OPEN for {self.channel_name} — probe failed")
   205→            elif self._state == CircuitState.CLOSED:
   206→                self._failure_count += 1
   207→                if self._failure_count >= self.failure_threshold:
   208→                    self._state = CircuitState.OPEN
   209→                    self._opened_at = time.time()
   210→                    logger.warning(
   211→                        f"Circuit OPEN for {self.channel_name} — "
   212→                        f"{self._failure_count} failures >= threshold {self.failure_threshold}"
   213→                    )

   228→@dataclass(slots=True)
   229→class DomainPreference:
   230→    """Per-domain channel preference with EMA-based learning.
   231→
   232→    Tracks success/failure counts per channel and uses exponential moving average
   233→    to compute a quality score. The channel with the highest EMA score becomes
   234→    the preferred_channel for subsequent requests to this domain.
   235→    
   236→    Memory Fix: slots=True reduces per-instance memory by ~40% (no __dict__).
   237→    """
   238→    domain: str
   239→    successes: Dict[str, int] = field(default_factory=lambda: {})
   240→    failures: Dict[str, int] = field(default_factory=lambda: {})
   241→    preferred_channel: str = "http_proxy"
   242→    last_updated: float = field(default_factory=time.time)
   243→    # EMA scores per channel (0.0-1.0)
   244→    _ema_scores: Dict[str, float] = field(default_factory=lambda: {})
   245→    _ema_alpha: float = 0.3  # EMA smoothing factor (higher = more responsive)
   246→
   247→    def record_channel_result(self, channel: str, success: bool):
   248→        """Record a channel result and update EMA score."""
   249→        if success:
   250→            self.successes[channel] = self.successes.get(channel, 0) + 1
   251→        else:
   252→            self.failures[channel] = self.failures.get(channel, 0) + 1
   253→        # EMA update: new_score = alpha * sample + (1 - alpha) * old_score
   254→        old_score = self._ema_scores.get(channel, 0.5)
   255→        sample = 1.0 if success else 0.0
   256→        self._ema_scores[channel] = self._ema_alpha * sample + (1 - self._ema_alpha) * old_score
   257→        self.last_updated = time.time()
   258→        # Re-compute preferred channel from EMA scores
   259→        if self._ema_scores:
   260→            best_channel = max(self._ema_scores, key=self._ema_scores.get)
   261→            if self._ema_scores[best_channel] > 0.3:  # Minimum quality threshold
   262→                self.preferred_channel = best_channel
   263→
   264→    def get_channel_score(self, channel: str) -> float:
   265→        """Get EMA quality score for a channel (0.0-1.0)."""
   266→        return self._ema_scores.get(channel, 0.5)

   271→# ═══════════════════════════════════════════════════════════════
   272→
   273→class OpenRouterKeyRotator:
   274→    """Round-robin API key rotation with failover on 429/401/403."""
   275→
   276→    def __init__(self, keys: List[str] = None, base_url: str = "https://openrouter.ai/api/v1"):
   277→        self._keys = keys or []
   278→        self._current_index = 0
   279→        self._base_url = base_url
   280→        self._key_errors: Dict[int, int] = {}
   281→        self._key_cooldown: Dict[int, float] = {}
   282→        self._lock = threading.Lock()
   283→
   284→    @classmethod
   285→    def from_env(cls) -> "OpenRouterKeyRotator":
   286→        keys = []
   287→        primary = os.getenv("OPENAI_API_KEY")
   288→        if primary:
   289→            keys.append(primary)
   290→        for i in range(1, 10):
   291→            key = os.getenv(f"OPENROUTER_KEY_{i}")
   292→            if key:
   293→                keys.append(key)
   294→        base_url = os.getenv("OPENAI_BASE_URL", "https://openrouter.ai/api/v1")
   295→        return cls(keys=keys, base_url=base_url)
   296→
   297→    @property
   298→    def base_url(self) -> str:
   299→        return self._base_url
   300→
   301→    def get_active_key(self) -> Optional[str]:
   302→        with self._lock:
   303→            if not self._keys:
   304→                return None
   305→            now = time.time()
   306→            if now >= self._key_cooldown.get(self._current_index, 0):
   307→                return self._keys[self._current_index]
   308→            for offset in range(len(self._keys)):
   309→                idx = (self._current_index + offset) % len(self._keys)
   310→                if now >= self._key_cooldown.get(idx, 0):
   311→                    self._current_index = idx
   312→                    KEY_ROTATION_COUNT.inc()
   313→                    return self._keys[idx]
   314→            return None
   315→
   316→    def report_error(self, status_code: int, error_type: str = "unknown"):
   317→        with self._lock:
   318→            idx = self._current_index
   319→            self._key_errors[idx] = self._key_errors.get(idx, 0) + 1
   320→            if status_code == 429:
   321→                self._key_cooldown[idx] = time.time() + 60
   322→            elif status_code in (401, 403):
   323→                self._key_cooldown[idx] = time.time() + 300
   324→            else:
   325→                self._key_cooldown[idx] = time.time() + 10
   326→            self._rotate_to_next()
   327→
   328→    def _rotate_to_next(self):
   329→        if len(self._keys) <= 1:
   330→            return
   331→        now = time.time()
   332→        for offset in range(1, len(self._keys)):
   333→            idx = (self._current_index + offset) % len(self._keys)
   334→            if now >= self._key_cooldown.get(idx, 0):
   335→                self._current_index = idx
   336→                KEY_ROTATION_COUNT.inc()
   337→                return
   338→        self._current_index = (self._current_index + 1) % len(self._keys)
   339→        KEY_ROTATION_COUNT.inc()
   340→
   341→    @property
   342→    def total_keys(self) -> int:
   343→        return len(self._keys)
   344→
   345→    @property
   346→    def available_keys(self) -> int:
   347→        now = time.time()
   348→        return sum(1 for i in range(len(self._keys))
   349→                   if now >= self._key_cooldown.get(i, 0))
   350→
   351→    def get_status(self) -> Dict[str, Any]:
   352→        now = time.time()
   353→        return {
   354→            "total_keys": self.total_keys,
   355→            "available_keys": self.available_keys,
   356→            "current_index": self._current_index,
   357→            "key_errors": dict(self._key_errors),
   358→            "cooldowns": {
   359→                i: max(0, self._key_cooldown.get(i, 0) - now)
   360→                for i in range(len(self._keys))
   361→            }
   362→        }

   366→# 2. DNS Flood Protector (from v2)
   367→# ═══════════════════════════════════════════════════════════════
   368→
   369→class DNSFloodProtector:
   370→    """Token-bucket + per-client rate limiting for DNS queries."""
   371→
   372→    def __init__(self, max_qps: int = 50, burst: int = 100,
   373→                 max_clients: int = 50000, client_ttl: float = 300.0):
   374→        self.max_qps = max_qps
   375→        self.burst = burst
   376→        self._tokens: float = float(burst)
   377→        self._last_refill: float = time.time()
   378→        self._lock = asyncio.Lock()
   379→        self._blocked_count = 0
   380→        self._client_queries: Dict[str, deque] = {}
   381→        # Memory Fix M-R2: Client IP eviction + max client cap
   382→        self._client_last_seen: Dict[str, float] = {}
   383→        self._max_clients = max_clients
   384→        self._client_ttl = client_ttl
   385→
   386→    def _evict_stale_clients(self) -> None:
   387→        """Memory Fix M-R2: Evict client IPs idle > client_ttl to prevent unbounded growth."""
   388→        now = time.time()
   389→        stale = [ip for ip, t in self._client_last_seen.items()
   390→                 if now - t > self._client_ttl]
   391→        for ip in stale:
   392→            self._client_queries.pop(ip, None)
   393→            self._client_last_seen.pop(ip, None)
   394→
   395→    async def allow(self, client_ip: str = "default") -> bool:
   396→        async with self._lock:
   397→            now = time.time()
   398→            # Memory Fix M-R2: Evict idle clients periodically
   399→            self._evict_stale_clients()
   400→            # Per-client rate check FIRST (before consuming global token)
   401→            if client_ip not in self._client_queries:
   402→                # Memory Fix M-R2: Cap total tracked clients
   403→                if len(self._client_queries) >= self._max_clients:
   404→                    # Evict oldest client to make room
   405→                    oldest_ip = min(self._client_last_seen, key=self._client_last_seen.get)
   406→                    self._client_queries.pop(oldest_ip, None)
   407→                    self._client_last_seen.pop(oldest_ip, None)
   408→                self._client_queries[client_ip] = deque(maxlen=100)
   409→            self._client_queries[client_ip].append(now)
   410→            self._client_last_seen[client_ip] = now  # Track last activity
   411→            recent = sum(1 for t in self._client_queries[client_ip] if now - t < 1.0)
   412→            if recent > 10:
   413→                self._blocked_count += 1
   414→                DNS_FLOOD_BLOCKED.inc()
   415→                return False
   416→            # Global token bucket check (after per-client passes)
   417→            elapsed = now - self._last_refill
   418→            self._tokens = min(self.burst, self._tokens + elapsed * self.max_qps)
   419→            self._last_refill = now
   420→            if self._tokens >= 1.0:
   421→                self._tokens -= 1.0
   422→                return True
   423→            self._blocked_count += 1
   424→            DNS_FLOOD_BLOCKED.inc()
   425→            return False

   432→@dataclass
   433→class ProxyEntry:
   434→    """Represents a single proxy endpoint (prox5-compatible)."""
   435→    endpoint: str           # host:port
   436→    protocol: str = "socks5"  # socks5, socks4, http
   437→    username: str = ""
   438→    password: str = ""
   439→    proxied_ip: str = ""
   440→    last_validated: float = 0.0
   441→    successes: int = 0
   442→    failures: int = 0
   443→    score: float = 1.0      # EMA quality score
   444→    region: str = ""
   445→
   446→    @property
   447→    def url(self) -> str:
   448→        if self.username and self.password:
   449→            return f"{self.protocol}://{self.username}:{self.password}@{self.endpoint}"
   450→        return f"{self.protocol}://{self.endpoint}"
   451→
   452→    @property
   453→    def is_stale(self) -> bool:
   454→        return time.time() - self.last_validated > 1800  # 30 min
   455→
   456→
   457→class ProxyPoolAdapter:
   458→    """
   459→    Python adapter for prox5-style proxy pool management.
   460→    Loads proxies from file (prox5/proxies.txt format), validates,
   461→    and rotates using Mystery Dialer pattern (retry on failure).
   462→    """
   463→
   464→    def __init__(self, proxy_file: str = None, max_workers: int = 10):
   465→        self._pool: List[ProxyEntry] = []
   466→        self._current_index = 0
   467→        self._lock = threading.Lock()
   468→        self._max_workers = max_workers
   469→        self._proxy_file = proxy_file
   470→        if proxy_file and os.path.exists(proxy_file):
   471→            self.load_from_file(proxy_file)
   472→
   473→    def load_from_file(self, filepath: str):
   474→        """
   475→        Load proxies from file. Supports formats:
   476→          - host:port:user:pass  (autoclaw/prox5 format)
   477→          - protocol://host:port  (URL format)
   478→          - host:port  (simple format)
   479→        """
   480→        count = 0
   481→        with open(filepath) as f:
   482→            for line in f:
   483→                line = line.strip()
   484→                if not line or line.startswith('#'):
   485→                    continue
   486→                entry = self._parse_proxy_line(line)
   487→                if entry:
   488→                    self._pool.append(entry)
   489→                    count += 1
   490→        logger.info(f"Loaded {count} proxies from {filepath}")
   491→        PROXY_POOL_SIZE.labels(protocol="all").set(len(self._pool))
   492→
   493→    def _parse_proxy_line(self, line: str) -> Optional[ProxyEntry]:
   494→        """Parse a proxy line in any supported format."""
   495→        if '://' in line:
   496→            # URL format: socks5://user:pass@host:port
   497→            parsed = urlparse(line)
   498→            protocol = parsed.scheme or "socks5"
   499→            host = parsed.hostname or ""
   500→            port = parsed.port or 1080
   501→            return ProxyEntry(
   502→                endpoint=f"{host}:{port}",
   503→                protocol=protocol,
   504→                username=parsed.username or "",
   505→                password=parsed.password or "",
   506→            )
   507→        parts = line.split(':')
   508→        if len(parts) == 4:
   509→            # host:port:user:pass format
   510→            return ProxyEntry(
   511→                endpoint=f"{parts[0]}:{parts[1]}",
   512→                username=parts[2],
   513→                password=parts[3],
   514→            )
   515→        if len(parts) == 2:
   516→            try:
   517→                int(parts[1])
   518→                return ProxyEntry(endpoint=line)
   519→            except ValueError:
   520→                pass
   521→        return None
   522→
   523→    def add_proxy(self, endpoint: str, protocol: str = "socks5",
   524→                  username: str = "", password: str = ""):
   525→        with self._lock:
   526→            self._pool.append(ProxyEntry(
   527→                endpoint=endpoint, protocol=protocol,
   528→                username=username, password=password
   529→            ))
   530→
   531→    def get_next(self) -> Optional[ProxyEntry]:
   532→        """Get next proxy using round-robin with Mystery Dialer retry."""
   533→        with self._lock:
   534→            if not self._pool:
   535→                return None
   536→            # Try current proxy first
   537→            for offset in range(len(self._pool)):
   538→                idx = (self._current_index + offset) % len(self._pool)
   539→                proxy = self._pool[idx]
   540→                if proxy.score > 0.2 and not proxy.is_stale:
   541→                    self._current_index = idx
   542→                    return proxy
   543→            # All stale/low-score, return best available
   544→            best = max(self._pool, key=lambda p: p.score)
   545→            return best
   546→
   547→    def report_success(self, endpoint: str, latency_ms: float = 0):
   548→        with self._lock:

  1112→    def _evict_stale_preferences(self) -> None:
  1113→        """Memory Fix M-R1: Evict DomainPreference entries idle > pref_ttl; hard cap at max_prefs."""
  1114→        now = time.time()
  1115→        stale = [d for d, p in self._prefs.items()
  1116→                 if now - p.last_updated > self._pref_ttl]
  1117→        for d in stale:
  1118→            del self._prefs[d]
  1119→        # Hard cap: evict oldest if still over limit
  1120→        if len(self._prefs) > self._max_prefs:
  1121→            sorted_d = sorted(self._prefs.items(), key=lambda x: x[1].last_updated)
  1122→            for d, _ in sorted_d[:len(self._prefs) - self._max_prefs]:
  1123→                del self._prefs[d]

  1125→    def _cache_get(self, key: str) -> Optional[Any]:
  1126→        if key in self._cache:
  1127→            data, ts = self._cache[key]
  1128→            if time.time() - ts < self._cache_ttl:
  1129→                return data
  1130→            del self._cache[key]
  1131→        return None

  1133→    def _cache_set(self, key: str, data: Any):
  1134→        # Evict oldest entries when cache exceeds max size
  1135→        if len(self._cache) >= self._cache_max:
  1136→            oldest_key = min(self._cache, key=lambda k: self._cache[k][1])
  1137→            del self._cache[oldest_key]
  1138→        self._cache[key] = (data, time.time())

  1175→    async def _try_http_proxy(self, url: str, domain: str, **kwargs) -> ChannelResult:
  1176→        """Route through proxy pool (prox5/https_proxy)."""
  1177→        start = time.time()
  1178→        proxy = self.proxy_pool.get_next()
  1179→
  1180→        if not proxy:
  1181→            return ChannelResult("http_proxy", False, error="No proxies available")
  1182→
  1183→        try:
  1184→            api_key = self.key_rotator.get_active_key()
  1185→            headers = {}
  1186→            if api_key:
  1187→                headers["Authorization"] = f"Bearer {api_key}"
  1188→
  1189→            # Memory Fix M-R3: Use shared httpx client instead of per-request creation
  1190→            client = await self._get_http_client(proxy=proxy.url)
  1191→            resp = await client.get(url, headers=headers)
  1192→            latency = (time.time() - start) * 1000
  1193→
  1194→            if resp.status_code in (401, 403, 429):
  1195→                self.key_rotator.report_error(
  1196→                    resp.status_code,
  1197→                    "rate_limit" if resp.status_code == 429 else "auth_error"
  1198→                )
  1199→
  1200→            self.proxy_pool.report_success(proxy.endpoint, latency)
  1201→
  1202→            # Only treat 2xx as success — non-2xx should trigger failover
  1203→            is_success = 200 <= resp.status_code < 300
  1204→            if is_success:
  1205→                self._cache_set(url, resp.content)
  1206→
  1207→            return ChannelResult(
  1208→                "http_proxy", is_success,
  1209→                data=resp.content, latency_ms=latency,
  1210→                status_code=resp.status_code, proxy_used=proxy.endpoint
  1211→            )
  1212→        except Exception as e:
  1213→            latency = (time.time() - start) * 1000
  1214→            self.proxy_pool.report_failure(proxy.endpoint)
  1215→            return ChannelResult("http_proxy", False, error=str(e), latency_ms=latency)

  1241→    async def _try_dns_tunnel(self, url: str, domain: str, **kwargs) -> ChannelResult:
  1242→        """Fall back to DNS tunneling (llm-dns-proxy).
  1243→        Verifies DNS server is actually reachable by sending a health-check query."""
  1244→        start = time.time()
  1245→        try:
  1246→            import socket
  1247→            dns_host = os.getenv("DNS_SERVER_HOST", "127.0.0.1")
  1248→            dns_port = int(os.getenv("DNS_SERVER_PORT", "5353"))
  1249→            sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
  1250→            sock.settimeout(2)
  1251→            # Send a minimal DNS query to verify the server is actually running
  1252→            # DNS header: ID=0x1234, RD=1, QDCOUNT=1, QNAME=., QTYPE=TXT, QCLASS=IN
  1253→            query = b'\x12\x34\x01\x00\x00\x01\x00\x00\x00\x00\x00\x00\x00\x00\x10\x00\x01'
  1254→            sock.sendto(query, (dns_host, dns_port))
  1255→            # Wait for any response (even SERVFAIL proves the server is alive)
  1256→            data, _ = sock.recvfrom(512)
  1257→            sock.close()
  1258→            if not data:
  1259→                return ChannelResult("dns_tunnel", False, error="DNS server returned empty response")
  1260→            return ChannelResult(
  1261→                "dns_tunnel", True,
  1262→                data=f"[DNS tunnel verified for {domain}]",
  1263→                latency_ms=(time.time() - start) * 1000
  1264→            )
  1265→        except socket.timeout:
  1266→            return ChannelResult("dns_tunnel", False, error="DNS server not reachable (timeout)")
  1267→        except ConnectionRefusedError:
  1268→            return ChannelResult("dns_tunnel", False, error="DNS server not running (connection refused)")
  1269→        except Exception as e:
  1270→            return ChannelResult("dns_tunnel", False, error=str(e))

  1304→    async def fetch(self, url: str, force_channel: str = None, **kwargs) -> ChannelResult:
  1305→        """
  1306→        Fetch a URL through the optimal channel with automatic failover.
  1307→        Now includes per-channel circuit breaker and EMA domain preference learning.
  1308→
  1309→        Channel selection priority:
  1310→          cached → http_proxy → socks_pool → dns_tunnel → mitm_stealth → connect_chain → http_direct
  1311→        """
  1312→        domain = self._extract_domain(url)
  1313→
  1314→        # Memory Fix M-R1: Evict stale domain preferences on each request
  1315→        self._evict_stale_preferences()
  1316→
  1317→        # Check cache
  1318→        cached = self._cache_get(url)
  1319→        if cached and not force_channel:
  1320→            return ChannelResult("cached", True, data=cached)
  1321→
  1322→        # Select channel (uses DomainPreference EMA if available)
  1323→        channel = self._select_channel(domain, force_channel)
  1324→
  1325→        # Channel implementations
  1326→        channel_map = {
  1327→            Channel.HTTP_PROXY: self._try_http_proxy,
  1328→            Channel.SOCKS_POOL: self._try_socks_pool,
  1329→            Channel.DNS_TUNNEL: self._try_dns_tunnel,
  1330→            Channel.MITM_STEALTH: self._try_mitm_stealth,
  1331→            Channel.CONNECT_CHAIN: self._try_connect_chain,
  1332→            Channel.HTTP_DIRECT: self._try_http_direct,
  1333→        }
  1334→
  1335→        # Try preferred channel (with circuit breaker check)
  1336→        if channel in channel_map:
  1337→            cb = self._circuit_breakers.get(channel.value)
  1338→            if cb and not await cb.allow_request():
  1339→                logger.debug(f"Circuit OPEN for {channel.value} — skipping")
  1340→            else:
  1341→                result = await channel_map[channel](url, domain, **kwargs)
  1342→                # Record result in circuit breaker and EMA
  1343→                if cb:
  1344→                    if result.success:
  1345→                        await cb.record_success()
  1346→                    else:
  1347→                        await cb.record_failure()
  1348→                # Update domain preference EMA
  1349→                pref = self._prefs.get(domain)
  1350→                if not pref:
  1351→                    pref = DomainPreference(domain=domain)
  1352→                    self._prefs[domain] = pref
  1353→                pref.record_channel_result(result.channel, result.success)
  1354→
  1355→                if result.success:
  1356→                    REQUESTS_TOTAL.labels(channel=result.channel, domain=domain, status="success").inc()
  1357→                    REQUESTS_DURATION.labels(channel=result.channel, domain=domain).observe(result.latency_ms / 1000)
  1358→                    return result
  1359→                # Record failure in Prometheus
  1360→                REQUESTS_TOTAL.labels(channel=result.channel, domain=domain, status="error").inc()
  1361→
  1362→        # Fallback: try all channels in priority order
  1363→        # NOTE: CONNECT_CHAIN was missing — added between MITM and HTTP_DIRECT
  1364→        fallback_order = [
  1365→            Channel.HTTP_PROXY, Channel.SOCKS_POOL, Channel.DNS_TUNNEL,
  1366→            Channel.MITM_STEALTH, Channel.CONNECT_CHAIN, Channel.HTTP_DIRECT,
  1367→        ]
  1368→        for ch in fallback_order:
  1369→            if ch == channel:
  1370→                continue  # Already tried
  1371→            if ch not in channel_map:
  1372→                continue
  1373→
  1374→            # Circuit breaker check
  1375→            cb = self._circuit_breakers.get(ch.value)
  1376→            if cb and not await cb.allow_request():
  1377→                logger.debug(f"Circuit OPEN for {ch.value} — skipping fallback")
  1378→                continue
  1379→
  1380→            result = await channel_map[ch](url, domain, **kwargs)
  1381→
  1382→            # Record in circuit breaker and EMA
  1383→            if cb:
  1384→                if result.success:
  1385→                    await cb.record_success()
  1386→                else:
  1387→                    await cb.record_failure()
  1388→            pref = self._prefs.get(domain)
  1389→            if pref:
  1390→                pref.record_channel_result(result.channel, result.success)
  1391→
  1392→            if result.success:
  1393→                CHANNEL_SWITCHES.labels(
  1394→                    from_channel=channel.value, to_channel=ch.value
  1395→                ).inc()
  1396→                REQUESTS_TOTAL.labels(channel=result.channel, domain=domain, status="success").inc()
  1397→                REQUESTS_DURATION.labels(channel=result.channel, domain=domain).observe(result.latency_ms / 1000)
  1398→                return result
  1399→            # Record failure
  1400→            REQUESTS_TOTAL.labels(channel=result.channel, domain=domain, status="error").inc()
  1401→
  1402→        REQUESTS_TOTAL.labels(channel="none", domain=domain, status="all_failed").inc()
  1403→        return ChannelResult("none", False, error=f"All channels exhausted for {domain}")
