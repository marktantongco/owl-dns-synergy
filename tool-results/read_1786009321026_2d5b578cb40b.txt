    45→_request_counts = {}  # email → int

    48→def get_next_token():
    49→    """Round-robin token selection across all accounts.
    50→    Auto-refreshes expired tokens. Skips known-exhausted accounts (from cache only)."""
    51→    global _token_idx
    52→    data = load_tokens()
    53→    if not data["accounts"]:
    54→        return None, None
    55→
    56→    n = len(data["accounts"])
    57→    with _token_lock:
    58→        idx = _token_idx % n
    59→        _token_idx += 1
    60→
    61→    # Try each account starting from idx
    62→    for i in range(n):
    63→        acc = data["accounts"][(idx + i) % n]
    64→        # Check token validity
    65→        from config import ACCESS_TOKEN_TTL, REFRESH_MARGIN
    66→        age = time.time() - acc.get("last_refreshed", 0)
    67→        if age < ACCESS_TOKEN_TTL - REFRESH_MARGIN:
    68→            # Token still valid — check exhausted cache (non-blocking, cached only)
    69→            if not _is_cached_exhausted(acc):
    70→                return acc["access_token"], acc
    71→            else:
    72→                continue
    73→        # Try refresh
    74→        from auth import refresh_token
    75→        new_token = refresh_token(acc)
    76→        if new_token:
    77→            if not _is_cached_exhausted(acc):
    78→                return new_token, acc
    79→            else:
    80→                continue
    81→
    82→    # Fallback: return first valid token even if exhausted
    83→    for acc in data["accounts"]:
    84→        return acc["access_token"], acc
    85→    return None, None
    86→
    87→
    88→# ── Exhausted accounts cache (wallet balance=0) ──
    89→# Only uses cache — does NOT make API calls during chat requests
    90→_exhausted_cache = {}  # email → {"balance": int, "checked": timestamp}
    91→_EXHAUSTED_CACHE_TTL = 300  # cache valid for 5 min
    92→

    94→def _is_cached_exhausted(acc):
    95→    """Check if account is known to be exhausted (cache only, no API call)."""
    96→    email = acc.get("email", "")
    97→    now = time.time()
    98→    cached = _exhausted_cache.get(email)
    99→    if cached and (now - cached["checked"]) < _EXHAUSTED_CACHE_TTL:
   100→        return cached["balance"] <= 0
   101→    # No cache or expired — assume NOT exhausted (don't block chat)
   102→    return False

   105→def _refresh_exhausted_cache():
   106→    """Background: check all accounts' wallet balances and update cache.
   107→    Called periodically, not during chat requests."""
   108→    data = load_tokens()
   109→    for acc in data["accounts"]:
   110→        email = acc.get("email", "")
   111→        try:
   112→            result = check_wallet(acc["access_token"])
   113→            if result.get("code") == 0:
   114→                balance = result["data"]["total_balance"]
   115→                _exhausted_cache[email] = {"balance": balance, "checked": time.time()}
   116→                if balance <= 0:
   117→                    print(f"[proxy] Exhausted: {email} (0 pts)")
   118→        except:
   119→            pass

   127→        except Exception as e:
   128→            print(f"[proxy] Wallet checker error: {e}")

   149→@app.route("/v1/chat/completions", methods=["POST"])
   150→def chat_completions():
   151→    """OpenAI-compatible chat completions proxy."""
   152→    body = request.get_json(force=True)
   153→    if not body:
   154→        return jsonify({"error": {"message": "Invalid JSON body"}}), 400
   155→
   156→    # Model mapping
   157→    client_model = body.get("model", DEFAULT_MODEL)
   158→    upstream_model = MODEL_MAP.get(client_model, DEFAULT_MODEL)
   159→
   160→    # Force stream=True for upstream (DeepSeek models 500 on non-stream)
   161→    upstream_body = dict(body)
   162→    upstream_body["stream"] = True
   163→    upstream_body["model"] = "x"  # ignored by upstream, but fill it
   164→
   165→    # Get token (round-robin)
   166→    access_token, acc = get_next_token()
   167→    if not access_token:
   168→        return jsonify({
   169→            "error": {
   170→                "message": "No valid tokens. Add accounts first via /login or manually edit tokens.json",
   171→                "type": "auth_error",
   172→            }
   173→        }), 401
   174→
   175→    # Increment request counter for this account
   176→    used_email = acc.get("email", "unknown")
   177→    _request_counts[used_email] = _request_counts.get(used_email, 0) + 1
   178→
   179→    headers = _sign_headers()
   180→    # Token already has "Bearer " prefix — don't double it
   181→    raw_token = access_token.replace("Bearer ", "")
   182→    headers["X-Authorization"] = f"Bearer {raw_token}"
   183→    headers["X-Request-Id"] = str(uuid.uuid4())
   184→    headers["X-Request-Model"] = upstream_model
   185→
   186→    client_wants_stream = body.get("stream", False)
   187→
   188→    try:
   189→        # Always request stream from upstream
   190→        upstream_resp = req_lib.post(
   191→            CHAT_COMPLETIONS,
   192→            json=upstream_body,
   193→            headers=headers,
   194→            stream=True,
   195→            timeout=600,
   196→            verify=False,
   197→        )
   198→
   199→        if upstream_resp.status_code != 200:
   200→            error_text = upstream_resp.text[:1000]
   201→            return jsonify({
   202→                "error": {
   203→                    "message": f"Upstream error {upstream_resp.status_code}: {error_text}",
   204→                    "type": "upstream_error",
   205→                    "code": upstream_resp.status_code,
   206→                }
   207→            }), upstream_resp.status_code
   208→
   209→        if client_wants_stream:
   210→            # Pass through SSE stream — filter reasoning, only forward content chunks
   211→            def generate():
   212→                for line in upstream_resp.iter_lines():
   213→                    if line:
   214→                        if line.startswith(b"data:"):
   215→                            raw = line[5:].strip()
   216→                            if raw == b"[DONE]":
   217→                                yield line + b"\n\n"
   218→                                continue
   219→                            try:
   220→                                chunk = json.loads(raw)
   221→                                choices = chunk.get("choices", [])
   222→                                if choices:
   223→                                    delta = choices[0].get("delta", {})
   224→                                    # Strip reasoning fields — Cline/OpenAI clients don't understand them
   225→                                    if "reasoning" in delta:
   226→                                        del delta["reasoning"]
   227→                                    if "reasoning_details" in delta:
   228→                                        del delta["reasoning_details"]
   229→                                    if "reasoning_content" in delta:
   230→                                        del delta["reasoning_content"]
   231→                                    # Skip chunks with empty content and no tool_calls (pure thinking)
   232→                                    has_content = bool(delta.get("content"))
   233→                                    has_tool_calls = bool(delta.get("tool_calls"))
   234→                                    has_finish = bool(choices[0].get("finish_reason"))
   235→                                    # Only forward if there's actual content/tool_calls/finish
   236→                                    if not has_content and not has_tool_calls and not has_finish:
   237→                                        continue
   238→                                    # Clean native_finish_reason
   239→                                    if "native_finish_reason" in choices[0]:
   240→                                        del choices[0]["native_finish_reason"]
   241→                                yield b"data: " + json.dumps(chunk).encode() + b"\n\n"
   242→                            except (json.JSONDecodeError, KeyError):
   243→                                yield line + b"\n\n"
   244→                yield b"data: [DONE]\n\n"
   245→            return Response(
   246→                generate(),
   247→                mimetype="text/event-stream",
   248→                headers={
   249→                    "Cache-Control": "no-cache",
   250→                    "X-Accel-Buffering": "no",
   251→                },
   252→            )
   253→        else:
   254→            # Aggregate stream → single JSON response (OpenAI non-stream format)
   255→            full_content = ""
   256→            full_tool_calls = []
   257→            finish_reason = None
   258→            model_name = upstream_model
   259→
   260→            for line in upstream_resp.iter_lines():
   261→                if not line:
   262→                    continue
   263→                line_str = line.decode("utf-8", errors="replace")
   264→                if line_str.startswith("data: "):
   265→                    data_str = line_str[6:]
   266→                    if data_str.strip() == "[DONE]":
   267→                        break
   268→                    try:
   269→                        chunk = json.loads(data_str)
   270→                        choices = chunk.get("choices", [])
   271→                        if choices:
   272→                            delta = choices[0].get("delta", {})
   273→                            if "content" in delta and delta["content"]:
   274→                                full_content += delta["content"]
   275→                            if "tool_calls" in delta and delta["tool_calls"]:
   276→                                for tc in delta["tool_calls"]:
   277→                                    idx = tc.get("index", 0)
   278→                                    while len(full_tool_calls) <= idx:
   279→                                        full_tool_calls.append({"id": "", "type": "function", "function": {"name": "", "arguments": ""}})
   280→                                    if "id" in tc and tc["id"]:
   281→                                        full_tool_calls[idx]["id"] = tc["id"]
   282→                                    if "type" in tc and tc["type"]:
   283→                                        full_tool_calls[idx]["type"] = tc["type"]
   284→                                    if "function" in tc:
   285→                                        fn = tc["function"]
   286→                                        if "name" in fn and fn["name"]:
   287→                                            full_tool_calls[idx]["function"]["name"] += fn["name"]
   288→                                        if "arguments" in fn and fn["arguments"]:
   289→                                            full_tool_calls[idx]["function"]["arguments"] += fn["arguments"]
   290→                            if choices[0].get("finish_reason"):
   291→                                finish_reason = choices[0]["finish_reason"]
   292→                        if chunk.get("model"):
   293→                            model_name = chunk["model"]
   294→                        if chunk.get("usage"):
   295→                            usage = chunk["usage"]
   296→                    except json.JSONDecodeError:
   297→                        pass
   298→
   299→            message = {"role": "assistant", "content": full_content if full_content else None}
   300→            if full_tool_calls:
   301→                message["tool_calls"] = full_tool_calls
   302→                if not finish_reason:
   303→                    finish_reason = "tool_calls"
   304→
   305→            response = {
   306→                "id": f"chatcmpl-{uuid.uuid4().hex[:12]}",
   307→                "object": "chat.completion",
   308→                "created": int(time.time()),
   309→                "model": client_model,
   310→                "choices": [{
   311→                    "index": 0,
   312→                    "message": message,
   313→                    "finish_reason": finish_reason or "stop",
   314→                }],
   315→                "usage": usage if 'usage' in dir() else {"prompt_tokens": 0, "completion_tokens": 0, "total_tokens": 0},
   316→            }
   317→            return jsonify(response)
   318→
   319→    except req_lib.exceptions.Timeout:
   320→        return jsonify({"error": {"message": "Upstream timeout"}}), 504
   321→    except Exception as e:
   322→        return jsonify({"error": {"message": f"Proxy error: {str(e)}"}}), 500

   608→    error = request.args.get("error")
   609→
   610→    # Find pending login by state — MUST match, no fallback in concurrent mode
   611→    pending = _pending_logins.get(state)
   612→    if not pending:
   613→        # No state match — do NOT fallback to "first pending" (causes cross-account
   614→        # contamination in concurrent mode). Just fail this callback.
   615→        print(f"[callback] Unknown state={state[:20]}... (not in {len(_pending_logins)} pending)")
   616→        return "<html><body><h1>Login Failed</h1><p>Unknown state</p></body></html>", 400
   617→
   618→    if error:
   619→        pending["error"] = error
   620→        return f"<html><body><h1>Login Failed</h1><p>{error}</p></body></html>", 400
   621→
   622→    if not code:
   623→        pending["error"] = "No code in callback"
   624→        return "<html><body><h1>Login Failed</h1><p>No code</p></body></html>", 400
   625→
   626→    device_id = pending.get("device_id")
   627→    proxy_used = pending.get("proxy")  # Reuse same proxy from URL generation
   628→
   629→    # Serialize token exchanges — AutoClaw API rate-limits per IP (630014).
   630→    # Without this, 3 concurrent callbacks all hit the API simultaneously → all fail.
   631→    # Lock ensures 1 exchange at a time, with small gap between.
   632→    with _token_exchange_lock:
   633→        print(f"[callback] Exchanging token for state={state[:12]}... device_id={device_id[:8]}... proxy={'yes' if proxy_used else 'none'}")
   634→        result = google_oauth_login(code, state, device_id, proxy=proxy_used)
   635→        time.sleep(2)  # Gap so next concurrent callback doesn't hit 630014
   636→    if not result:
   637→        print(f"[callback] Token exchange FAILED for state={state[:12]}... (google_oauth_login returned None)")
   638→        pending["error"] = "Token exchange failed"
   639→        return "<html><body><h1>Login Failed</h1><p>Token exchange failed</p></body></html>", 500
   640→    print(f"[callback] Token exchange OK for state={state[:12]}... user_id={result.get('user_id')}")
   641→
   642→    # Extract email from JWT jti field
   643→    import base64 as _b64
   644→    email = result.get("user_name") or f"user_{result['user_id']}"
   645→    try:
   646→        token_part = result["access_token"].replace("Bearer ", "").split(".")[1]
   647→        token_part += "=" * (4 - len(token_part) % 4)
   648→        payload = json.loads(_b64.urlsafe_b64decode(token_part))
   649→        if payload.get("jti"):
   650→            email = payload["jti"]
   651→    except:
   652→        pass
   653→
   654→    add_token(
   655→        email=email,
   656→        access_token=result["access_token"],
   657→        refresh_token=result["refresh_token"],
   658→        user_id=result["user_id"],
   659→        device_id=device_id,
   660→    )
   661→    pending["result"] = {
   662→        "email": email,
   663→        "user_id": result.get("user_id"),
   664→        "first_login": result.get("first_login"),
   665→    }
   666→    return f"""<html><body>
   667→<h1>Login Success!</h1>
   668→<p>Email: {email}</p>
   669→<p>User ID: {result.get('user_id')}</p>
   670→<p>You can close this tab.</p>
   671→</body></html>"""
