   129→class CircuitState(Enum):
   130→    """3-state circuit breaker: CLOSED → OPEN → HALF_OPEN → CLOSED"""
   131→    CLOSED = 0      # Normal operation — requests flow through
   132→    OPEN = 1        # Circuit tripped — all requests rejected
   133→    HALF_OPEN = 2  # Probe state — one request allowed to test recovery
   134→
   135→
   136→class ChannelCircuitBreaker:
   137→    """Per-channel 3-state circuit breaker with Prometheus tracking.
   138→
   139→    States:
   140→      CLOSED: Normal. All requests pass. Failure count tracked.
   141→      OPEN: Circuit tripped. All requests fail-fast. Timer starts.
   142→      HALF_OPEN: One probe request allowed. If success → CLOSED, if fail → OPEN.
   143→
   144→    Args:
   145→        channel_name: Name for logging/metrics
   146→        failure_threshold: Consecutive failures before opening (default 5)
   147→        recovery_timeout: Seconds before half-open probe (default 30)
   148→        success_threshold: Consecutive successes in HALF_OPEN to close (default 2)
   149→    """
   150→    def __init__(self, channel_name: str, failure_threshold: int = 5,
   151→                 recovery_timeout: float = 30.0, success_threshold: int = 2):
   152→        self.channel_name = channel_name
   153→        self.failure_threshold = failure_threshold
   154→        self.recovery_timeout = recovery_timeout
   155→        self.success_threshold = success_threshold
   156→        self._state = CircuitState.CLOSED
   157→        self._failure_count = 0
   158→        self._success_count = 0
   159→        self._opened_at: float = 0.0
   160→        self._lock = asyncio.Lock()
   161→
   162→    @property
   163→    def state(self) -> CircuitState:
   164→        """Current state, with automatic HALF_OPEN transition on timeout."""
   165→        if self._state == CircuitState.OPEN:
   166→            if time.time() - self._opened_at >= self.recovery_timeout:
   167→                self._state = CircuitState.HALF_OPEN
   168→        return self._state
   169→    @property
   170→    def is_open(self) -> bool:
   171→        """True if circuit is OPEN (requests should be rejected)."""
   172→        return self.state == CircuitState.OPEN
   173→    async def allow_request(self) -> bool:
   174→        """Check if a request should be allowed through."""
   175→        async with self._lock:
   176→            state = self.state
   177→            if state == CircuitState.CLOSED:
   178→                return True
   179→            elif state == CircuitState.HALF_OPEN:
   180→                # Allow ONE probe request through
   181→                return True
   182→            else:  # OPEN
   183→                return False
   184→    async def record_success(self):
   185→        """Record a successful request."""
   186→        async with self._lock:
   187→            if self._state == CircuitState.HALF_OPEN:
   188→                self._success_count += 1
   189→                if self._success_count >= self.success_threshold:
   190→                    self._state = CircuitState.CLOSED
   191→                    self._failure_count = 0
   192→                    self._success_count = 0
   193→                    logger.info(f"Circuit CLOSED for {self.channel_name} — recovered")
   194→            elif self._state == CircuitState.CLOSED:
   195→                self._failure_count = 0  # Reset on success
   196→    async def record_failure(self):
   197→        """Record a failed request."""
   198→        async with self._lock:
   199→            if self._state == CircuitState.HALF_OPEN:
   200→                # Probe failed → back to OPEN
   201→                self._state = CircuitState.OPEN
   202→                self._opened_at = time.time()
   203→                self._success_count = 0
   204→                logger.warning(f"Circuit back to OPEN for {self.channel_name} — probe failed")
   205→            elif self._state == CircuitState.CLOSED:
   206→                self._failure_count += 1
   207→                if self._failure_count >= self.failure_threshold:
   208→                    self._state = CircuitState.OPEN
   209→                    self._opened_at = time.time()
   210→                    logger.warning(
   211→                        f"Circuit OPEN for {self.channel_name} — "
   212→                        f"{self._failure_count} failures >= threshold {self.failure_threshold}"
   213→                    )

   228→@dataclass
   229→class DomainPreference:
   230→    """Per-domain channel preference with EMA-based learning.
   231→
   232→    Tracks success/failure counts per channel and uses exponential moving average
   233→    to compute a quality score. The channel with the highest EMA score becomes
   234→    the preferred_channel for subsequent requests to this domain.
   235→    """
   236→    domain: str
   237→    successes: Dict[str, int] = field(default_factory=lambda: {})
   238→    failures: Dict[str, int] = field(default_factory=lambda: {})
   239→    preferred_channel: str = "http_proxy"
   240→    last_updated: float = field(default_factory=time.time)
   241→    # EMA scores per channel (0.0-1.0)
   242→    _ema_scores: Dict[str, float] = field(default_factory=lambda: {})
   243→    _ema_alpha: float = 0.3  # EMA smoothing factor (higher = more responsive)
   244→
   245→    def record_channel_result(self, channel: str, success: bool):
   246→        """Record a channel result and update EMA score."""
   247→        if success:
   248→            self.successes[channel] = self.successes.get(channel, 0) + 1
   249→        else:
   250→            self.failures[channel] = self.failures.get(channel, 0) + 1
   251→        # EMA update: new_score = alpha * sample + (1 - alpha) * old_score
   252→        old_score = self._ema_scores.get(channel, 0.5)
   253→        sample = 1.0 if success else 0.0
   254→        self._ema_scores[channel] = self._ema_alpha * sample + (1 - self._ema_alpha) * old_score
   255→        self.last_updated = time.time()
   256→        # Re-compute preferred channel from EMA scores
   257→        if self._ema_scores:
   258→            best_channel = max(self._ema_scores, key=self._ema_scores.get)
   259→            if self._ema_scores[best_channel] > 0.3:  # Minimum quality threshold
   260→                self.preferred_channel = best_channel
   261→
   262→    def get_channel_score(self, channel: str) -> float:
   263→        """Get EMA quality score for a channel (0.0-1.0)."""
   264→        return self._ema_scores.get(channel, 0.5)

   269→# ═══════════════════════════════════════════════════════════════

   271→class OpenRouterKeyRotator:
   272→    """Round-robin API key rotation with failover on 429/401/403."""
   273→
   274→    def __init__(self, keys: List[str] = None, base_url: str = "https://openrouter.ai/api/v1"):
   275→        self._keys = keys or []
   276→        self._current_index = 0
   277→        self._base_url = base_url
   278→        self._key_errors: Dict[int, int] = {}
   279→        self._key_cooldown: Dict[int, float] = {}
   280→        self._lock = threading.Lock()
   281→
   282→    @classmethod
   283→    def from_env(cls) -> "OpenRouterKeyRotator":
   284→        keys = []
   285→        primary = os.getenv("OPENAI_API_KEY")
   286→        if primary:
   287→            keys.append(primary)
   288→        for i in range(1, 10):
   289→            key = os.getenv(f"OPENROUTER_KEY_{i}")
   290→            if key:
   291→                keys.append(key)
   292→        base_url = os.getenv("OPENAI_BASE_URL", "https://openrouter.ai/api/v1")
   293→        return cls(keys=keys, base_url=base_url)
   294→
   295→    @property
   296→    def base_url(self) -> str:
   297→        return self._base_url
   298→
   299→    def get_active_key(self) -> Optional[str]:
   300→        with self._lock:
   301→            if not self._keys:
   302→                return None
   303→            now = time.time()
   304→            if now >= self._key_cooldown.get(self._current_index, 0):
   305→                return self._keys[self._current_index]
   306→            for offset in range(len(self._keys)):
   307→                idx = (self._current_index + offset) % len(self._keys)
   308→                if now >= self._key_cooldown.get(idx, 0):
   309→                    self._current_index = idx
   310→                    KEY_ROTATION_COUNT.inc()
   311→                    return self._keys[idx]
   312→            return None
   313→
   314→    def report_error(self, status_code: int, error_type: str = "unknown"):
   315→        with self._lock:
   316→            idx = self._current_index
   317→            self._key_errors[idx] = self._key_errors.get(idx, 0) + 1
   318→            if status_code == 429:
   319→                self._key_cooldown[idx] = time.time() + 60
   320→            elif status_code in (401, 403):
   321→                self._key_cooldown[idx] = time.time() + 300
   322→            else:
   323→                self._key_cooldown[idx] = time.time() + 10
   324→            self._rotate_to_next()
   325→
   326→    def _rotate_to_next(self):
   327→        if len(self._keys) <= 1:
   328→            return
   329→        now = time.time()
   330→        for offset in range(1, len(self._keys)):
   331→            idx = (self._current_index + offset) % len(self._keys)
   332→            if now >= self._key_cooldown.get(idx, 0):
   333→                self._current_index = idx
   334→                KEY_ROTATION_COUNT.inc()
   335→                return
   336→        self._current_index = (self._current_index + 1) % len(self._keys)
   337→        KEY_ROTATION_COUNT.inc()
   338→
   339→    @property
   340→    def total_keys(self) -> int:
   341→        return len(self._keys)
   342→
   343→    @property
   344→    def available_keys(self) -> int:
   345→        now = time.time()
   346→        return sum(1 for i in range(len(self._keys))
   347→                   if now >= self._key_cooldown.get(i, 0))
   348→
   349→    def get_status(self) -> Dict[str, Any]:
   350→        now = time.time()
   351→        return {
   352→            "total_keys": self.total_keys,
   353→            "available_keys": self.available_keys,
   354→            "current_index": self._current_index,
   355→            "key_errors": dict(self._key_errors),
   356→            "cooldowns": {
   357→                i: max(0, self._key_cooldown.get(i, 0) - now)
   358→                for i in range(len(self._keys))
   359→            }
   360→        }

   364→# 2. DNS Flood Protector (from v2)
   365→# ═══════════════════════════════════════════════════════════════
   366→
   367→class DNSFloodProtector:
   368→    """Token-bucket + per-client rate limiting for DNS queries."""
   369→
   370→    def __init__(self, max_qps: int = 50, burst: int = 100,
   371→                 max_clients: int = 50000, client_ttl: float = 300.0):
   372→        self.max_qps = max_qps
   373→        self.burst = burst
   374→        self._tokens: float = float(burst)
   375→        self._last_refill: float = time.time()
   376→        self._lock = asyncio.Lock()
   377→        self._blocked_count = 0
   378→        self._client_queries: Dict[str, deque] = {}
   379→        # Memory Fix M-R2: Client IP eviction + max client cap
   380→        self._client_last_seen: Dict[str, float] = {}
   381→        self._max_clients = max_clients
   382→        self._client_ttl = client_ttl
   383→
   384→    def _evict_stale_clients(self) -> None:
   385→        """Memory Fix M-R2: Evict client IPs idle > client_ttl to prevent unbounded growth."""
   386→        now = time.time()
   387→        stale = [ip for ip, t in self._client_last_seen.items()
   388→                 if now - t > self._client_ttl]
   389→        for ip in stale:
   390→            self._client_queries.pop(ip, None)
   391→            self._client_last_seen.pop(ip, None)
   392→
   393→    async def allow(self, client_ip: str = "default") -> bool:
   394→        async with self._lock:
   395→            now = time.time()
   396→            # Memory Fix M-R2: Evict idle clients periodically
   397→            self._evict_stale_clients()
   398→            # Per-client rate check FIRST (before consuming global token)
   399→            if client_ip not in self._client_queries:
   400→                # Memory Fix M-R2: Cap total tracked clients
   401→                if len(self._client_queries) >= self._max_clients:
   402→                    # Evict oldest client to make room
   403→                    oldest_ip = min(self._client_last_seen, key=self._client_last_seen.get)
   404→                    self._client_queries.pop(oldest_ip, None)
   405→                    self._client_last_seen.pop(oldest_ip, None)
   406→                self._client_queries[client_ip] = deque(maxlen=100)
   407→            self._client_queries[client_ip].append(now)
   408→            self._client_last_seen[client_ip] = now  # Track last activity
   409→            recent = sum(1 for t in self._client_queries[client_ip] if now - t < 1.0)
   410→            if recent > 10:
   411→                self._blocked_count += 1
   412→                DNS_FLOOD_BLOCKED.inc()
   413→                return False
   414→            # Global token bucket check (after per-client passes)
   415→            elapsed = now - self._last_refill
   416→            self._tokens = min(self.burst, self._tokens + elapsed * self.max_qps)
   417→            self._last_refill = now
   418→            if self._tokens >= 1.0:
   419→                self._tokens -= 1.0
   420→                return True
   421→            self._blocked_count += 1
   422→            DNS_FLOOD_BLOCKED.inc()
   423→            return False

   430→@dataclass
   431→class ProxyEntry:
   432→    """Represents a single proxy endpoint (prox5-compatible)."""
   433→    endpoint: str           # host:port
   434→    protocol: str = "socks5"  # socks5, socks4, http
   435→    username: str = ""
   436→    password: str = ""
   437→    proxied_ip: str = ""
   438→    last_validated: float = 0.0
   439→    successes: int = 0
   440→    failures: int = 0
   441→    score: float = 1.0      # EMA quality score
   442→    region: str = ""
   443→
   444→    @property
   445→    def url(self) -> str:
   446→        if self.username and self.password:
   447→            return f"{self.protocol}://{self.username}:{self.password}@{self.endpoint}"
   448→        return f"{self.protocol}://{self.endpoint}"
   449→
   450→    @property
   451→    def is_stale(self) -> bool:
   452→        return time.time() - self.last_validated > 1800  # 30 min
   453→
   454→
   455→class ProxyPoolAdapter:
   456→    """
   457→    Python adapter for prox5-style proxy pool management.
   458→    Loads proxies from file (prox5/proxies.txt format), validates,
   459→    and rotates using Mystery Dialer pattern (retry on failure).
   460→    """
   461→
   462→    def __init__(self, proxy_file: str = None, max_workers: int = 10):
   463→        self._pool: List[ProxyEntry] = []
   464→        self._current_index = 0
   465→        self._lock = threading.Lock()
   466→        self._max_workers = max_workers
   467→        self._proxy_file = proxy_file
   468→        if proxy_file and os.path.exists(proxy_file):
   469→            self.load_from_file(proxy_file)
   470→
   471→    def load_from_file(self, filepath: str):
   472→        """
   473→        Load proxies from file. Supports formats:
   474→          - host:port:user:pass  (autoclaw/prox5 format)
   475→          - protocol://host:port  (URL format)
   476→          - host:port  (simple format)
   477→        """
   478→        count = 0
   479→        with open(filepath) as f:
   480→            for line in f:
   481→                line = line.strip()
   482→                if not line or line.startswith('#'):
   483→                    continue
   484→                entry = self._parse_proxy_line(line)
   485→                if entry:
   486→                    self._pool.append(entry)
   487→                    count += 1
   488→        logger.info(f"Loaded {count} proxies from {filepath}")
   489→        PROXY_POOL_SIZE.labels(protocol="all").set(len(self._pool))
   490→
   491→    def _parse_proxy_line(self, line: str) -> Optional[ProxyEntry]:
   492→        """Parse a proxy line in any supported format."""
   493→        if '://' in line:
   494→            # URL format: socks5://user:pass@host:port
   495→            parsed = urlparse(line)
   496→            protocol = parsed.scheme or "socks5"
   497→            host = parsed.hostname or ""
   498→            port = parsed.port or 1080
   499→            return ProxyEntry(
   500→                endpoint=f"{host}:{port}",
   501→                protocol=protocol,
   502→                username=parsed.username or "",
   503→                password=parsed.password or "",
   504→            )
   505→        parts = line.split(':')
   506→        if len(parts) == 4:
   507→            # host:port:user:pass format
   508→            return ProxyEntry(
   509→                endpoint=f"{parts[0]}:{parts[1]}",
   510→                username=parts[2],
   511→                password=parts[3],
   512→            )
   513→        if len(parts) == 2:
   514→            try:
   515→                int(parts[1])
   516→                return ProxyEntry(endpoint=line)
   517→            except ValueError:
   518→                pass
   519→        return None
   520→
   521→    def add_proxy(self, endpoint: str, protocol: str = "socks5",
   522→                  username: str = "", password: str = ""):
   523→        with self._lock:
   524→            self._pool.append(ProxyEntry(
   525→                endpoint=endpoint, protocol=protocol,
   526→                username=username, password=password
   527→            ))
   528→
   529→    def get_next(self) -> Optional[ProxyEntry]:
   530→        """Get next proxy using round-robin with Mystery Dialer retry."""
   531→        with self._lock:
   532→            if not self._pool:
   533→                return None
   534→            # Try current proxy first
   535→            for offset in range(len(self._pool)):
   536→                idx = (self._current_index + offset) % len(self._pool)
   537→                proxy = self._pool[idx]
   538→                if proxy.score > 0.2 and not proxy.is_stale:
   539→                    self._current_index = idx
   540→                    return proxy
   541→            # All stale/low-score, return best available
   542→            best = max(self._pool, key=lambda p: p.score)
   543→            return best
   544→
   545→    def report_success(self, endpoint: str, latency_ms: float = 0):
   546→        with self._lock:
   547→            for p in self._pool:
   548→                if p.endpoint == endpoint:
   549→                    p.successes += 1

  1103→    def _evict_stale_preferences(self) -> None:
  1104→        """Memory Fix M-R1: Evict DomainPreference entries idle > pref_ttl; hard cap at max_prefs."""
  1105→        now = time.time()
  1106→        stale = [d for d, p in self._prefs.items()
  1107→                 if now - p.last_updated > self._pref_ttl]
  1108→        for d in stale:
  1109→            del self._prefs[d]
  1110→        # Hard cap: evict oldest if still over limit
  1111→        if len(self._prefs) > self._max_prefs:
  1112→            sorted_d = sorted(self._prefs.items(), key=lambda x: x[1].last_updated)
  1113→            for d, _ in sorted_d[:len(self._prefs) - self._max_prefs]:
  1114→                del self._prefs[d]

  1116→    def _cache_get(self, key: str) -> Optional[Any]:
  1117→        if key in self._cache:
  1118→            data, ts = self._cache[key]
  1119→            if time.time() - ts < self._cache_ttl:
  1120→                return data
  1121→            del self._cache[key]
  1122→        return None
  1123→
  1124→    def _cache_set(self, key: str, data: Any):
  1125→        # Evict oldest entries when cache exceeds max size
  1126→        if len(self._cache) >= self._cache_max:
  1127→            oldest_key = min(self._cache, key=lambda k: self._cache[k][1])
  1128→            del self._cache[oldest_key]
  1129→        self._cache[key] = (data, time.time())

  1145→        """Select the optimal channel for a domain."""
  1146→        if force_channel:
  1147→            try:
  1148→                return Channel(force_channel)
  1149→            except ValueError:
  1150→                pass
  1151→
  1152→        pref = self._prefs.get(domain)

  1154→            try:

  1156→            except ValueError:

  1159→        # Default priority: proxy pool → SOCKS → DNS → MITM → direct
  1160→        if self.proxy_pool.valid_count > 0:
  1161→            return Channel.HTTP_PROXY
  1162→        return Channel.DNS_TUNNEL

  1166→    async def _try_http_proxy(self, url: str, domain: str, **kwargs) -> ChannelResult:
  1167→        """Route through proxy pool (prox5/https_proxy)."""
  1168→        start = time.time()
  1169→        proxy = self.proxy_pool.get_next()
  1170→
  1171→        if not proxy:
  1172→            return ChannelResult("http_proxy", False, error="No proxies available")
  1173→
  1174→        try:
  1175→            api_key = self.key_rotator.get_active_key()
  1176→            headers = {}
  1177→            if api_key:
  1178→                headers["Authorization"] = f"Bearer {api_key}"
  1179→
  1180→            # Memory Fix M-R3: Use shared httpx client instead of per-request creation
  1181→            client = await self._get_http_client(proxy=proxy.url)
  1182→            resp = await client.get(url, headers=headers)
  1183→            latency = (time.time() - start) * 1000
  1184→
  1185→            if resp.status_code in (401, 403, 429):
  1186→                self.key_rotator.report_error(
  1187→                    resp.status_code,
  1188→                    "rate_limit" if resp.status_code == 429 else "auth_error"
  1189→                )
  1190→
  1191→            self.proxy_pool.report_success(proxy.endpoint, latency)
  1192→
  1193→            # Only treat 2xx as success — non-2xx should trigger failover
  1194→            is_success = 200 <= resp.status_code < 300
  1195→            if is_success:
  1196→                self._cache_set(url, resp.content)
  1197→
  1198→            return ChannelResult(
  1199→                "http_proxy", is_success,
  1200→                data=resp.content, latency_ms=latency,
  1201→                status_code=resp.status_code, proxy_used=proxy.endpoint
  1202→            )
  1203→        except Exception as e:
  1204→            latency = (time.time() - start) * 1000
  1205→            self.proxy_pool.report_failure(proxy.endpoint)
  1206→            return ChannelResult("http_proxy", False, error=str(e), latency_ms=latency)

  1208→    async def _try_socks_pool(self, url: str, domain: str, **kwargs) -> ChannelResult:

  1232→    async def _try_dns_tunnel(self, url: str, domain: str, **kwargs) -> ChannelResult:
  1233→        """Fall back to DNS tunneling (llm-dns-proxy).
  1234→        Verifies DNS server is actually reachable by sending a health-check query."""
  1235→        start = time.time()
  1236→        try:
  1237→            import socket
  1238→            dns_host = os.getenv("DNS_SERVER_HOST", "127.0.0.1")
  1239→            dns_port = int(os.getenv("DNS_SERVER_PORT", "5353"))
  1240→            sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
  1241→            sock.settimeout(2)
  1242→            # Send a minimal DNS query to verify the server is actually running
  1243→            # DNS header: ID=0x1234, RD=1, QDCOUNT=1, QNAME=., QTYPE=TXT, QCLASS=IN
  1244→            query = b'\x12\x34\x01\x00\x00\x01\x00\x00\x00\x00\x00\x00\x00\x00\x10\x00\x01'
  1245→            sock.sendto(query, (dns_host, dns_port))
  1246→            # Wait for any response (even SERVFAIL proves the server is alive)
  1247→            data, _ = sock.recvfrom(512)
  1248→            sock.close()
  1249→            if not data:
  1250→                return ChannelResult("dns_tunnel", False, error="DNS server returned empty response")
  1251→            return ChannelResult(
  1252→                "dns_tunnel", True,
  1253→                data=f"[DNS tunnel verified for {domain}]",
  1254→                latency_ms=(time.time() - start) * 1000
  1255→            )
  1256→        except socket.timeout:
  1257→            return ChannelResult("dns_tunnel", False, error="DNS server not reachable (timeout)")
  1258→        except ConnectionRefusedError:
  1259→            return ChannelResult("dns_tunnel", False, error="DNS server not running (connection refused)")
  1260→        except Exception as e:
  1261→            return ChannelResult("dns_tunnel", False, error=str(e))

  1295→    async def fetch(self, url: str, force_channel: str = None, **kwargs) -> ChannelResult:
  1296→        """
  1297→        Fetch a URL through the optimal channel with automatic failover.
  1298→        Now includes per-channel circuit breaker and EMA domain preference learning.
  1299→
  1300→        Channel selection priority:
  1301→          cached → http_proxy → socks_pool → dns_tunnel → mitm_stealth → connect_chain → http_direct
  1302→        """
  1303→        domain = self._extract_domain(url)
  1304→
  1305→        # Memory Fix M-R1: Evict stale domain preferences on each request
  1306→        self._evict_stale_preferences()
  1307→
  1308→        # Check cache
  1309→        cached = self._cache_get(url)
  1310→        if cached and not force_channel:
  1311→            return ChannelResult("cached", True, data=cached)
  1312→
  1313→        # Select channel (uses DomainPreference EMA if available)
  1314→        channel = self._select_channel(domain, force_channel)
  1315→
  1316→        # Channel implementations
  1317→        channel_map = {
  1318→            Channel.HTTP_PROXY: self._try_http_proxy,
  1319→            Channel.SOCKS_POOL: self._try_socks_pool,
  1320→            Channel.DNS_TUNNEL: self._try_dns_tunnel,
  1321→            Channel.MITM_STEALTH: self._try_mitm_stealth,
  1322→            Channel.CONNECT_CHAIN: self._try_connect_chain,
  1323→            Channel.HTTP_DIRECT: self._try_http_direct,
  1324→        }
  1325→
  1326→        # Try preferred channel (with circuit breaker check)
  1327→        if channel in channel_map:
  1328→            cb = self._circuit_breakers.get(channel.value)
  1329→            if cb and not await cb.allow_request():
  1330→                logger.debug(f"Circuit OPEN for {channel.value} — skipping")
  1331→            else:
  1332→                result = await channel_map[channel](url, domain, **kwargs)
  1333→                # Record result in circuit breaker and EMA
  1334→                if cb:
  1335→                    if result.success:
  1336→                        await cb.record_success()
  1337→                    else:
  1338→                        await cb.record_failure()
  1339→                # Update domain preference EMA
  1340→                pref = self._prefs.get(domain)
  1341→                if not pref:
  1342→                    pref = DomainPreference(domain=domain)
  1343→                    self._prefs[domain] = pref
  1344→                pref.record_channel_result(result.channel, result.success)
  1345→
  1346→                if result.success:
  1347→                    REQUESTS_TOTAL.labels(channel=result.channel, domain=domain, status="success").inc()
  1348→                    REQUESTS_DURATION.labels(channel=result.channel, domain=domain).observe(result.latency_ms / 1000)
  1349→                    return result
  1350→                # Record failure in Prometheus
  1351→                REQUESTS_TOTAL.labels(channel=result.channel, domain=domain, status="error").inc()
  1352→
  1353→        # Fallback: try all channels in priority order
  1354→        # NOTE: CONNECT_CHAIN was missing — added between MITM and HTTP_DIRECT
  1355→        fallback_order = [
  1356→            Channel.HTTP_PROXY, Channel.SOCKS_POOL, Channel.DNS_TUNNEL,
  1357→            Channel.MITM_STEALTH, Channel.CONNECT_CHAIN, Channel.HTTP_DIRECT,
  1358→        ]
  1359→        for ch in fallback_order:
  1360→            if ch == channel:
  1361→                continue  # Already tried
  1362→            if ch not in channel_map:
  1363→                continue
  1364→
  1365→            # Circuit breaker check
  1366→            cb = self._circuit_breakers.get(ch.value)
  1367→            if cb and not await cb.allow_request():
  1368→                logger.debug(f"Circuit OPEN for {ch.value} — skipping fallback")
  1369→                continue
  1370→
  1371→            result = await channel_map[ch](url, domain, **kwargs)
  1372→
  1373→            # Record in circuit breaker and EMA
  1374→            if cb:
  1375→                if result.success:
  1376→                    await cb.record_success()
  1377→                else:
  1378→                    await cb.record_failure()
  1379→            pref = self._prefs.get(domain)
  1380→            if pref:
  1381→                pref.record_channel_result(result.channel, result.success)
  1382→
  1383→            if result.success:
  1384→                CHANNEL_SWITCHES.labels(
  1385→                    from_channel=channel.value, to_channel=ch.value
  1386→                ).inc()
  1387→                REQUESTS_TOTAL.labels(channel=result.channel, domain=domain, status="success").inc()
  1388→                REQUESTS_DURATION.labels(channel=result.channel, domain=domain).observe(result.latency_ms / 1000)
  1389→                return result
  1390→            # Record failure
  1391→            REQUESTS_TOTAL.labels(channel=result.channel, domain=domain, status="error").inc()
  1392→
  1393→        REQUESTS_TOTAL.labels(channel="none", domain=domain, status="all_failed").inc()
  1394→        return ChannelResult("none", False, error=f"All channels exhausted for {domain}")
