    22→class SimpleSpinner:
    23→    def __init__(self, message="Loading"):
    24→        self.message = message
    25→        self.running = False
    26→        self.thread = None
    27→        self.spinner_chars = "|/-\\"
    28→        self.current = 0
    29→
    30→    def _spin(self):
    31→        while self.running:
    32→            sys.stdout.write(f"\r{self.message} {self.spinner_chars[self.current % len(self.spinner_chars)]}")
    33→            sys.stdout.flush()
    34→            self.current += 1
    35→            time.sleep(0.1)
    36→        sys.stdout.write("\r" + " " * (len(self.message) + 2) + "\r")
    37→        sys.stdout.flush()
    38→
    39→    def start(self):
    40→        self.running = True
    41→        self.thread = threading.Thread(target=self._spin)
    42→        self.thread.start()
    43→
    44→    def stop(self):
    45→        self.running = False
    46→        if self.thread:
    47→            self.thread.join()

    50→class DNSLLMClient:
    51→    def __init__(self, server_host: str = "127.0.0.1", server_port: int = 5353, crypto_key: bytes = None, verbose: bool = False, poll_interval: float = 0.1, model: str = "LLM"):
    52→        self.server_host = server_host
    53→        self.server_port = server_port
    54→        self.crypto = CryptoManager(crypto_key)
    55→        self.chunker = DNSChunker()
    56→        self.sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
    57→        self.sock.settimeout(30.0)
    58→        self.verbose = verbose
    59→        self.poll_interval = poll_interval
    60→        self.model = model
    61→        self.client_version = get_version_string()
    62→        # Use a single session ID for the entire client session to avoid collisions
    63→        # Use 3-digit session ID (000-999) for much better uniqueness
    64→        # Mix timestamp and random for collision avoidance
    65→        time_component = int(time.time()) % 1000
    66→        random_component = uuid.uuid4().int % 1000
    67→        session_num = (time_component + random_component) % 1000
    68→        self.session_id = f"{session_num:03d}"  # Zero-padded 3 digits
    69→
    70→    def _send_dns_query(self, query_name: str) -> Optional[str]:
    71→        """Send a DNS query and return the TXT record response."""
    72→        try:
    73→            # Manual DNS record construction to avoid dnslib issues
    74→            question = DNSQuestion(query_name, QTYPE.TXT)
    75→            header = DNSHeader(id=random.randint(1, 65535))
    76→            q = DNSRecord(header, q=question)
    77→            query_data = q.pack()
    78→
    79→            self.sock.sendto(query_data, (self.server_host, self.server_port))
    80→
    81→            response_data, _ = self.sock.recvfrom(4096)
    82→            response = DNSRecord.parse(response_data)
    83→
    84→            for rr in response.rr:
    85→                if rr.rtype == QTYPE.TXT:
    86→                    # Handle TXT RDATA which can be a list of strings (each ≤255 bytes)
    87→                    rdata = rr.rdata
    88→
    89→                    # dnslib TXT RDATA has a 'data' attribute that's a list of byte strings
    90→                    if hasattr(rdata, 'data') and isinstance(rdata.data, list):
    91→                        # Join all byte strings and decode
    92→                        txt_bytes = b''.join(rdata.data)
    93→                        txt_data = txt_bytes.decode('utf-8', errors='ignore')
    94→                    else:
    95→                        # Fallback to string conversion
    96→                        txt_data = str(rdata)
    97→                        # Handle quoted TXT records
    98→                        if txt_data.startswith('"') and txt_data.endswith('"'):
    99→                            txt_data = txt_data[1:-1]
   100→
   101→                    return txt_data
   102→
   103→            return None
   104→
   105→        except Exception as e:
   106→            if self.verbose:
   107→                click.echo(f"DNS query error: {e}")
   108→            return None

   110→    def send_message(self, message: str, show_spinner: bool = True, streaming: bool = True) -> Optional[str]:
   111→        """Send a message to the LLM through DNS and get the response.
   112→
   113→        Args:
   114→            message: The message to send
   115→            show_spinner: Whether to show spinner during processing
   116→            streaming: Whether to display streaming response (default True)
   117→        """
   118→        # Use the persistent session ID for this client instance
   119→        session_id = self.session_id
   120→        if self.verbose:
   121→            click.echo(f"Using session ID: {session_id}")
   122→        spinner = None
   123→
   124→        try:
   125→            if self.verbose:
   126→                click.echo(f"Encrypting message...")
   127→            encrypted_data = self.crypto.encrypt(message)
   128→
   129→            if self.verbose:
   130→                click.echo(f"Creating message chunks...")
   131→            chunks = self.chunker.create_chunks(encrypted_data, session_id)
   132→
   133→            if show_spinner and not self.verbose:
   134→                spinner = SimpleSpinner("...")
   135→                spinner.start()
   136→
   137→            if self.verbose:
   138→                click.echo(f"Sending {len(chunks)} chunks...")
   139→
   140→            for i, chunk in enumerate(chunks):
   141→                if self.verbose:
   142→                    click.echo(f"Sending chunk {i+1}/{len(chunks)}")
   143→                response = self._send_dns_query(chunk)
   144→                if response != "OK" and self.verbose:
   145→                    click.echo(f"Warning: Unexpected response for chunk {i+1}: {response}")
   146→
   147→            if spinner:
   148→                spinner.stop()
   149→
   150→            if streaming and not self.verbose:
   151→                return self._handle_streaming_response(session_id)
   152→            else:
   153→                return self._handle_traditional_response(session_id, show_spinner)
   154→
   155→        finally:
   156→            if spinner:
   157→                spinner.stop()

   159→    def _handle_streaming_response(self, session_id: str) -> Optional[str]:
   160→        """Handle streaming response by displaying individual chunks immediately."""
   161→        import sys
   162→
   163→        displayed_chunks = set()  # Track which chunks we've already displayed
   164→        final_response = ""
   165→        start_time = time.time()
   166→        max_wait_time = 60  # Maximum wait time in seconds
   167→        last_chunk_time = time.time()
   168→
   169→        # Initial wait for processing to start
   170→        time.sleep(self.poll_interval)
   171→
   172→        while True:
   173→            # Check for timeout
   174→            current_time = time.time()
   175→            if current_time - start_time > max_wait_time:
   176→                if self.verbose:
   177→                    click.echo(f"\nTimeout waiting for response after {max_wait_time}s")
   178→                break
   179→
   180→            # Check if no new chunks for too long (indicates completion)
   181→            if current_time - last_chunk_time > 10:  # 10 seconds without new chunks
   182→                if final_response and '[EOS]' in final_response:
   183→                    break
   184→                elif len(final_response) > 50:  # Only timeout if we have substantial content
   185→                    if not final_response.endswith('\n'):
   186→                        sys.stdout.write('\n')
   187→                        sys.stdout.flush()
   188→                    final_response += "[Response may be incomplete - connection timed out]"
   189→                    break
   190→
   191→            response_chunks = self._get_current_response_chunks(session_id)
   192→
   193→            if not response_chunks:
   194→                time.sleep(self.poll_interval)
   195→                continue
   196→
   197→            # Process each new chunk individually and display immediately
   198→            new_chunks_found = False
   199→            for chunk_index, chunk_data in response_chunks.items():
   200→                if chunk_index not in displayed_chunks:
   201→                    displayed_chunks.add(chunk_index)
   202→                    new_chunks_found = True
   203→                    last_chunk_time = current_time
   204→
   205→                    try:
   206→                        # Try to decrypt this individual chunk
   207→                        parts = chunk_data.split(':', 2)
   208→                        if len(parts) == 3:
   209→                            encrypted_data = parts[2].encode('ascii')
   210→                            try:
   211→                                # Try streaming chunk decryption
   212→                                chunk_text = self.crypto.decrypt_chunk(encrypted_data)
   213→                            except Exception:
   214→                                # Fall back to regular decryption if chunk method fails
   215→                                chunk_text = self.crypto.decrypt(encrypted_data)
   216→
   217→                            # Display chunk content immediately (remove [EOS] marker)
   218→                            display_text = chunk_text.replace('[EOS]', '')
   219→                            if display_text:
   220→                                sys.stdout.write(display_text)
   221→                                sys.stdout.flush()
   222→                                final_response += display_text
   223→
   224→                            # Check if this chunk contains [EOS] marker
   225→                            if '[EOS]' in chunk_text:
   226→                                if not final_response.endswith('\n'):
   227→                                    sys.stdout.write('\n')
   228→                                    sys.stdout.flush()
   229→                                return final_response.rstrip()
   230→
   231→                    except Exception as e:
   232→                        if self.verbose:
   233→                            click.echo(f"\nError decrypting chunk {chunk_index}: {e}")
   234→                        continue
   235→
   236→            if not new_chunks_found:
   237→                time.sleep(self.poll_interval)
   238→
   239→        # Return final response (timeout case)
   240→        return final_response.rstrip() if final_response else None

   242→    def _get_current_response_chunks(self, session_id: str) -> dict:
   243→        """Get current response chunks from server."""
   244→        response_chunks = {}
   245→        chunk_index = 0
   246→        max_chunks = 100  # Increased limit for large responses
   247→        total_chunks = None
   248→        consecutive_not_found = 0
   249→        max_consecutive_not_found = 50  # Allow more gaps - server may be generating chunks slowly
   250→
   251→        while chunk_index < max_chunks and consecutive_not_found < max_consecutive_not_found:
   252→            query = format_dns_query("g", session_id, chunk_index)
   253→            response = self._send_dns_query(query)
   254→
   255→            if response == "NOT_FOUND":
   256→                consecutive_not_found += 1
   257→                chunk_index += 1
   258→                continue
   259→            else:
   260→                consecutive_not_found = 0  # Reset counter on successful retrieval
   261→
   262→            if response and ':' in response:
   263→                parts = response.split(':', 2)
   264→                if len(parts) == 3:
   265→                    try:
   266→                        current_index = int(parts[0])
   267→                        chunk_total = int(parts[1])
   268→                        response_chunks[current_index] = response
   269→
   270→                        # Update total_chunks if we haven't seen it yet or if it's larger
   271→                        if total_chunks is None or chunk_total > total_chunks:
   272→                            total_chunks = chunk_total
   273→                            # Don't limit max_chunks yet - server might still be generating more
   274→
   275→                        # If we have all chunks we expect, we can break early
   276→                        if total_chunks and len(response_chunks) >= total_chunks:
   277→                            # Verify we have a contiguous sequence from 0 to total_chunks-1
   278→                            if all(i in response_chunks for i in range(total_chunks)):
   279→                                break
   280→                    except (ValueError, IndexError):
   281→                        if self.verbose:
   282→                            click.echo(f"Error parsing chunk response: {response}")
   283→
   284→            chunk_index += 1
   285→
   286→        # If we know how many chunks we should have, try to wait for missing ones
   287→        if total_chunks and len(response_chunks) < total_chunks:
   288→            missing_chunks = [i for i in range(total_chunks) if i not in response_chunks]
   289→            if self.verbose:
   290→                click.echo(f"Waiting for {len(missing_chunks)} missing chunks: {missing_chunks}")
   291→
   292→            # Implement chunk-level retries with exponential backoff for flaky networks
   293→            max_retries = 5
   294→            for retry in range(max_retries):
   295→                if not missing_chunks:
   296→                    break
   297→
   298→                if self.verbose:
   299→                    click.echo(f"Retry {retry + 1}/{max_retries}: Requesting {len(missing_chunks)} missing chunks...")
   300→
   301→                # Exponential backoff: 0.5, 1.0, 2.0, 4.0, 8.0 seconds
   302→                import time
   303→                wait_time = 0.5 * (2 ** retry)
   304→                time.sleep(wait_time)
   305→
   306→                # Request specific missing chunks
   307→                chunks_retrieved_this_round = 0
   308→                for missing_idx in missing_chunks[:]:
   309→                    query = format_dns_query("g", session_id, missing_idx)
   310→                    response = self._send_dns_query(query)
   311→
   312→                    if self._validate_chunk_response(response, missing_idx):
   313→                        response_chunks[missing_idx] = response
   314→                        missing_chunks.remove(missing_idx)
   315→                        chunks_retrieved_this_round += 1
   316→                        if self.verbose:
   317→                            click.echo(f"  ✓ Retrieved missing chunk {missing_idx}")
   318→                    elif self.verbose and response and response != "NOT_FOUND":
   319→                        click.echo(f"  ✗ Invalid chunk {missing_idx}: {response[:30]}...")
   320→
   321→                if self.verbose:
   322→                    click.echo(f"  Retrieved {chunks_retrieved_this_round} chunks this round")
   323→
   324→                if not missing_chunks:  # All chunks found
   325→                    break
   326→
   327→            if self.verbose and missing_chunks:
   328→                click.echo(f"Still missing {len(missing_chunks)} chunks after all retries: {missing_chunks}")
   329→
   330→        return response_chunks

   332→    def _validate_chunk_response(self, response: str, expected_index: int) -> bool:
   333→        """Validate chunk response format and integrity.
   334→
   335→        Args:
   336→            response: Raw TXT record response
   337→            expected_index: Expected chunk index
   338→
   339→        Returns:
   340→            True if valid, False otherwise
   341→        """
   342→        if not response or response == "NOT_FOUND":
   343→            return False
   344→
   345→        parts = response.split(':', 2)
   346→        if len(parts) != 3:
   347→            return False
   348→
   349→        try:
   350→            actual_index = int(parts[0])
   351→            total_chunks = int(parts[1])
   352→            data = parts[2]
   353→
   354→            # Validate index matches expectation
   355→            if actual_index != expected_index:
   356→                return False
   357→
   358→            # Validate data is not empty
   359→            if not data:
   360→                return False
   361→
   362→            # Basic format validation - should be base64-like Fernet token
   363→            if len(data) < 20:  # Fernet tokens are at least ~44 chars
   364→                return False
   365→
   366→            return True
   367→
   368→        except (ValueError, IndexError):
   369→            return False

   371→    def _handle_traditional_response(self, session_id: str, show_spinner: bool) -> Optional[str]:
   372→        """Handle traditional non-streaming response."""
   373→        spinner = None
   374→
   375→        try:
   376→            if show_spinner and not self.verbose:
   377→                spinner = SimpleSpinner("-->")
   378→                spinner.start()
   379→            elif self.verbose:
   380→                click.echo("Message sent, waiting for processing...")
   381→
   382→            # Wait longer for server to generate response, especially for long responses
   383→            time.sleep(1)  # Increased wait time for server generation
   384→
   385→            if spinner:
   386→                spinner.stop()
   387→                spinner = SimpleSpinner("<--")
   388→                spinner.start()
   389→            elif self.verbose:
   390→                click.echo("Retrieving response chunks...")
   391→
   392→            # Use the improved chunk retrieval method
   393→            # Try multiple times until we get a complete response (marked with [EOS])
   394→            response_chunks = {}
   395→            max_attempts = 5
   396→
   397→            for attempt in range(max_attempts):
   398→                response_chunks = self._get_current_response_chunks(session_id)
   399→
   400→                if response_chunks:
   401→                    # Try to reassemble and check if it's complete
   402→                    try:
   403→                        reassembled = self.chunker.reassemble_response(response_chunks)
   404→                        if reassembled:
   405→                            decrypted = self.crypto.decrypt(reassembled)
   406→                            if '[EOS]' in decrypted:
   407→                                # Response is complete
   408→                                break
   409→                    except Exception:
   410→                        pass  # Decryption failed, keep trying
   411→
   412→                if attempt < max_attempts - 1:  # Don't sleep on last attempt
   413→                    if self.verbose:
   414→                        click.echo(f"Response incomplete, waiting... (attempt {attempt + 1}/{max_attempts})")
   415→                    time.sleep(2)  # Wait before retrying
   416→
   417→            if spinner:
   418→                spinner.stop()
   419→
   420→            if not response_chunks:
   421→                if not self.verbose:
   422→                    click.echo("No response received from server")
   423→                return None
   424→
   425→            if self.verbose:
   426→                click.echo(f"Received {len(response_chunks)} response chunks")
   427→
   428→            try:
   429→                complete_encrypted_response = self.chunker.reassemble_response(response_chunks)
   430→                decrypted_response = self.crypto.decrypt(complete_encrypted_response)
   431→                return decrypted_response
   432→            except Exception as e:
   433→                if self.verbose:
   434→                    click.echo(f"Error decrypting response: {e}")
   435→                    click.echo(f"Encrypted data length: {len(complete_encrypted_response) if complete_encrypted_response else 0}")
   436→                    if complete_encrypted_response:
   437→                        click.echo(f"Encrypted data starts with: {complete_encrypted_response[:50]}")
   438→                return None
   439→
   440→        finally:
   441→            if spinner:
   442→                spinner.stop()

   444→    def get_server_info(self) -> Optional[Dict[str, str]]:
   445→        """Get the server version and model info."""
   446→        import json
   447→        try:
   448→            version_query = format_dns_query("v")
   449→            response = self._send_dns_query(version_query)
   450→            if response and response != "NOT_FOUND":
   451→                # Handle quoted TXT records
   452→                if response.startswith('"') and response.endswith('"'):
   453→                    response = response[1:-1]
   454→
   455→                # Try to parse as JSON (new format)
   456→                try:
   457→                    return json.loads(response)
   458→                except json.JSONDecodeError:
   459→                    # Fallback to old format (just version string)
   460→                    return {"version": response, "model": None}
   461→            return None
   462→        except Exception:
   463→            return None

   465→    def test_connection(self) -> bool:
   466→        """Test basic connectivity to the DNS server."""
   467→        try:
   468→            # Try a simple DNS query to test connectivity
   469→            test_query = format_dns_query("t")
   470→            question = DNSQuestion(test_query, QTYPE.TXT)
   471→            header = DNSHeader(id=random.randint(1, 65535))
   472→            q = DNSRecord(header, q=question)
   473→            query_data = q.pack()
   474→
   475→            # Set shorter timeout for connection test
   476→            original_timeout = self.sock.gettimeout()
   477→            self.sock.settimeout(5.0)
   478→
   479→            self.sock.sendto(query_data, (self.server_host, self.server_port))
   480→            response_data, _ = self.sock.recvfrom(4096)
   481→
   482→            # Restore original timeout
   483→            self.sock.settimeout(original_timeout)
   484→
   485→            # If we got any response, connection is working
   486→            return True
   487→        except Exception:
   488→            return False

   511→@cli.command()
   512→@click.option('--server', default='127.0.0.1', help='DNS server address')
   513→@click.option('--port', default=5353, help='DNS server port')
   514→@click.option('--message', '-m', help='Single message to send')
   515→@click.option('--verbose', '-v', is_flag=True, help='Enable verbose output')
   516→@click.option('--poll-interval', default=0.1, help='Polling interval in seconds for streaming responses')
   517→@click.option('--model', default='LLM', help='Model name to display in chat (for display purposes only)')
   518→def chat(server, port, message, verbose, poll_interval, model):
   519→    """Start a chat session with the LLM through DNS."""
   520→    client = DNSLLMClient(server, port, verbose=verbose, poll_interval=poll_interval, model=model)
   521→
   522→    # Test connection first
   523→    if not verbose:
   524→        spinner = SimpleSpinner("Testing connection")
   525→        spinner.start()
   526→
   527→    connection_ok = client.test_connection()
   528→
   529→    if not verbose:
   530→        spinner.stop()
   531→
   532→    if connection_ok:
   533→        # Get server info
   534→        server_info = client.get_server_info()
   535→        client_version = client.client_version
   536→
   537→        if server_info:
   538→            server_version = server_info.get("version")
   539→            server_model = server_info.get("model")
   540→
   541→            if server_model:
   542→                click.echo(f"✓ Connected to DNS server at {server}:{port}")
   543→                click.echo(f"  Server: {server_version} (model: {server_model})")
   544→                click.echo(f"  Client: {client_version}")
   545→            else:
   546→                click.echo(f"✓ Connected to DNS server at {server}:{port} (server: {server_version}, client: {client_version})")
   547→
   548→            # Check for version mismatch
   549→            if server_version and server_version != client_version:
   550→                click.echo(f"⚠️  WARNING: Version mismatch detected!")
   551→                click.echo(f"   Server version: {server_version}")
   552→                click.echo(f"   Client version: {client_version}")
   553→        else:
   554→            click.echo(f"✓ Connected to DNS server at {server}:{port} (client: {client_version})")
   555→            click.echo("⚠️  Could not retrieve server version - using older server?")
   556→
   557→        if verbose:
   558→            click.echo("Connection test successful - chat is ready!")
   559→    else:
   560→        click.echo(f"✗ Failed to connect to DNS server at {server}:{port}")
   561→        click.echo("Please check that the server is running and accessible.")
   562→        client.close()
   563→        return
   564→
   565→    try:
   566→        if message:
   567→            response = client.send_message(message, show_spinner=not verbose, streaming=not verbose)
   568→            if response and verbose:
   569→                click.echo(response)
   570→            elif not verbose:
   571→                click.echo()  # Add newline after streaming response
   572→        else:
   573→            click.echo("Chat session started. Type '/quit' to exit, '/help' for commands.")
   574→            click.echo()
   575→
   576→            while True:
   577→                try:
   578→                    user_input = input("> ")
