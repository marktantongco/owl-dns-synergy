   467→  1166→        except Exception as e:
   468→  1167→            latency = (time.time() - start) * 1000
   469→  1168→            self.proxy_pool.report_failure(proxy.endpoint)

   496→  1195→    async def _try_dns_tunnel(self, url: str, domain: str, **kwargs) -> ChannelResult:
   497→  1196→        """Fall back to DNS tunneling (llm-dns-proxy).
   498→  1197→        Verifies DNS server is actually reachable by sending a health-check query."""
   499→  1198→        start = time.time()
   500→  1199→        try:
   501→  1200→            import socket
   502→  1201→            dns_host = os.getenv("DNS_SERVER_HOST", "127.0.0.1")
   503→  1202→            dns_port = int(os.getenv("DNS_SERVER_PORT", "5353"))
   504→  1203→            sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
   505→  1204→            sock.settimeout(2)
   506→  1205→            # Send a minimal DNS query to verify the server is actually running
   507→  1206→            # DNS header: ID=0x1234, RD=1, QDCOUNT=1, QNAME=., QTYPE=TXT, QCLASS=IN
   508→  1207→            query = b'\x12\x34\x01\x00\x00\x01\x00\x00\x00\x00\x00\x00\x00\x00\x10\x00\x01'
   509→  1208→            sock.sendto(query, (dns_host, dns_port))
   510→  1209→            # Wait for any response (even SERVFAIL proves the server is alive)
   511→  1210→            data, _ = sock.recvfrom(512)
   512→  1211→            sock.close()
   513→  1212→            if not data:
   514→  1213→                return ChannelResult("dns_tunnel", False, error="DNS server returned empty response")
   515→  1214→            return ChannelResult(
   516→  1215→                "dns_tunnel", True,
   517→  1216→                data=f"[DNS tunnel verified for {domain}]",
   518→  1217→                latency_ms=(time.time() - start) * 1000
   519→  1218→            )
   520→  1219→        except socket.timeout:
   521→  1220→            return ChannelResult("dns_tunnel", False, error="DNS server not reachable (timeout)")
   522→  1221→        except ConnectionRefusedError:
   523→  1222→            return ChannelResult("dns_tunnel", False, error="DNS server not running (connection refused)")
   524→  1223→        except Exception as e:
   525→  1224→            return ChannelResult("dns_tunnel", False, error=str(e))

   559→  1258→    async def fetch(self, url: str, force_channel: str = None, **kwargs) -> ChannelResult:
   560→  1259→        """
   561→  1260→        Fetch a URL through the optimal channel with automatic failover.
   562→  1261→        Now includes per-channel circuit breaker and EMA domain preference learning.
   563→  1262→
   564→  1263→        Channel selection priority:
   565→  1264→          cached → http_proxy → socks_pool → dns_tunnel → mitm_stealth → connect_chain → http_direct
   566→  1265→        """
   567→  1266→        domain = self._extract_domain(url)
   568→  1267→
   569→  1268→        # Check cache
   570→  1269→        cached = self._cache_get(url)
   571→  1270→        if cached and not force_channel:
   572→  1271→            return ChannelResult("cached", True, data=cached)
   573→  1272→
   574→  1273→        # Select channel (uses DomainPreference EMA if available)
   575→  1274→        channel = self._select_channel(domain, force_channel)
   576→  1275→
   577→  1276→        # Channel implementations
   578→  1277→        channel_map = {
   579→  1278→            Channel.HTTP_PROXY: self._try_http_proxy,
   580→  1279→            Channel.SOCKS_POOL: self._try_socks_pool,
   581→  1280→            Channel.DNS_TUNNEL: self._try_dns_tunnel,
   582→  1281→            Channel.MITM_STEALTH: self._try_mitm_stealth,
   583→  1282→            Channel.CONNECT_CHAIN: self._try_connect_chain,
   584→  1283→            Channel.HTTP_DIRECT: self._try_http_direct,
   585→  1284→        }
   586→  1285→
   587→  1286→        # Try preferred channel (with circuit breaker check)
   588→  1287→        if channel in channel_map:
   589→  1288→            cb = self._circuit_breakers.get(channel.value)
   590→  1289→            if cb and not await cb.allow_request():
   591→  1290→                logger.debug(f"Circuit OPEN for {channel.value} — skipping")
   592→  1291→            else:
   593→  1292→                result = await channel_map[channel](url, domain, **kwargs)
   594→  1293→                # Record result in circuit breaker and EMA
   595→  1294→                if cb:
   596→  1295→                    if result.success:
   597→  1296→                        await cb.record_success()
   598→  1297→                    else:
   599→  1298→                        await cb.record_failure()
   600→  1299→                # Update domain preference EMA
   601→  1300→                pref = self._prefs.get(domain)
   602→  1301→                if not pref:
   603→  1302→                    pref = DomainPreference(domain=domain)
   604→  1303→                    self._prefs[domain] = pref
   605→  1304→                pref.record_channel_result(result.channel, result.success)
   606→  1305→
   607→  1306→                if result.success:
   608→  1307→                    REQUESTS_TOTAL.labels(channel=result.channel, domain=domain, status="success").inc()
   609→  1308→                    REQUESTS_DURATION.labels(channel=result.channel, domain=domain).observe(result.latency_ms / 1000)
   610→  1309→                    return result
   611→  1310→                # Record failure in Prometheus
   612→  1311→                REQUESTS_TOTAL.labels(channel=result.channel, domain=domain, status="error").inc()
   613→  1312→
   614→  1313→        # Fallback: try all channels in priority order
   615→  1314→        # NOTE: CONNECT_CHAIN was missing — added between MITM and HTTP_DIRECT
   616→  1315→        fallback_order = [
   617→  1316→            Channel.HTTP_PROXY, Channel.SOCKS_POOL, Channel.DNS_TUNNEL,
   618→  1317→            Channel.MITM_STEALTH, Channel.CONNECT_CHAIN, Channel.HTTP_DIRECT,
   619→  1318→        ]
   620→  1319→        for ch in fallback_order:
   621→  1320→            if ch == channel:
   622→  1321→                continue  # Already tried
   623→  1322→            if ch not in channel_map:
   624→  1323→                continue
   625→  1324→
   626→  1325→            # Circuit breaker check
   627→  1326→            cb = self._circuit_breakers.get(ch.value)
   628→  1327→            if cb and not await cb.allow_request():
   629→  1328→                logger.debug(f"Circuit OPEN for {ch.value} — skipping fallback")
   630→  1329→                continue
   631→  1330→
   632→  1331→            result = await channel_map[ch](url, domain, **kwargs)
   633→  1332→
   634→  1333→            # Record in circuit breaker and EMA
   635→  1334→            if cb:
   636→  1335→                if result.success:
   637→  1336→                    await cb.record_success()
   638→  1337→                else:
   639→  1338→                    await cb.record_failure()
   640→  1339→            pref = self._prefs.get(domain)
   641→  1340→            if pref:
   642→  1341→                pref.record_channel_result(result.channel, result.success)
   643→  1342→
   644→  1343→            if result.success:
   645→  1344→                CHANNEL_SWITCHES.labels(
   646→  1345→                    from_channel=channel.value, to_channel=ch.value
   647→  1346→                ).inc()
   648→  1347→                REQUESTS_TOTAL.labels(channel=result.channel, domain=domain, status="success").inc()
   649→  1348→                REQUESTS_DURATION.labels(channel=result.channel, domain=domain).observe(result.latency_ms / 1000)
   650→  1349→                return result
   651→  1350→            # Record failure
   652→  1351→            REQUESTS_TOTAL.labels(channel=result.channel, domain=domain, status="error").inc()
   653→  1352→
   654→  1353→        REQUESTS_TOTAL.labels(channel="none", domain=domain, status="all_failed").inc()
   655→  1354→        return ChannelResult("none", False, error=f"All channels exhausted for {domain}")
