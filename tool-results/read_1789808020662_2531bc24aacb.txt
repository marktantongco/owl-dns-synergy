   114→   114→function createUsageTracker(onComplete) {
   115→   115→  const state = { toolCalls: new Map(), text: "", reasoning: "", inputTokens: 0, outputTokens: 0 };
   116→   116→  let buffer = "";
   117→   117→  let settled = false;
   118→   118→  const finish = () => {
   119→   119→    if (settled) return;
   120→   120→    settled = true;
   121→   121→    onComplete({
   122→   122→      prompt_tokens: state.inputTokens,
   123→   123→      completion_tokens: state.outputTokens,
   124→   124→      total_tokens: state.inputTokens + state.outputTokens,
   125→   125→    });
   126→   126→  };
   127→   127→  return {
   128→   128→    push(chunk) {
   129→   129→      buffer += chunk.toString();
   130→   130→      const lines = buffer.split(/\r?\n/);
   131→   131→      buffer = lines.pop() || "";
   132→   132→      for (const line of lines) {
   133→   133→        const parsed = parseChatSseLine(line);
   134→   134→        if (parsed) applyChatChunk(state, parsed);
   135→   135→      }
   136→   136→    },
   137→   137→    finish,
   138→   138→  };
   139→   139→}

   149→   149→function trimResponseHistories() {
   150→   150→  const cutoff = Date.now() - RESPONSE_HISTORY_TTL_MS;
   151→   151→  for (const [id, entry] of responseHistories) {
   152→   152→    if (entry.createdAt < cutoff) responseHistories.delete(id);
   153→   153→  }
   154→   154→  while (responseHistories.size > RESPONSE_HISTORY_MAX) {
   155→   155→    const oldest = responseHistories.keys().next().value;
   156→   156→    if (!oldest) break;
   157→   157→    responseHistories.delete(oldest);
   158→   158→  }
   159→   159→}

   161→   161→function responseToChatMessage(response) {
   162→   162→  const text = response.output
   163→   163→    ?.filter((item) => item.type === "message")
   164→   164→    .flatMap((item) => item.content || [])
   165→   165→    .filter((part) => part.type === "output_text")
   166→   166→    .map((part) => part.text || "")
   167→   167→    .join("") || "";
   168→   168→  const toolCalls = (response.output || [])
   169→   169→    .filter((item) => item.type === "function_call")
   170→   170→    .map((item) => ({
   171→   171→      id: item.call_id,
   172→   172→      type: "function",
   173→   173→      function: { name: item.name, arguments: item.arguments || "{}" },
   174→   174→    }));
   175→   175→  return {
   176→   176→    role: "assistant",
   177→   177→    content: text,
   178→   178→    ...(toolCalls.length ? { tool_calls: toolCalls } : {}),
   179→   179→  };
   180→   180→}

   182→   182→function rememberResponse(response, messages, client) {
   183→   183→  if (!response?.id) return;
   184→   184→  responseHistories.set(response.id, {
   185→   185→    client,
   186→   186→    createdAt: Date.now(),
   187→   187→    messages: [...messages, responseToChatMessage(response)],
   188→   188→  });
   189→   189→  trimResponseHistories();
   190→   190→}

   213→   213→function serveStaticDashboard(req, res, pathname) {
   214→   214→  if (req.method !== "GET" && req.method !== "HEAD") return false;
   215→   215→  if (pathname === "/api" || pathname.startsWith("/api/") || pathname.startsWith("/v1/") || pathname === "/healthz") return false;
   216→   216→
   217→   217→  let relativePath;
   218→   218→  try {
   219→   219→    relativePath = pathname === "/" ? "index.html" : decodeURIComponent(pathname).replace(/^[/\\]+/, "");
   220→   220→  } catch {
   221→   221→    sendError(res, "Invalid path", "invalid_request", 400);
   222→   222→    return true;
   223→   223→  }
   224→   224→  const hasExtension = path.extname(relativePath) !== "";
   225→   225→  const filePath = path.resolve(PUBLIC_DIR, relativePath);
   226→   226→  const publicPrefix = `${path.resolve(PUBLIC_DIR)}${path.sep}`;
   227→   227→  if (filePath !== path.resolve(PUBLIC_DIR) && !filePath.startsWith(publicPrefix)) {
   228→   228→    sendError(res, "Invalid path", "invalid_request", 400);
   229→   229→    return true;
   230→   230→  }
   231→   231→
   232→   232→  let selectedPath = filePath;
   233→   233→  try {
   234→   234→    if (!fs.statSync(selectedPath).isFile()) throw new Error("not a file");
   235→   235→  } catch {
   236→   236→    if (hasExtension) return false;
   237→   237→    selectedPath = path.join(PUBLIC_DIR, "index.html");
   238→   238→    try {
   239→   239→      if (!fs.statSync(selectedPath).isFile()) return false;
   240→   240→    } catch {
   241→   241→      return false;
   242→   242→    }
   243→   243→  }
   244→   244→
   245→   245→  try {
   246→   246→    const content = fs.readFileSync(selectedPath);
   247→   247→    const extension = path.extname(selectedPath).toLowerCase();
   248→   248→    res.writeHead(200, {
   249→   249→      "Content-Type": MIME_TYPES[extension] || "application/octet-stream",
   250→   250→      "Content-Length": content.length,
   251→   251→      "Cache-Control": extension === ".html" ? "no-cache" : "public, max-age=3600",
   252→   252→    });
   253→   253→    if (req.method !== "HEAD") res.end(content); else res.end();
   254→   254→  } catch {
   255→   255→    sendError(res, "Unable to load dashboard", "api_error", 500);
   256→   256→  }
   257→   257→  return true;
   258→   258→}

   381→   381→function parseJsonc(source) {
   382→   382→  let stripped = "";
   383→   383→  let inString = false;
   384→   384→  let escaped = false;
   385→   385→  let lineComment = false;
   386→   386→  let blockComment = false;
   387→   387→  for (let index = 0; index < source.length; index += 1) {
   388→   388→    const char = source[index];
   389→   389→    const next = source[index + 1];
   390→   390→    if (lineComment) {
   391→   391→      if (char === "\n") { lineComment = false; stripped += char; }
   392→   392→      continue;
   393→   393→    }
   394→   394→    if (blockComment) {
   395→   395→      if (char === "*" && next === "/") { blockComment = false; index += 1; }
   396→   396→      else if (char === "\n") stripped += "\n";
   397→   397→      continue;
   398→   398→    }
   399→   399→    if (!inString && char === "/" && next === "/") { lineComment = true; index += 1; continue; }
   400→   400→    if (!inString && char === "/" && next === "*") { blockComment = true; index += 1; continue; }
   401→   401→    stripped += char;
   402→   402→    if (inString) {
   403→   403→      if (escaped) escaped = false;
   404→   404→      else if (char === "\\") escaped = true;
   405→   405→      else if (char === '"') inString = false;
   406→   406→    } else if (char === '"') inString = true;
   407→   407→  }
   408→   408→
   409→   409→  let normalized = "";
   410→   410→  inString = false;
   411→   411→  escaped = false;
   412→   412→  for (let index = 0; index < stripped.length; index += 1) {
   413→   413→    const char = stripped[index];
   414→   414→    if (inString) {
   415→   415→      normalized += char;
   416→   416→      if (escaped) escaped = false;
   417→   417→      else if (char === "\\") escaped = true;
   418→   418→      else if (char === '"') inString = false;
   419→   419→      continue;
   420→   420→    }
   421→   421→    if (char === '"') { inString = true; normalized += char; continue; }
   422→   422→    if (char === ",") {
   423→   423→      let lookahead = index + 1;
   424→   424→      while (/\s/.test(stripped[lookahead] || "")) lookahead += 1;
   425→   425→      if (stripped[lookahead] === "}" || stripped[lookahead] === "]") continue;
   426→   426→    }
   427→   427→    normalized += char;
   428→   428→  }
   429→   429→  return normalized.trim() ? JSON.parse(normalized) : {};

   591→   591→function updateRootTomlValues(source, values) {
   592→   592→  const newline = source.includes("\r\n") ? "\r\n" : "\n";
   593→   593→  const lines = source ? source.split(/\r?\n/) : [];
   594→   594→  const firstSection = lines.findIndex((line) => /^\s*\[/.test(line));
   595→   595→  const rootEnd = firstSection === -1 ? lines.length : firstSection;
   596→   596→  const seen = new Set();
   597→   597→  const keys = Object.keys(values);
   598→   598→  for (let index = 0; index < rootEnd; index += 1) {
   599→   599→    const match = lines[index].match(/^\s*([A-Za-z0-9_-]+)\s*=/);
   600→   600→    if (!match || !Object.hasOwn(values, match[1])) continue;
   601→   601→    lines[index] = `${match[1]} = ${tomlValue(values[match[1]])}`;
   602→   602→    seen.add(match[1]);
   603→   603→  }
   604→   604→  const missing = keys.filter((key) => !seen.has(key)).map((key) => `${key} = ${tomlValue(values[key])}`);
   605→   605→  if (missing.length) lines.splice(rootEnd, 0, ...missing, ...(rootEnd ? [] : [""]));
   606→   606→  return lines.join(newline).replace(/\s*$/, "") + newline;
   607→   607→}

   609→   609→function updateTomlSection(source, header, values, removeKeys = []) {
   610→   610→  const newline = source.includes("\r\n") ? "\r\n" : "\n";
   611→   611→  const lines = source.split(/\r?\n/);
   612→   612→  const start = lines.findIndex((line) => line.trim() === header);
   613→   613→  const canonicalLines = Object.entries(values).map(([key, value]) => `${key} = ${tomlValue(value)}`);
   614→   614→
   615→   615→  if (start === -1) {
   616→   616→    const output = source.replace(/\s*$/, "");
   617→   617→    return `${output}${output ? `${newline}${newline}` : ""}${header}${newline}${canonicalLines.join(newline)}${newline}`;
   618→   618→  }
   619→   619→
   620→   620→  let end = start + 1;
   621→   621→  while (end < lines.length && !/^\s*\[/.test(lines[end])) end += 1;
   622→   622→  const canonicalKeys = new Set([...Object.keys(values), ...removeKeys]);
   623→   623→  const preserved = lines.slice(start + 1, end).filter((line) => {
   624→   624→    const match = line.match(/^\s*([A-Za-z0-9_-]+)\s*=/);
   625→   625→    return !match || !canonicalKeys.has(match[1]);
   626→   626→  });
   627→   627→  lines.splice(start + 1, end - start - 1, ...canonicalLines, ...preserved);
   628→   628→  return lines.join(newline).replace(/\s*$/, "") + newline;
   629→   629→}

   631→   631→function rootTomlValue(source, key) {
   632→   632→  const root = source.split(/\r?\n/).slice(0, (() => {
   633→   633→    const index = source.split(/\r?\n/).findIndex((line) => /^\s*\[/.test(line));
   634→   634→    return index === -1 ? source.split(/\r?\n/).length : index;
   635→   635→  })()).join("\n");
   636→   636→  const match = root.match(new RegExp(`^\\s*${key}\\s*=\\s*"([^"]*)"`, "m"));
   637→   637→  return match?.[1] || null;
   638→   638→}
   639→   639→
   640→   640→function tomlSection(source, header) {
   641→   641→  const lines = source.split(/\r?\n/);
   642→   642→  const start = lines.findIndex((line) => line.trim() === header);
   643→   643→  if (start === -1) return "";
   644→   644→  let end = start + 1;
   645→   645→  while (end < lines.length && !/^\s*\[/.test(lines[end])) end += 1;
   646→   646→  return lines.slice(start + 1, end).join("\n");
   647→   647→}

   738→   738→async function handleDashboardApi(req, res, pathname) {
   739→   739→  if (req.method === "GET" && pathname === "/api/dashboard/overview") {
   740→   740→    return sendJSON(res, dashboardOverview());

   742→   742→  if (req.method === "GET" && pathname === "/api/dashboard/requests") {
   743→   743→    const rawLimit = new URL(req.url, "http://localhost").searchParams.get("limit");
   744→   744→    const limit = Math.min(Math.max(parseInt(rawLimit || "10", 10) || 10, 1), 100);
   745→   745→    const requests = telemetry.recent(2_000).slice(-limit).reverse();
   746→   746→    return sendJSON(res, { requests });
   747→   747→  }
   748→   748→  if (req.method === "POST" && pathname === "/api/dashboard/codex/setup") {
   749→   749→    if (!isLoopbackRequest(req)) return sendDashboardError(res, "Configurar o Codex só é permitido a partir deste computador.", 403);
   750→   750→    let body;
   751→   751→    try { body = await readBody(req, config.MAX_BODY_BYTES); }
   752→   752→    catch (error) { return sendDashboardError(res, error.message, error.statusCode || 400); }

   754→   754→    if (!MODELS.some((model) => model.id === requestedModel)) return sendDashboardError(res, "Modelo inválido para o Codex.", 400);
   755→   755→    try {
   756→   756→      const codex = configureCodexDesktop(requestedModel);
   757→   757→      return sendJSON(res, {
   758→   758→        message: "Configuração salva. O proxy não fecha nem reinicia o Codex; o modelo GLM será carregado somente na próxima abertura manual do aplicativo.",
   759→   759→        codex,
   760→   760→        overview: dashboardOverview(),
   761→   761→      });
   762→   762→    } catch (error) {
   763→   763→      return sendDashboardError(res, `Não foi possível configurar o Codex: ${error.message}`, 500);
   764→   764→    }
   765→   765→  }
   766→   766→  const toolSetupMatch = pathname.match(/^\/api\/dashboard\/(opencode|claude)\/setup$/);
   767→   767→  if (req.method === "POST" && toolSetupMatch) {
   768→   768→    if (!isLoopbackRequest(req)) return sendDashboardError(res, "Configurar ferramentas só é permitido a partir deste computador.", 403);
   769→   769→    let body;
   770→   770→    try { body = await readBody(req, config.MAX_BODY_BYTES); }
   771→   771→    catch (error) { return sendDashboardError(res, error.message, error.statusCode || 400); }
   772→   772→    const requestedModel = typeof body.model === "string" ? body.model : MODELS[0]?.id;
   773→   773→    if (!MODELS.some((model) => model.id === requestedModel)) return sendDashboardError(res, "Modelo inválido para a integração.", 400);
   774→   774→    try {
   775→   775→      const tool = toolSetupMatch[1];
   776→   776→      const status = tool === "opencode"
   777→   777→        ? configureOpenCode(requestedModel)
   778→   778→        : configureClaudeCode(requestedModel);
   779→   779→      const displayName = tool === "opencode" ? "OpenCode" : "Claude Code";
   780→   780→      return sendJSON(res, {
   781→   781→        message: status.available
   782→   782→          ? `${displayName} preparado com ${requestedModel}. A chave será passada somente ao processo aberto pelo painel.`
   783→   783→          : `${displayName} foi configurado com ${requestedModel}, mas ainda precisa ser instalado para abrir.`,
   784→   784→        [tool]: status,
   785→   785→        overview: dashboardOverview(),
   786→   786→      });
   787→   787→    } catch (error) {
   788→   788→      return sendDashboardError(res, `Não foi possível preparar a integração: ${error.message}`, 500);
   789→   789→    }
   790→   790→  }
   791→   791→  if (req.method === "POST" && pathname === "/api/dashboard/openrouter/setup") {
   792→   792→    if (!isLoopbackRequest(req)) return sendDashboardError(res, "Configurar o OpenRouter só é permitido a partir deste computador.", 403);
   793→   793→    let body;
   794→   794→    try { body = await readBody(req, config.MAX_BODY_BYTES); }
   795→   795→    catch (error) { return sendDashboardError(res, error.message, error.statusCode || 400); }
   796→   796→    try {
   797→   797→      const fallback = openRouter.save({ apiKey: body.apiKey, model: body.model });
   798→   798→      return sendJSON(res, {
   799→   799→        message: `OpenRouter configurado. Falhas do AutoClaw usarão ${fallback.model} automaticamente.`,
   800→   800→        fallback,
   801→   801→        overview: dashboardOverview(),
   802→   802→      });
   803→   803→    } catch (error) {
   804→   804→      return sendDashboardError(res, error.message, 400);
   805→   805→    }

   820→   820→// SSE buffering (OpenAI-specific: assemble streamed chunks into a single response)
   821→   821→function bufferSSE(upstreamRes, modelId) {
   822→   822→  return new Promise((resolve, reject) => {
   823→   823→    let raw = "";
   824→   824→    upstreamRes.on("data",  (c) => (raw += c));
   825→   825→    upstreamRes.on("error", reject);
   826→   826→    upstreamRes.on("end",   () => {
   827→   827→      try {
   828→   828→        let content = "", reasoning = "";
   829→   829→        let id      = `chatcmpl-${generateId()}`;
   830→   830→        let model   = modelId;
   831→   831→        let promptTokens = 0, completionTokens = 0;
   832→   832→        let finishReason = "stop";
   833→   833→        const toolCalls = {};
   834→   834→
   835→   835→        for (const line of raw.split("\n")) {
   836→   836→          if (!line.startsWith("data: ") || line === "data: [DONE]") continue;
   837→   837→          const chunk = JSON.parse(line.slice(6));
   838→   838→          if (chunk.id)    id    = chunk.id;
   839→   839→          if (chunk.model) model = chunk.model;
   840→   840→          const delta = chunk.choices?.[0]?.delta;
   841→   841→          if (delta?.content)          content   += delta.content;
   842→   842→          if (delta?.reasoning_content) reasoning += delta.reasoning_content;
   843→   843→          // Accumulate tool calls
   844→   844→          for (const tc of delta?.tool_calls || []) {
   845→   845→            if (!toolCalls[tc.index]) toolCalls[tc.index] = { id: "", name: "", arguments: "" };
   846→   846→            if (tc.id)                  toolCalls[tc.index].id = tc.id;
   847→   847→            if (tc.function?.name)      toolCalls[tc.index].name = tc.function.name;
   848→   848→            if (tc.function?.arguments) toolCalls[tc.index].arguments += tc.function.arguments;
   849→   849→          }
   850→   850→          const fr = chunk.choices?.[0]?.finish_reason;
   851→   851→          if (fr) finishReason = fr;
   852→   852→          if (chunk.usage) {
   853→   853→            promptTokens     = chunk.usage.prompt_tokens     ?? 0;
   854→   854→            completionTokens = chunk.usage.completion_tokens ?? 0;
   855→   855→          }
   856→   856→        }
   857→   857→
   858→   858→        // Build sorted tool_calls array
   859→   859→        const sortedToolCalls = Object.keys(toolCalls)
   860→   860→          .sort((a, b) => Number(a) - Number(b))
   861→   861→          .map((idx) => {
   862→   862→            const tc = toolCalls[idx];
   863→   863→            return {
   864→   864→              id: tc.id,
   865→   865→              type: "function",
   866→   866→              function: { name: tc.name, arguments: tc.arguments },
   867→   867→            };
   868→   868→          });
   869→   869→
   870→   870→        resolve({
   871→   871→          id,
   872→   872→          object:  "chat.completion",
   873→   873→          created: Math.floor(Date.now() / 1000),
   874→   874→          model,
   875→   875→          choices: [{
   876→   876→            index:         0,
   877→   877→            message:       {
   878→   878→              role:    "assistant",
   879→   879→              content,
   880→   880→              ...(reasoning ? { reasoning_content: reasoning } : {}),
   881→   881→              ...(sortedToolCalls.length ? { tool_calls: sortedToolCalls } : {}),
   882→   882→            },
   883→   883→            finish_reason: finishReason,
   884→   884→          }],
   885→   885→          usage: {
   886→   886→            prompt_tokens:     promptTokens,
   887→   887→            completion_tokens: completionTokens,
   888→   888→            total_tokens:      promptTokens + completionTokens,
   889→   889→          },
   890→   890→        });
   891→   891→      } catch (err) {
   892→   892→        reject(new Error(`Failed to parse upstream SSE: ${err.message}`));
   893→   893→      }
   894→   894→    });
   895→   895→  });
   896→   896→}

  1204→  1204→  if (!isSafeModelId(body.model)) {
