   466→   466→#  Main — process single account
   467→   467→# ══════════════════════════════════════════════════════════════════════
   468→   468→async def process_account(email, password, test_only=False):
   469→   469→    tag = "[TEST] " if test_only else ""
   470→   470→    # Check proxy for this account
   471→   471→    from auth import _next_proxy
   472→   472→    _px = _next_proxy()
   473→   473→    _px_host = _px["server"].split("//")[-1] if _px else "direct"
   474→   474→    log(f"{tag}Processing: {email}" + (f" via {_px_host}" if _px else ""))
   475→   475→
   476→   476→    # Check if already in tokens.json
   477→   477→    tokens = load_tokens()
   478→   478→    existing = [a for a in tokens.get("accounts", []) if a.get("email") == email]
   479→   479→    if existing and not test_only:
   480→   480→        log(f"Already have token for {email}, skipping. Use --force to re-login.", "OK")
   481→   481→        return {"success": True, "email": email, "skipped": True}
   482→   482→
   483→   483→    # 1. Get OAuth URL (uses rotating proxy from config)
   484→   484→    oauth_url, state, device_id = get_oauth_url()
   485→   485→    if not oauth_url:
   486→   486→        return {"success": False, "email": email, "error": "oauth_url_failed"}
   487→   487→
   488→   488→    dbg(f"state={state}, device_id={device_id}")
   489→   489→
   490→   490→    # 2. Browser automation — Google login
   491→   491→    result = await automate_google_login(oauth_url, email, password)
   492→   492→
   493→   493→    if result.get("error") and not result.get("login_done"):
   494→   494→        log(f"Login failed: {result['error']}", "ERR")
   495→   495→        return {"success": False, "email": email, "error": result["error"]}
   496→   496→
   497→   497→    # 3. Check if proxy captured the callback (use state for concurrent tracking)
   498→   498→    log("Checking if token was captured...", "WAIT")
   499→   499→    account = check_login_result(state)
   500→   500→
   501→   501→    if account:
   502→   502→        log(f"Login success! Email: {account.get('email')}, User ID: {account.get('user_id')}", "OK")
   503→   503→        return {
   504→   504→            "success": True,
   505→   505→            "email": account.get("email", email),
   506→   506→            "user_id": account.get("user_id"),
   507→   507→            "first_login": account.get("first_login"),
   508→   508→        }
   509→   509→
   510→   510→    # Fallback: check tokens.json directly
   511→   511→    await asyncio.sleep(2)
   512→   512→    tokens = load_tokens()
   513→   513→    new_acc = [a for a in tokens.get("accounts", []) if a.get("email") == email]
   514→   514→    if new_acc:
   515→   515→        log(f"Token found in tokens.json for {email}", "OK")
   516→   516→        return {"success": True, "email": email, "user_id": new_acc[0].get("user_id")}
   517→   517→
   518→   518→    log("Token not captured! Check proxy logs.", "ERR")
   519→   519→    return {"success": False, "email": email, "error": "token_not_captured"}
   520→   520→

   522→   522→# ══════════════════════════════════════════════════════════════════════
   523→   523→#  Concurrent batch runner
   524→   524→# ══════════════════════════════════════════════════════════════════════
   525→   525→async def run_batch(accounts, test_only=False, concurrent=1):
   526→   526→    """Run multiple accounts concurrently with semaphore."""
   527→   527→    sem = asyncio.Semaphore(concurrent)
   528→   528→
   529→   529→    async def _run(email, password):
   530→   530→        async with sem:
   531→   531→            return await process_account(email, password, test_only=test_only)
   532→   532→
   533→   533→    tasks = []
   534→   534→    for acc in accounts:
   535→   535→        email, pwd = acc.split(":", 1)
   536→   536→        tasks.append(_run(email, pwd))
   537→   537→
   538→   538→    results = await asyncio.gather(*tasks, return_exceptions=True)
   539→   539→
   540→   540→    processed = []
   541→   541→    for i, r in enumerate(results):
   542→   542→        if isinstance(r, Exception):
   543→   543→            email = accounts[i].split(":", 1)[0]
   544→   544→            processed.append({"success": False, "email": email, "error": str(r)})
   545→   545→        else:
   546→   546→            processed.append(r)
   547→   547→
   548→   548→    return processed
