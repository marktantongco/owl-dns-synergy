   176→def make_table(data, col_widths=None, header_rows=1):
   177→    """Create a styled table with palette colors."""
   178→    if col_widths is None:
   179→        col_widths = [CONTENT_W / len(data[0])] * len(data[0])
   180→    
   181→    t = Table(data, colWidths=col_widths, repeatRows=header_rows)
   182→    
   183→    style_cmds = [
   184→        ('BACKGROUND', (0, 0), (-1, header_rows-1), HEADER_FILL),
   185→        ('TEXTCOLOR', (0, 0), (-1, header_rows-1), colors.white),
   186→        ('FONTNAME', (0, 0), (-1, header_rows-1), 'Carlito-Bold'),
   187→        ('FONTSIZE', (0, 0), (-1, header_rows-1), 9),
   188→        ('FONTNAME', (0, header_rows), (-1, -1), 'Carlito'),
   189→        ('FONTSIZE', (0, header_rows), (-1, -1), 9),
   190→        ('LEADING', (0, 0), (-1, -1), 12),
   191→        ('ALIGN', (0, 0), (-1, -1), 'LEFT'),
   192→        ('VALIGN', (0, 0), (-1, -1), 'TOP'),
   193→        ('GRID', (0, 0), (-1, -1), 0.5, BORDER),
   194→        ('TOPPADDING', (0, 0), (-1, -1), 4),
   195→        ('BOTTOMPADDING', (0, 0), (-1, -1), 4),
   196→        ('LEFTPADDING', (0, 0), (-1, -1), 4),
   197→        ('RIGHTPADDING', (0, 0), (-1, -1), 4),
   198→    ]
   199→    
   200→    # Alternating row colors
   201→    for i in range(header_rows, len(data)):
   202→        if i % 2 == 0:
   203→            style_cmds.append(('BACKGROUND', (0, i), (-1, i), TABLE_STRIPE))
   204→        else:
   205→            style_cmds.append(('BACKGROUND', (0, i), (-1, i), PAGE_BG))
   206→    
   207→    t.setStyle(TableStyle(style_cmds))
   208→    return t

   433→    'the same SHA-256 hash used by the cache. When two requests arrive simultaneously for the same resource, '
   434→    'the second request awaits the first request\'s future rather than issuing a duplicate network call. This '
   435→    'is critical for high-concurrency scraping scenarios where multiple threads or agents may target the same '
   436→    'URL simultaneously. The deduplicator uses asyncio.Lock for thread-safe registration and cleanup of futures, '
   437→    'with proper exception propagation to all waiting consumers.'
   438→))

   440→story.append(heading3('3.2.3 QualityScorer'))
   441→story.append(body(
   442→    'The QualityScorer implements an exponential moving average (EMA) with configurable decay factor (default '
   443→    '0.9) to track proxy quality over time. Each proxy receives a score between 0.0 (completely unreliable) '
   444→    'and 1.0 (perfect reliability), updated on every request outcome. The EMA formula is: new_score = '
   445→    'old_score * decay + (success ? 1.0 : 0.0) * (1 - decay). This ensures recent failures weigh heavily '
   446→    'while historical performance provides stability. The scorer also maintains a latency history (last 100 '
   447→    'measurements per proxy) for latency-aware proxy selection. The get_best_proxy() method selects the proxy '

   455→    'rate-limit responses (429, 503), the rate decreases by 50% toward a configurable minimum (default 0.1 '
