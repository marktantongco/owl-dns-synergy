   930→   930→   930→    async def _try_http_proxy(self, url: str, domain: str, **kwargs) -> ChannelResult:
   931→   931→   931→        """Route through proxy pool (prox5/https_proxy)."""
   932→   932→   932→        import httpx
   933→   933→   933→        start = time.time()
   934→   934→   934→        proxy = self.proxy_pool.get_next()
   935→   935→   935→
   936→   936→   936→        if not proxy:
   937→   937→   937→            return ChannelResult("http_proxy", False, error="No proxies available")
   938→   938→   938→
   939→   939→   939→        try:
   940→   940→   940→            api_key = self.key_rotator.get_active_key()
   941→   941→   941→            headers = {}
   942→   942→   942→            if api_key:
   943→   943→   943→                headers["Authorization"] = f"Bearer {api_key}"
   944→   944→   944→
   945→   945→   945→            async with httpx.AsyncClient(
   946→   946→   946→                proxy=proxy.url, timeout=30.0
   947→   947→   947→            ) as client:
   948→   948→   948→                resp = await client.get(url, headers=headers)
   949→   949→   949→                latency = (time.time() - start) * 1000
   950→   950→   950→
   951→   951→   951→                if resp.status_code in (401, 403, 429):
   952→   952→   952→                    self.key_rotator.report_error(
   953→   953→   953→                        resp.status_code,
   954→   954→   954→                        "rate_limit" if resp.status_code == 429 else "auth_error"
   955→   955→   955→                    )
   956→   956→   956→
   957→   957→   957→                self.proxy_pool.report_success(proxy.endpoint, latency)
   958→   958→   958→                self._cache_set(url, resp.content)
   959→   959→   959→
   960→   960→   960→                return ChannelResult(
   961→   961→   961→                    "http_proxy", True,
   962→   962→   962→                    data=resp.content, latency_ms=latency,
   963→   963→   963→                    status_code=resp.status_code, proxy_used=proxy.endpoint
   964→   964→   964→                )
   965→   965→   965→        except Exception as e:
   966→   966→   966→            latency = (time.time() - start) * 1000
   967→   967→   967→            self.proxy_pool.report_failure(proxy.endpoint)
   968→   968→   968→            return ChannelResult("http_proxy", False, error=str(e), latency_ms=latency)

  1045→  1045→  1045→    async def fetch(self, url: str, force_channel: str = None, **kwargs) -> ChannelResult:
  1046→  1046→  1046→        """
  1047→  1047→  1047→        Fetch a URL through the optimal channel with automatic failover.
  1048→  1048→  1048→
  1049→  1049→  1049→        Channel selection priority:
  1050→  1050→  1050→          cached → http_proxy → socks_pool → dns_tunnel → mitm_stealth → connect_chain → http_direct
  1051→  1051→  1051→        """
  1052→  1052→  1052→        domain = self._extract_domain(url)
  1053→  1053→  1053→
  1054→  1054→  1054→        # Check cache
  1055→  1055→  1055→        cached = self._cache_get(url)
  1056→  1056→  1056→        if cached and not force_channel:
  1057→  1057→  1057→            return ChannelResult("cached", True, data=cached)
  1058→  1058→  1058→
  1059→  1059→  1059→        # Select channel
  1060→  1060→  1060→        channel = self._select_channel(domain, force_channel)
  1061→  1061→  1061→
  1062→  1062→  1062→        # Channel implementations
  1063→  1063→  1063→        channel_map = {
  1064→  1064→  1064→            Channel.HTTP_PROXY: self._try_http_proxy,
  1065→  1065→  1065→            Channel.SOCKS_POOL: self._try_socks_pool,
  1066→  1066→  1066→            Channel.DNS_TUNNEL: self._try_dns_tunnel,
  1067→  1067→  1067→            Channel.MITM_STEALTH: self._try_mitm_stealth,
  1068→  1068→  1068→            Channel.CONNECT_CHAIN: self._try_connect_chain,
  1069→  1069→  1069→            Channel.HTTP_DIRECT: self._try_http_direct,
  1070→  1070→  1070→        }
  1071→  1071→  1071→
  1072→  1072→  1072→        # Try preferred channel
  1073→  1073→  1073→        if channel in channel_map:
  1074→  1074→  1074→            result = await channel_map[channel](url, domain, **kwargs)
  1075→  1075→  1075→            if result.success:
  1076→  1076→  1076→                REQUESTS_TOTAL.labels(channel=result.channel, domain=domain, status="success").inc()
  1077→  1077→  1077→                REQUESTS_DURATION.labels(channel=result.channel, domain=domain).observe(result.latency_ms / 1000)
  1078→  1078→  1078→                return result
  1079→  1079→  1079→
  1080→  1080→  1080→        # Fallback: try all channels in priority order
  1081→  1081→  1081→        fallback_order = [
  1082→  1082→  1082→            Channel.HTTP_PROXY, Channel.SOCKS_POOL, Channel.DNS_TUNNEL,
  1083→  1083→  1083→            Channel.MITM_STEALTH, Channel.HTTP_DIRECT,
  1084→  1084→  1084→        ]
  1085→  1085→  1085→        for ch in fallback_order:
  1086→  1086→  1086→            if ch == channel:
  1087→  1087→  1087→                continue  # Already tried
  1088→  1088→  1088→            if ch in channel_map:
  1089→  1089→  1089→                result = await channel_map[ch](url, domain, **kwargs)
  1090→  1090→  1090→                if result.success:
  1091→  1091→  1091→                    CHANNEL_SWITCHES.labels(
  1092→  1092→  1092→                        from_channel=channel.value, to_channel=ch.value
  1093→  1093→  1093→                    ).inc()
  1094→  1094→  1094→                    REQUESTS_TOTAL.labels(channel=result.channel, domain=domain, status="success").inc()
  1095→  1095→  1095→                    return result
  1096→  1096→  1096→
  1097→  1097→  1097→        REQUESTS_TOTAL.labels(channel="none", domain=domain, status="all_failed").inc()
  1098→  1098→  1098→        return ChannelResult("none", False, error=f"All channels exhausted for {domain}")
