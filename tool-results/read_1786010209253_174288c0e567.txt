   311→   311→// ── Fetch all wallets in background with limited concurrency ──
   312→   312→let _walletFetchInFlight = false;
   313→   313→let _walletFetchAbort = false;
   314→   314→
   315→   315→async function fetchAllWalletsBackground() {
   316→   316→  // Prevent overlapping fetches
   317→   317→  if (_walletFetchInFlight) {
   318→   318→    _walletFetchAbort = true;
   319→   319→    // Wait a bit for previous to notice abort
   320→   320→    await new Promise(r => setTimeout(r, 200));
   321→   321→  }
   322→   322→  _walletFetchInFlight = true;
   323→   323→  _walletFetchAbort = false;
   324→   324→
   325→   325→  const toFetch = accountsData.filter(acc => {
   326→   326→    const w = walletCache[acc.email];
   327→   327→    return !w || w.balance == null;
   328→   328→  });
   329→   329→
   330→   330→  if (toFetch.length === 0) {
   331→   331→    _walletFetchInFlight = false;
   332→   332→    return;
   333→   333→  }
   334→   334→
   335→   335→  // Process in small batches to avoid overwhelming the server
   336→   336→  const BATCH = 5;
   337→   337→  for (let i = 0; i < toFetch.length; i += BATCH) {
   338→   338→    if (_walletFetchAbort) break;
   339→   339→
   340→   340→    const batch = toFetch.slice(i, i + BATCH);
   341→   341→    const promises = batch.map(async (acc) => {
   342→   342→      if (_walletFetchAbort) return;
   343→   343→      try {
   344→   344→        const resp = await fetch(API + '/api/wallet/' + encodeURIComponent(acc.email));
   345→   345→        const data = await resp.json();
   346→   346→        if (data.code === 0 && data.data) {
   347→   347→          walletCache[acc.email] = { email: acc.email, balance: data.data.total_balance, status: data.data.total_balance > 0 ? 'active' : 'exhausted' };
   348→   348→        } else {
   349→   349→          walletCache[acc.email] = { email: acc.email, balance: 0, status: 'error' };
   350→   350→        }
   351→   351→      } catch (e) {
   352→   352→        walletCache[acc.email] = { email: acc.email, balance: 0, status: 'error' };
   353→   353→      }
   354→   354→      // Update totals after each wallet completes — credits bar counts up live
   355→   355→      updateTotals();
   356→   356→    });
   357→   357→
   358→   358→    await Promise.all(promises);
   359→   359→  }
   360→   360→
   361→   361→  _walletFetchInFlight = false;
   362→   362→}

   364→   364→// ── Render single page of cards ──
   365→   365→function renderPage() {
   366→   366→  const grid = document.getElementById('cards-grid');
   367→   367→  const total = accountsData.length;
   368→   368→  const totalPages = Math.ceil(total / perPage);
   369→   369→  if (currentPage > totalPages) currentPage = totalPages;
   370→   370→  if (currentPage < 1) currentPage = 1;
   371→   371→
   372→   372→  const start = (currentPage - 1) * perPage;
   373→   373→  const end = Math.min(start + perPage, total);
   374→   374→  const pageAccounts = accountsData.slice(start, end);
   375→   375→
   376→   376→  // Render cards for this page
   377→   377→  grid.innerHTML = '';
   378→   378→  for (let i = 0; i < pageAccounts.length; i++) {
   379→   379→    const acc = pageAccounts[i];
   380→   380→    const cardId = 'card-' + acc.email.replace(/[^a-zA-Z0-9]/g, '');
   381→   381→    const emailShort = acc.email.length > 32 ? acc.email.substring(0, 29) + '...' : acc.email;
   382→   382→
   383→   383→    // Token status
   384→   384→    let tokenBadge, tokenDot, tokenBadgeClass;
   385→   385→    if (acc.expired) {
   386→   386→      tokenBadge = 'disabled'; tokenDot = 'dot-red'; tokenBadgeClass = 'badge-disabled';
   387→   387→    } else if (acc.remaining_hours < 2) {
   388→   388→      tokenBadge = 'expiring'; tokenDot = 'dot-yellow'; tokenBadgeClass = 'badge-expiring';
   389→   389→    } else {
   390→   390→      tokenBadge = 'active'; tokenDot = 'dot-green'; tokenBadgeClass = 'badge-active';
   391→   391→    }
   392→   392→
   393→   393→    const reqCount = (acc.request_count || 0) + ' req';
   394→   394→
   395→   395→    // Check if wallet already cached

   502→   502→      let fillClass = 'fill-blue';
   503→   503→      if (isExhausted) fillClass = 'fill-red';

   529→   529→function updateTotals() {
   530→   530→  let totalPoints = 0, totalMax = 0, loaded = 0;
   531→   531→  const target = accountsData.length;
   532→   532→  for (const acc of accountsData) {
   533→   533→    const w = walletCache[acc.email];
   534→   534→    if (w && w.balance != null) {
   535→   535→      totalPoints += w.balance;
   536→   536→      totalMax += 2300;
   537→   537→      loaded++;
   538→   538→    }
   539→   539→  }
   540→   540→  const bar = document.getElementById('credits-bar');
   541→   541→  if (totalMax > 0 || loaded > 0 || target > 0) {
   542→   542→    bar.style.display = 'inline-flex';
   543→   543→    // Animate the number — smooth count up
   544→   544→    const pointsEl = document.getElementById('total-points');
   545→   545→    const maxEl = document.getElementById('total-max');
   546→   546→    const progEl = document.getElementById('credits-progress');
   547→   547→    const current = parseInt(pointsEl.textContent.replace(/[^0-9]/g, '')) || 0;
   548→   548→    const target_val = totalPoints;
   549→   549→
   550→   550→    // Show loading progress
   551→   551→    if (loaded < target) {
   552→   552→      progEl.textContent = `(${loaded}/${target})`;
   553→   553→    } else {
   554→   554→      progEl.textContent = '';
   555→   555→    }
   556→   556→
   557→   557→    // Smooth count animation — only start if not already animating
   558→   558→    if (current !== target_val && pointsEl.dataset.animating !== '1') {
   559→   559→      pointsEl.dataset.animating = '1';
   560→   560→      requestAnimationFrame(function tick() {
   561→   561→        const cur = parseInt(pointsEl.textContent.replace(/[^0-9]/g, '')) || 0;
   562→   562→        const d = target_val - cur;
   563→   563→        if (d === 0) {
   564→   564→          pointsEl.textContent = target_val.toLocaleString();
   565→   565→          pointsEl.dataset.animating = '';
   566→   566→          return;
   567→   567→        }
   568→   568→        // Adaptive step: bigger jumps for large diffs, small for precision
   569→   569→        const s = Math.max(1, Math.ceil(Math.abs(d) / 15));
   570→   570→        if (Math.abs(d) <= s) {
   571→   571→          pointsEl.textContent = target_val.toLocaleString();
   572→   572→          pointsEl.dataset.animating = '';
   573→   573→        } else {
   574→   574→          pointsEl.textContent = (cur + Math.sign(d) * s).toLocaleString();
   575→   575→          requestAnimationFrame(tick);
   576→   576→        }
   577→   577→      });
   578→   578→    }
   579→   579→    maxEl.textContent = totalMax.toLocaleString();
   580→   580→    // Update dot color based on progress
   581→   581→    const dot = bar.querySelector('.credits-dot');
   582→   582→    if (dot) {
   583→   583→      const pct = totalMax > 0 ? (totalPoints / totalMax) * 100 : 0;
   584→   584→      dot.style.background = pct > 50 ? 'var(--green)' : pct > 20 ? 'var(--yellow)' : 'var(--red)';
   585→   585→    }
   586→   586→  } else {
   587→   587→    bar.style.display = 'none';
   588→   588→  }
   589→   589→}

   612→   612→async function refreshAllTokens() {
   613→   613→  // Disable button, show spinner
   614→   614→  const btn = event.target.closest('button');

   619→   619→    const data = await resp.json();
   620→   620→
   621→   621→    if (data.status === 'running') {
   622→   622→      // Already running — start polling
   623→   623→      showToast('Refresh already in progress...', 'success');

   625→   625→      showToast(`Refreshing ${data.total} tokens...`, 'success');
   626→   626→    }
   627→   627→
   628→   628→    // Start polling progress
   629→   629→    if (_refreshAllPolling) clearInterval(_refreshAllPolling);
   630→   630→    _refreshAllPolling = setInterval(async () => {
   631→   631→      try {
   632→   632→        const pr = await fetch(API + '/api/refresh-progress').then(r => r.json());

   634→   634→

   637→   637→          btn.innerHTML = btn.querySelector('svg').outerHTML + ` ${pr.done}/${pr.total}`;
   638→   638→        }
   639→   639→
   640→   640→        if (!pr.running) {
   641→   641→          clearInterval(_refreshAllPolling);
   642→   642→          _refreshAllPolling = null;
   643→   643→          const msg = `Refreshed: ${pr.success} ok, ${pr.fail} failed`;
   644→   644→          showToast(msg, pr.fail > 0 ? 'error' : 'success');

   646→   646→          if (btn) {
   647→   647→            btn.classList.remove('spinning');
   648→   648→            btn.disabled = false;

   650→   650→          }
   651→   651→          loadAccounts();
   652→   652→        }

   657→   657→
   658→   658→  } catch (e) {

   660→   660→    if (btn) {
   661→   661→      btn.classList.remove('spinning');
   662→   662→      btn.disabled = false;
