    92→async function runOpenAIUpstream(body, modelId, { preferOpenRouter = false } = {}) {
    93→  const fallback = openRouter.load();
    94→  if (preferOpenRouter && fallback.configured) {
    95→    const response = await callUpstreamOpenRouter(config, fallback, body, log);
    96→    return { response, errorBody: "", provider: "openrouter", reason: "responses_primary" };
    97→  }
    98→  try {
    99→    const autoclaw = await callAutoClawWithRetry({
   100→      primary: () => callUpstreamOpenAI(KNOWN_IDS, config.CLIENT_HEADERS, getToken, body, modelId, log, config.UPSTREAM_URL),
   101→      log,
   102→    });
   103→    if (autoclaw.response.statusCode < 400 || !fallback.configured) return autoclaw;
   104→    log.warn(`AutoClaw status=${autoclaw.response.statusCode}; switching to OpenRouter model=${fallback.model}`);
   105→  } catch (error) {
   106→    if (!fallback.configured) throw error;
   107→    log.warn(`AutoClaw unavailable (${error.message}); switching to OpenRouter model=${fallback.model}`);
   108→  }
   109→
   110→  const response = await callUpstreamOpenRouter(config, fallback, body, log);
   111→  return { response, errorBody: "", provider: "openrouter", reason: "autoclaw_failed" };
   112→}

   114→function createUsageTracker(onComplete) {
   115→  const state = { toolCalls: new Map(), text: "", reasoning: "", inputTokens: 0, outputTokens: 0 };
   116→  let buffer = "";
   117→  let settled = false;
   118→  const finish = () => {
   119→    if (settled) return;
   120→    settled = true;
   121→    onComplete({
   122→      prompt_tokens: state.inputTokens,
   123→      completion_tokens: state.outputTokens,
   124→      total_tokens: state.inputTokens + state.outputTokens,
   125→    });
   126→  };
   127→  return {
   128→    push(chunk) {
   129→      buffer += chunk.toString();
   130→      const lines = buffer.split(/\r?\n/);
   131→      buffer = lines.pop() || "";
   132→      for (const line of lines) {
   133→        const parsed = parseChatSseLine(line);
   134→        if (parsed) applyChatChunk(state, parsed);
   135→      }
   136→    },
   137→    finish,
   138→  };
   139→}

   141→// Responses clients (including Codex Desktop) can refer to the preceding
   142→// response instead of resending the whole conversation. The upstream is Chat
   143→// Completions, so retain a short, per-client history locally and expand it
   144→// before forwarding. It is intentionally memory-only and expires on restart.
   145→const responseHistories = new Map();
   146→const RESPONSE_HISTORY_MAX = 256;
   147→const RESPONSE_HISTORY_TTL_MS = 6 * 60 * 60 * 1000;
   148→
   149→function trimResponseHistories() {
   150→  const cutoff = Date.now() - RESPONSE_HISTORY_TTL_MS;
   151→  for (const [id, entry] of responseHistories) {
   152→    if (entry.createdAt < cutoff) responseHistories.delete(id);
   153→  }
   154→  while (responseHistories.size > RESPONSE_HISTORY_MAX) {
   155→    const oldest = responseHistories.keys().next().value;
   156→    if (!oldest) break;
   157→    responseHistories.delete(oldest);
   158→  }
   159→}

   161→function responseToChatMessage(response) {
   162→  const text = response.output
   163→    ?.filter((item) => item.type === "message")
   164→    .flatMap((item) => item.content || [])
   165→    .filter((part) => part.type === "output_text")
   166→    .map((part) => part.text || "")
   167→    .join("") || "";
   168→  const toolCalls = (response.output || [])
   169→    .filter((item) => item.type === "function_call")
   170→    .map((item) => ({
   171→      id: item.call_id,
   172→      type: "function",
   173→      function: { name: item.name, arguments: item.arguments || "{}" },
   174→    }));
   175→  return {
   176→    role: "assistant",
   177→    content: text,
   178→    ...(toolCalls.length ? { tool_calls: toolCalls } : {}),
   179→  };
   180→}

   182→function rememberResponse(response, messages, client) {
   183→  if (!response?.id) return;
   184→  responseHistories.set(response.id, {
   185→    client,
   186→    createdAt: Date.now(),
   187→    messages: [...messages, responseToChatMessage(response)],
   188→  });
   189→  trimResponseHistories();
   190→}

   192→function resolvePreviousResponse(responseId, client) {
   193→  if (!responseId) return [];
   194→  trimResponseHistories();
   195→  const entry = responseHistories.get(responseId);
   196→  if (!entry || entry.client !== client) return null;
   197→  return entry.messages;
   198→}

   213→function serveStaticDashboard(req, res, pathname) {
   214→  if (req.method !== "GET" && req.method !== "HEAD") return false;
   215→  if (pathname === "/api" || pathname.startsWith("/api/") || pathname.startsWith("/v1/") || pathname === "/healthz") return false;
   216→
   217→  let relativePath;
   218→  try {
   219→    relativePath = pathname === "/" ? "index.html" : decodeURIComponent(pathname).replace(/^[/\\]+/, "");
   220→  } catch {
   221→    sendError(res, "Invalid path", "invalid_request", 400);
   222→    return true;
   223→  }
   224→  const hasExtension = path.extname(relativePath) !== "";
   225→  const filePath = path.resolve(PUBLIC_DIR, relativePath);
   226→  const publicPrefix = `${path.resolve(PUBLIC_DIR)}${path.sep}`;
   227→  if (filePath !== path.resolve(PUBLIC_DIR) && !filePath.startsWith(publicPrefix)) {
   228→    sendError(res, "Invalid path", "invalid_request", 400);
   229→    return true;
   230→  }
   231→
   232→  let selectedPath = filePath;
   233→  try {
   234→    if (!fs.statSync(selectedPath).isFile()) throw new Error("not a file");
   235→  } catch {
   236→    if (hasExtension) return false;
   237→    selectedPath = path.join(PUBLIC_DIR, "index.html");
   238→    try {
   239→      if (!fs.statSync(selectedPath).isFile()) return false;
   240→    } catch {
   241→      return false;
   242→    }
   243→  }
   244→
   245→  try {
   246→    const content = fs.readFileSync(selectedPath);
   247→    const extension = path.extname(selectedPath).toLowerCase();
   248→    res.writeHead(200, {
   249→      "Content-Type": MIME_TYPES[extension] || "application/octet-stream",
   250→      "Content-Length": content.length,
   251→      "Cache-Control": extension === ".html" ? "no-cache" : "public, max-age=3600",
   252→    });
   253→    if (req.method !== "HEAD") res.end(content); else res.end();
   254→  } catch {
   255→    sendError(res, "Unable to load dashboard", "api_error", 500);
   256→  }
   257→  return true;
   258→}

   279→function dashboardOverview() {
   280→  const records = dashboardRecords();
   281→  const usage = { requests: records.length, inputTokens: 0, outputTokens: 0, totalTokens: 0, successRate: 0, avgLatencyMs: 0 };
   282→  let successful = 0;
   283→  let totalLatency = 0;
   284→  const clients = new Map();
   285→  const modelUsage = new Map();
   286→  const now = Date.now();
   287→  const activity = Array.from({ length: 12 }, (_, index) => {
   288→    const start = now - (12 - index) * 2 * 60 * 60 * 1000;
   289→    return { start, end: start + 2 * 60 * 60 * 1000, requests: 0 };
   290→  });
   291→
   292→  for (const entry of records) {
   293→    const inputTokens = numberOrZero(entry.inputTokens);
   294→    const outputTokens = numberOrZero(entry.outputTokens);
   295→    const totalTokens = numberOrZero(entry.totalTokens) || inputTokens + outputTokens;
   296→    const status = numberOrZero(entry.status);
   297→    usage.inputTokens += inputTokens;
   298→    usage.outputTokens += outputTokens;
   299→    usage.totalTokens += totalTokens;
   300→    totalLatency += numberOrZero(entry.latencyMs);
   301→    if (status >= 200 && status < 400) successful += 1;
   302→
   303→    const clientId = entry.client || "local";
   304→    if (!clients.has(clientId)) clients.set(clientId, { id: clientId, requests: 0, totalTokens: 0, errors: 0, lastSeen: entry.ts });
   305→    const client = clients.get(clientId);
   306→    client.requests += 1;
   307→    client.totalTokens += totalTokens;
   308→    if (status >= 400 || status === 0) client.errors += 1;
   309→    if (new Date(entry.ts).getTime() > new Date(client.lastSeen).getTime()) client.lastSeen = entry.ts;
   310→
   311→    const key = entry.model || "Modelo não informado";
   312→    if (!modelUsage.has(key)) modelUsage.set(key, { id: key, name: key, requests: 0, totalTokens: 0 });
   313→    const model = modelUsage.get(key);
   314→    model.requests += 1;
   315→    model.totalTokens += totalTokens;
   316→
   317→    const timestamp = new Date(entry.ts).getTime();
   318→    const bucket = activity.find((point) => timestamp >= point.start && timestamp < point.end);
   319→    if (bucket) bucket.requests += 1;
   320→  }
   321→
   322→  usage.successRate = usage.requests ? successful / usage.requests : 0;
   323→  usage.avgLatencyMs = usage.requests ? Math.round(totalLatency / usage.requests) : 0;
   324→  const clientRows = [...clients.values()]
   325→    .map((client) => ({ ...client, errorRate: client.requests ? client.errors / client.requests : 0 }))
   326→    .sort((a, b) => new Date(b.lastSeen) - new Date(a.lastSeen));
   327→
   328→  let tokenOk = true;
   329→  let tokenError = null;
   330→  try { getToken(); } catch (error) { tokenOk = false; tokenError = error.message; }
   331→  const baseUrl = `${proxyBaseUrl()}/v1`;
   332→  const codex = getCodexStatus();
   333→  const opencode = getOpenCodeStatus();
   334→  const claude = getClaudeCodeStatus();
   335→  return {
   336→    health: {
   337→      ok: tokenOk,
   338→      status: tokenOk ? "Proxy online" : "Aguardando AutoClaw",
   339→      primaryReady: tokenOk,
   340→      ...(tokenError ? { error: tokenError } : {}),
   341→    },
   342→    endpoint: {
   343→      baseUrl,
   344→      responsesUrl: `${baseUrl}/responses`,
   345→      anthropicUrl: `${proxyBaseUrl(18792)}/v1/messages`,
   346→    },
   347→    rateLimit: rateLimitSnapshot(),
   348→    fallback: {
   349→      ready: openRouter.status().configured,
   350→      mode: openRouter.status().configured ? "openrouter" : "not-configured",
   351→      model: openRouter.status().model,
   352→      url: "https://openrouter.ai/keys",
   353→    },
   354→    usage,
   355→    models: MODELS.map((model) => ({ id: model.id, name: model.name, contextWindow: model.contextWindow, maxTokens: model.maxTokens })),
   356→    clients: clientRows.slice(0, 20),
   357→    modelUsage: [...modelUsage.values()].sort((a, b) => b.requests - a.requests),
   358→    activity: activity.map((point) => ({
   359→      label: new Intl.DateTimeFormat("pt-BR", { hour: "2-digit" }).format(new Date(point.start)),
   360→      requests: point.requests,
   361→    })),
   362→    codex,
   363→    opencode,
   364→    claude,
   365→  };
   366→}

   381→function parseJsonc(source) {
   382→  let stripped = "";
   383→  let inString = false;
   384→  let escaped = false;
   385→  let lineComment = false;
   386→  let blockComment = false;
   387→  for (let index = 0; index < source.length; index += 1) {
   388→    const char = source[index];
   389→    const next = source[index + 1];
   390→    if (lineComment) {
   391→      if (char === "\n") { lineComment = false; stripped += char; }
   392→      continue;
   393→    }
   394→    if (blockComment) {
   395→      if (char === "*" && next === "/") { blockComment = false; index += 1; }
   396→      else if (char === "\n") stripped += "\n";
   397→      continue;
   398→    }
   399→    if (!inString && char === "/" && next === "/") { lineComment = true; index += 1; continue; }
   400→    if (!inString && char === "/" && next === "*") { blockComment = true; index += 1; continue; }
   401→    stripped += char;
   402→    if (inString) {
   403→      if (escaped) escaped = false;
   404→      else if (char === "\\") escaped = true;
   405→      else if (char === '"') inString = false;
   406→    } else if (char === '"') inString = true;
   407→  }
   408→
   409→  let normalized = "";
   410→  inString = false;
   411→  escaped = false;
   412→  for (let index = 0; index < stripped.length; index += 1) {
   413→    const char = stripped[index];
   414→    if (inString) {
   415→      normalized += char;
   416→      if (escaped) escaped = false;
   417→      else if (char === "\\") escaped = true;
   418→      else if (char === '"') inString = false;
   419→      continue;
   420→    }
   421→    if (char === '"') { inString = true; normalized += char; continue; }
   422→    if (char === ",") {
   423→      let lookahead = index + 1;
   424→      while (/\s/.test(stripped[lookahead] || "")) lookahead += 1;
   425→      if (stripped[lookahead] === "}" || stripped[lookahead] === "]") continue;
   426→    }
   427→    normalized += char;
   428→  }
   429→  return normalized.trim() ? JSON.parse(normalized) : {};
   430→}

   432→function readJsonLike(file) {
   433→  try {
   434→    const source = fs.readFileSync(file, "utf8");
   435→    return { source, value: parseJsonc(source) };
   436→  } catch (error) {
   437→    if (error.code === "ENOENT") return { source: "", value: {} };
   438→    throw new Error(`Não foi possível ler ${file}: ${error.message}`);
   439→  }
   440→}

   591→function updateRootTomlValues(source, values) {
   592→  const newline = source.includes("\r\n") ? "\r\n" : "\n";
   593→  const lines = source ? source.split(/\r?\n/) : [];
   594→  const firstSection = lines.findIndex((line) => /^\s*\[/.test(line));
   595→  const rootEnd = firstSection === -1 ? lines.length : firstSection;
   596→  const seen = new Set();
   597→  const keys = Object.keys(values);
   598→  for (let index = 0; index < rootEnd; index += 1) {
   599→    const match = lines[index].match(/^\s*([A-Za-z0-9_-]+)\s*=/);
   600→    if (!match || !Object.hasOwn(values, match[1])) continue;
   601→    lines[index] = `${match[1]} = ${tomlValue(values[match[1]])}`;
   602→    seen.add(match[1]);
   603→  }
   604→  const missing = keys.filter((key) => !seen.has(key)).map((key) => `${key} = ${tomlValue(values[key])}`);
   605→  if (missing.length) lines.splice(rootEnd, 0, ...missing, ...(rootEnd ? [] : [""]));
   606→  return lines.join(newline).replace(/\s*$/, "") + newline;
   607→}

   609→function updateTomlSection(source, header, values, removeKeys = []) {
   610→  const newline = source.includes("\r\n") ? "\r\n" : "\n";
   611→  const lines = source.split(/\r?\n/);
   612→  const start = lines.findIndex((line) => line.trim() === header);
   613→  const canonicalLines = Object.entries(values).map(([key, value]) => `${key} = ${tomlValue(value)}`);
   614→
   615→  if (start === -1) {
   616→    const output = source.replace(/\s*$/, "");
   617→    return `${output}${output ? `${newline}${newline}` : ""}${header}${newline}${canonicalLines.join(newline)}${newline}`;
   618→  }
   619→
   620→  let end = start + 1;
   621→  while (end < lines.length && !/^\s*\[/.test(lines[end])) end += 1;
   622→  const canonicalKeys = new Set([...Object.keys(values), ...removeKeys]);
   623→  const preserved = lines.slice(start + 1, end).filter((line) => {
   624→    const match = line.match(/^\s*([A-Za-z0-9_-]+)\s*=/);
   625→    return !match || !canonicalKeys.has(match[1]);
   626→  });
   627→  lines.splice(start + 1, end - start - 1, ...canonicalLines, ...preserved);
   628→  return lines.join(newline).replace(/\s*$/, "") + newline;
   629→}

   631→function rootTomlValue(source, key) {
   632→  const root = source.split(/\r?\n/).slice(0, (() => {
   633→    const index = source.split(/\r?\n/).findIndex((line) => /^\s*\[/.test(line));
   634→    return index === -1 ? source.split(/\r?\n/).length : index;
   635→  })()).join("\n");
   636→  const match = root.match(new RegExp(`^\\s*${key}\\s*=\\s*"([^"]*)"`, "m"));
   637→  return match?.[1] || null;
   638→}

   640→function tomlSection(source, header) {
   641→  const lines = source.split(/\r?\n/);
   642→  const start = lines.findIndex((line) => line.trim() === header);
   643→  if (start === -1) return "";
   644→  let end = start + 1;
   645→  while (end < lines.length && !/^\s*\[/.test(lines[end])) end += 1;
   646→  return lines.slice(start + 1, end).join("\n");
   647→}

   704→function launchLocalTool(tool) {

   717→    cwd: __dirname,
   718→    env: tool === "opencode"

   726→    detached: true,
   727→    stdio: "ignore",
   728→    windowsHide: false,
   729→  });
   730→  child.unref();

   820→// SSE buffering (OpenAI-specific: assemble streamed chunks into a single response)
   821→function bufferSSE(upstreamRes, modelId) {
   822→  return new Promise((resolve, reject) => {
   823→    let raw = "";
   824→    upstreamRes.on("data",  (c) => (raw += c));
   825→    upstreamRes.on("error", reject);
   826→    upstreamRes.on("end",   () => {
   827→      try {
   828→        let content = "", reasoning = "";
   829→        let id      = `chatcmpl-${generateId()}`;
   830→        let model   = modelId;
   831→        let promptTokens = 0, completionTokens = 0;
   832→        let finishReason = "stop";
   833→        const toolCalls = {};
   834→
   835→        for (const line of raw.split("\n")) {
   836→          if (!line.startsWith("data: ") || line === "data: [DONE]") continue;
   837→          const chunk = JSON.parse(line.slice(6));
   838→          if (chunk.id)    id    = chunk.id;
   839→          if (chunk.model) model = chunk.model;
   840→          const delta = chunk.choices?.[0]?.delta;
   841→          if (delta?.content)          content   += delta.content;
   842→          if (delta?.reasoning_content) reasoning += delta.reasoning_content;
   843→          // Accumulate tool calls
   844→          for (const tc of delta?.tool_calls || []) {
   845→            if (!toolCalls[tc.index]) toolCalls[tc.index] = { id: "", name: "", arguments: "" };
   846→            if (tc.id)                  toolCalls[tc.index].id = tc.id;
   847→            if (tc.function?.name)      toolCalls[tc.index].name = tc.function.name;
   848→            if (tc.function?.arguments) toolCalls[tc.index].arguments += tc.function.arguments;
   849→          }
   850→          const fr = chunk.choices?.[0]?.finish_reason;
   851→          if (fr) finishReason = fr;
   852→          if (chunk.usage) {
   853→            promptTokens     = chunk.usage.prompt_tokens     ?? 0;
   854→            completionTokens = chunk.usage.completion_tokens ?? 0;
   855→          }
   856→        }
   857→
   858→        // Build sorted tool_calls array
   859→        const sortedToolCalls = Object.keys(toolCalls)
   860→          .sort((a, b) => Number(a) - Number(b))
   861→          .map((idx) => {
   862→            const tc = toolCalls[idx];
   863→            return {
   864→              id: tc.id,
   865→              type: "function",
   866→              function: { name: tc.name, arguments: tc.arguments },
   867→            };
   868→          });
   869→
   870→        resolve({
   871→          id,
   872→          object:  "chat.completion",
   873→          created: Math.floor(Date.now() / 1000),
   874→          model,
   875→          choices: [{
   876→            index:         0,
   877→            message:       {
   878→              role:    "assistant",
   879→              content,
   880→              ...(reasoning ? { reasoning_content: reasoning } : {}),
   881→              ...(sortedToolCalls.length ? { tool_calls: sortedToolCalls } : {}),
   882→            },
   883→            finish_reason: finishReason,
   884→          }],
   885→          usage: {
   886→            prompt_tokens:     promptTokens,
   887→            completion_tokens: completionTokens,
   888→            total_tokens:      promptTokens + completionTokens,
   889→          },
   890→        });
   891→      } catch (err) {
   892→        reject(new Error(`Failed to parse upstream SSE: ${err.message}`));
   893→      }
   894→    });
   895→  });
   896→}

   964→        ? message.content.length
   965→        : JSON.stringify(message?.content ?? "").length,
   966→      toolCalls: Array.isArray(message?.tool_calls) ? message.tool_calls.length : 0,
   967→    })),
   968→    tools: Array.isArray(body.tools)
   969→      ? body.tools.map((tool) => tool?.function?.name || tool?.name || tool?.type).slice(0, 64)
   970→      : [],
   971→    toolChoice: typeof body.tool_choice === "object" ? body.tool_choice?.type : body.tool_choice,
   972→    maxTokens: body.max_completion_tokens ?? body.max_tokens,
   973→    streamOptions: body.stream_options,
   974→  }));
   975→
   976→  let upstreamRes;
   977→  let upstreamErrBody = "";
   978→  let upstreamProvider = "autoclaw";
   979→  try {
   980→    const upstream = await runOpenAIUpstream(body, modelId);
   981→    upstreamRes = upstream.response;
   982→    upstreamErrBody = upstream.errorBody;
   983→    upstreamProvider = upstream.provider;
   984→  } catch (err) {
   985→    const status  = err.message.includes("Cannot read AutoClaw token") ? 503 : 502;
   986→    const errType = status === 503 ? "service_unavailable" : "upstream_error";
   987→    logJsonl({ model: modelId, status, ip: requestIp, latencyMs: Date.now() - startTime, error: errType });
   988→    recordProxyRequest({ model: modelId, status, ip: requestIp, startTime, stream, upstream: upstreamProvider });
   989→    return sendError(res, err.message, errType, status);
   990→  }
   991→
   992→  log.debug(`← upstream status=${upstreamRes.statusCode}`);
   993→
   994→  // Save request details + status to file (not terminal)
   995→  const lastMsg = body.messages?.[body.messages.length - 1];
   996→  logRequest({
   997→    timestamp: new Date().toISOString(),
   998→    model: modelId,
   999→    status: upstreamRes.statusCode,
  1000→    last_message: typeof lastMsg?.content === "string"
  1001→      ? lastMsg.content.substring(0, 300)
  1002→      : JSON.stringify(lastMsg?.content).substring(0, 300),
  1003→    message_count: body.messages?.length || 0,
  1004→  });
  1005→
  1006→  logJsonl({ model: modelId, status: upstreamRes.statusCode, ip: requestIp, latencyMs: Date.now() - startTime });
  1007→
  1008→  // 401 → invalidate cached token so next request gets a fresh one
  1009→  if (upstreamRes.statusCode === 401 && upstreamProvider === "autoclaw") {
  1010→    invalidateToken();
  1011→    recordProxyRequest({ model: modelId, status: 401, ip: requestIp, startTime, stream, upstream: upstreamProvider });
  1012→    return sendError(res,
  1013→      "AutoClaw token expired — invalidated cache, retry the request",
  1014→      "authentication_error", 401
  1015→    );
  1016→  }
  1017→
  1018→  // Any other upstream error → pass body through
  1019→  if (upstreamRes.statusCode >= 400) {
  1020→    const errBody = upstreamErrBody || await collectResponse(upstreamRes);
  1021→    recordProxyRequest({ model: modelId, status: upstreamRes.statusCode, ip: requestIp, startTime, stream, upstream: upstreamProvider });
  1022→    try {
  1023→      const parsed = JSON.parse(errBody);
  1024→      log.error(`Upstream error ${upstreamRes.statusCode}:`, parsed.error?.message || errBody);
  1025→      sendJSON(res, parsed, upstreamRes.statusCode);
  1026→    } catch {
  1027→      // Response wasn't JSON (e.g., nginx HTML error like 413)
  1028→      const cleanMsg = errBody.match(/<title>(.*?)<\/title>/i)?.[1] || errBody || "Upstream error";
  1029→      log.error(`Upstream error ${upstreamRes.statusCode}:`, cleanMsg);
  1030→      sendError(res, cleanMsg, "api_error", upstreamRes.statusCode);
  1031→    }
  1032→    return;
  1033→  }
  1034→
  1035→  if (stream) {
  1036→    // Pipe SSE straight through to the client
  1037→    res.writeHead(200, {
  1038→      "Content-Type":    "text/event-stream",
  1039→      "Cache-Control":   "no-cache",
  1040→      "Connection":      "keep-alive",
  1041→      "X-Accel-Buffering": "no",
  1042→    });
  1043→    const tracker = createUsageTracker((usage) => {
  1044→      recordProxyRequest({ model: modelId, status: 200, ip: requestIp, startTime, usage, stream: true, upstream: upstreamProvider });
  1045→    });
  1046→    upstreamRes.on("data", (chunk) => tracker.push(chunk));
  1047→    upstreamRes.on("end", tracker.finish);
  1048→    upstreamRes.on("error", () => {
  1049→      recordProxyRequest({ model: modelId, status: 502, ip: requestIp, startTime, stream: true, upstream: upstreamProvider });
  1050→    });
  1051→    upstreamRes.pipe(res);
  1052→    return;
  1053→  }
  1054→
  1055→  // Non-stream: buffer SSE, assemble full response object
  1056→  try {
  1057→    const response = await bufferSSE(upstreamRes, modelId);
  1058→    sendJSON(res, response);
  1059→    recordProxyRequest({ model: modelId, status: 200, ip: requestIp, startTime, usage: response.usage, upstream: upstreamProvider });
  1060→  } catch (err) {
  1061→    recordProxyRequest({ model: modelId, status: 502, ip: requestIp, startTime, stream, upstream: upstreamProvider });
  1062→    sendError(res, err.message, "api_error", 502);
  1063→  }
  1064→}

  1074→  }

  1083→  if (previousMessages === null) {

  1092→

  1094→  log.info(`responses model=${modelId} upstream=${route.upstreamModelId} body=${route.bodyModelId} stream=${stream}`);

  1096→  let upstreamErrBody = "";

  1100→    upstreamRes = upstream.response;
  1101→    upstreamErrBody = upstream.errorBody;
  1102→    upstreamProvider = upstream.provider;
  1103→  } catch (error) {
  1104→    const status = error.message.includes("Cannot read AutoClaw token") ? 503 : 502;
  1105→    const errorType = status === 503 ? "service_unavailable" : "upstream_error";
  1106→    logJsonl({ model: modelId, status, ip: requestIp, latencyMs: Date.now() - startTime, error: errorType });
  1107→    recordProxyRequest({ protocol: "responses", model: modelId, status, ip: requestIp, startTime, stream, upstream: upstreamProvider });
  1108→    return sendError(res, error.message, errorType, status);
  1109→  }
  1110→
  1111→  const lastMessage = chatBody.messages?.[chatBody.messages.length - 1];
  1112→  logRequest({
  1113→    timestamp: new Date().toISOString(),
  1114→    model: modelId,
  1115→    status: upstreamRes.statusCode,
  1116→    last_message: typeof lastMessage?.content === "string"
  1117→      ? lastMessage.content.substring(0, 300)
  1118→      : JSON.stringify(lastMessage?.content).substring(0, 300),
  1119→    message_count: chatBody.messages?.length || 0,
  1120→  });
  1121→  logJsonl({ model: modelId, status: upstreamRes.statusCode, ip: requestIp, latencyMs: Date.now() - startTime });
  1122→
  1123→  if (upstreamRes.statusCode === 401 && upstreamProvider === "autoclaw") {
  1124→    invalidateToken();
  1125→    recordProxyRequest({ protocol: "responses", model: modelId, status: 401, ip: requestIp, startTime, stream, upstream: upstreamProvider });
  1126→    return sendError(res, "AutoClaw token expired — invalidated cache, retry the request", "authentication_error", 401);
  1127→  }
  1128→  if (upstreamRes.statusCode >= 400) {
  1129→    const errBody = upstreamErrBody || await collectResponse(upstreamRes);
  1130→    recordProxyRequest({ protocol: "responses", model: modelId, status: upstreamRes.statusCode, ip: requestIp, startTime, stream, upstream: upstreamProvider });
  1131→    try {
  1132→      sendJSON(res, JSON.parse(errBody), upstreamRes.statusCode);
  1133→    } catch {
  1134→      const cleanMessage = errBody.match(/<title>(.*?)<\/title>/i)?.[1] || errBody || "Upstream error";
  1135→      sendError(res, cleanMessage, "api_error", upstreamRes.statusCode);
  1136→    }
  1137→    return;
  1138→  }
  1139→
  1140→  if (!stream) {
  1141→    try {
  1142→      const raw = await collectResponse(upstreamRes);
  1143→      const response = createResponseFromChat({ model: modelId, result: collectChatSse(raw) });
  1144→      sendJSON(res, response);
  1145→      rememberResponse(response, chatBody.messages, requestIp);
  1146→      recordProxyRequest({ protocol: "responses", model: modelId, status: 200, ip: requestIp, startTime, usage: response.usage, upstream: upstreamProvider });
  1147→    } catch (error) {
  1148→      recordProxyRequest({ protocol: "responses", model: modelId, status: 502, ip: requestIp, startTime, stream, upstream: upstreamProvider });
  1149→      sendError(res, `Failed to parse upstream response: ${error.message}`, "api_error", 502);
  1150→    }
  1151→    return;
  1152→  }
  1153→
  1154→  res.writeHead(200, {
  1155→    "Content-Type": "text/event-stream",
  1156→    "Cache-Control": "no-cache",
  1157→    "Connection": "keep-alive",
  1158→    "X-Accel-Buffering": "no",
  1159→  });

  1162→  let finished = false;
  1163→  const complete = () => {
  1164→    if (finished) return;
  1165→    finished = true;
  1166→    if (buffer.trim()) {
  1167→      const finalChunk = parseChatSseLine(buffer.trim());
  1168→      if (finalChunk) writer.consume(finalChunk);
  1169→    }
  1170→    const response = writer.complete();
  1171→    rememberResponse(response, chatBody.messages, requestIp);
  1172→    recordProxyRequest({ protocol: "responses", model: modelId, status: 200, ip: requestIp, startTime, usage: response.usage, stream: true, upstream: upstreamProvider });
  1173→    res.end();
  1174→  };
  1175→  upstreamRes.on("data", (chunk) => {
  1176→    buffer += chunk.toString();
  1177→    const lines = buffer.split(/\r?\n/);
  1178→    buffer = lines.pop() || "";
  1179→    for (const line of lines) writer.consume(parseChatSseLine(line));
  1180→  });

  1182→  upstreamRes.on("error", (error) => {
  1183→    if (finished) return;
  1184→    finished = true;
  1185→    writer.fail(error.message || "Upstream stream failed");
  1186→    recordProxyRequest({ protocol: "responses", model: modelId, status: 502, ip: requestIp, startTime, stream: true, upstream: upstreamProvider });
  1187→    res.end();
  1188→  });
  1189→  req.on("aborted", () => upstreamRes.destroy());
